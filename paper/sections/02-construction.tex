\section{The construction}\label{sec:construction}

The parameters $X_0$, $\kappa$, $H$, $M_1$, $L_0$, $y$, $x_k$, $\gamma_k$, $\alpha_k$, $\kappa_0$ are those of Section~\ref{sec:params}.

\subsection{The base, the rigid set and the semigroup}
\begin{itemize}
\item \emph{Base.} Below $\Xm:=X_0-H(X_0)/2$ the system is exactly $\N$. Its primes are the rational primes $p<X_0$, placed at
their own values with no jitter. On $[\Xm,X_0)$ it is $\N$ together with the at most $H(X_0)/2+1$ jittered primes that are decided
at slots with $s_j-H(s_j)/2<X_0$ and land there.
\item \emph{Jittered primes.} Every prime placed by the construction (Section~\ref{sec:greedy}) has value $\ge X_0-H(X_0)/2$ and is
called \emph{jittered}.
\item \emph{Elements.} $\NN$ is the free commutative semigroup on all primes, rational and jittered. An element is an exponent
vector, and its value is $n=\prod p^{e_p}$. $N(x)$ counts elements of value $\le x$, with multiplicity. An element is
\emph{composite} if $\sum e_p\ge2$. Its \emph{latest prime} is the prime factor decided last (Section~\ref{sec:greedy}).
\item \emph{The rigid set} $\mathcal D$ is the set of $X_0$-smooth integers, i.e.\ integers all of whose prime factors are $<X_0$.
Write $\Psi(x,X_0):=\#\mathcal D\cap[1,x]$ and $\Psi(I,X_0):=\#\mathcal D\cap I$. Every $d\in\mathcal D$ with $d\ge X_0$ is composite.
\end{itemize}

\subsection{The target}\label{sec:target}
\emph{Oscillations.} For $k\ge1$ put $x_k^+:=x_k+2H(x_k)$ and
\begin{equation}\label{eq:Rk}
  R_k(x):=\alpha_kx^{\theta}\cos(\gamma_k\log x)\,\phi\bigl(\log(x/x_k^+)\bigr),\qquad R:=\sum_kR_k,
\end{equation}
where:
\begin{itemize}
\item $\phi\in C^\infty$, $0\le\phi\le1$, $\phi=0$ on $(-\infty,0]$, $\phi=1$ on $[1,\infty)$ and $\phi'\ge0$. So $R_k\equiv0$ on
$[1,x_k+2H(x_k)]$. The term $2H$ makes Lemma~\ref{lem:adapted} exact: $\gamma_k$ reads primes decided at slots up to
$x_k+0.51H(x_k)$, and $T$ on $[1,u]$ is read by slots $\le u-H(u)$.
\item The \emph{amplitude condition}:
\begin{equation}\label{eq:ampl}
  \alpha_k\gamma_kx_k^{\theta-1}=\kappa'_k/\log x_k,\qquad \kappa'_k:=\kappa_0/\log^2x_k,\quad \kappa_0\in[X_0^{-1},\tfrac14].
\end{equation}
It gives $|R'(x)|\le\kappa_+/\log x$ with $\kappa_+=0.8$ (see \eqref{eq:G1}).
% Logged correction 76 (CORRECTIONS-vs-v3.md): gamma_k is defined for every omega by the rule of Lemma goodheights, from the
% system on [1,x_k) of this section; v3's convention "x_k means x_k^+ in Section 3" also moved this information cutoff to
% x_k^+, where the choice would read slots that depend on gamma_k. The convention is now split.
\item $\gamma_k\in[\Gamma_k,2\Gamma_k]$ is defined for every $\omega$ by the rule of Lemma~\ref{lem:goodheights}, as a measurable
function of the system on $[1,x_k)$ only, and $\alpha_k\in[1,2]$.
\end{itemize}
% Logged correction 76 (R17, C76-1, C76-4): x_k^+ is used throughout Section 3; the range of gamma_k with Gamma_k from x_k is
% included among the statements that hold up to the factor 1 + 2H(x_k)/x_k.
In Section~\ref{sec:zeros}, $x_k$ denotes $x_k^+$ throughout (in particular in $R_k$, $\widehat R_k$, $\Gamma_k$, $\alpha_k$, $\delta_k$, $M_k$
and $s^*_k$), and $a_k$ denotes
the scale $x_k$ of this section, which is the information cutoff of the choice of $\gamma_k$: that choice reads only the system on
$[1,a_k)$. There $\alpha_k$ is defined from $x_k^+$, so \eqref{eq:ampl} holds exactly with $x_k^+$ in place of $x_k$, and so do the
bounds derived from it (the proof of \eqref{eq:G1} below, and \eqref{eq:Rkprime}), since $R_k$ vanishes below $x_k^+$. Read with $x_k$,
\eqref{eq:ampl}, \eqref{eq:Rkprime} and the range $\gamma_k\in[\Gamma_k,2\Gamma_k]$ with $\Gamma_k$ computed from $x_k$
(Section~\ref{sec:params}) hold up to a factor in $[1,1+2H(x_k)/x_k]$, as $u^{1-\theta}/\log^3u$ increases; every bound built on them
has far more room than this factor.

\emph{The rigid density.} With $\kappa:=X_0^{-5/12+2\eps_\kappa}$, $\eps_\kappa:=0.01$, put
\begin{equation}\label{eq:rhodet}
  \rhodet(u):=\frac{\Psi([u,(1+\kappa)u],X_0)}{\kappa u}\qquad(u\ge X_0).
\end{equation}
This is a linear-window average with linear normalisation. Each rigid $n\ge X_0(1+\kappa)$ contributes a ramp of total mass
$\log(1+\kappa)/\kappa=1-\kappa/2+O(\kappa^2)$ on $[n/(1+\kappa),n]$.

\emph{The absorbed set} (random and adapted). Put $L_*:=10^7$, $h(u):=u^{\theta/2}$ and $y(n):=y(x)=H(x)L_0(x)(\log x)^6$, where
$x$ is the dyadic scale with $n\in[x,2x)$ and $H$, $L_0$ are the parameters of Section~\ref{sec:params}; so $y$ is frozen per
scale. Then
\begin{equation}\label{eq:absorbed}
  \Abs:=\{n\in\NN:\ \log n\ge L_*,\ \text{the latest prime $r$ of $n$ is jittered, and } r<y(n)\},
\end{equation}
with any cofactor. Also put $\rho_{\Abs}(u):=\#(\Abs\cap[u,u+h(u)])/h(u)$.

\emph{The target.} Put $T(x):=\lfloor x\rfloor$ for $x<X_0$, and for $x\ge X_0$
\begin{equation}\label{eq:target}
\begin{gathered}
  T(x):=X_0+\bigl[\Psi(x,X_0)-\Psi(X_0,X_0)\bigr]+\#(\Abs\cap[X_0,x])+\int_{X_0}^xT'_{\mathrm{cont}}(u)\,du,\\
  T'_{\mathrm{cont}}:=1-\rhodet-\bar\rho_{\Abs}+R',\qquad \bar\rho_{\Abs}(u):=\min\bigl(\rho_{\Abs}(u),0.1/\log u\bigr).
\end{gathered}
\end{equation}
% Logged correction 36: X_0 is an even integer, so the jump at X_0 is the floor jump at the rigid integer X_0.
So on $[X_0,\infty)$, $T$ has a unit jump at every $d\in\mathcal D\cap[X_0,\infty)$ and at every $n\in\Abs$, and no other jump. At
$d=X_0$ this is the step from $\lfloor x\rfloor=X_0-1$ to $T(X_0)=X_0$: as $X_0$ is even, it lies in $\mathcal D$, and the term
$\Psi(x,X_0)-\Psi(X_0,X_0)$ counts the $d\in\mathcal D\cap(X_0,x]$.

\begin{remark}[The stopped process]\label{rem:stopped}
% v3: Remark 1.0
Every probabilistic statement in Sections~\ref{sec:zeros}--\ref{sec:queue} concerns the process stopped at the first failure time
$\tau$ of Definition~\ref{def:good}.
\begin{itemize}
\item Before $\tau$, i.e.\ on $\Good(<u)$, the event of Lemma~\ref{lem:lemmaP} for the window $[u,u+h(u)]$ holds, since that
lemma needs only $\Good(<t_u)$, and $t_u=9u/(\log u)^{11}+H(u)<u$.
% Logged correction 66: v3 attributes this to Lemma adapted; the reason is t_u < u in Lemma lemmaP.
So $\rho_{\Abs}(u)\le(\log u)^{-4}<0.1/\log u$ and $\bar\rho_{\Abs}(u)=\rho_{\Abs}(u)$.
\item The cap can bind only after $\tau$, where nothing is claimed. Its role is to keep $T$, the slots and the construction defined on
every $\omega$.
\item The cap is adapted, being a deterministic function of the adapted $\rho_{\Abs}$.
\end{itemize}
\end{remark}

\begin{lemma}[Adaptedness]\label{lem:adapted}
% v3: Lemma 1.1
$T$ restricted to $[1,u]$ is determined by the decisions and jitters at slots $s_j\le u-H(u)$.
\end{lemma}
\begin{proof}
% Logged correction 67: "y(n) <= n^{o(1)} << u - H(u)" made explicit.
An element $n\le u+h(u)$ of $\Abs$ has latest prime $r<y(n)\le n^{0.01}$: at a scale $x$ where $\Abs\ne\emptyset$ we have
$\log x\ge\Lambda^2e^{s_H}\ge393\Lambda^2$, while $\log y\le22.6\Lambda+9\log\log x$, or $\log2y\le12.1(\log\log x)^2$ in the standard
regime (Section~\ref{sec:params}), and both are $\le0.01\log x$. As $\log H(r)\le8(\log\log r)^2+10^{-6}\log r\le3\cdot10^{-5}\log r$
(Theorem~\ref{thm:caplw}(c), as in the proof of Proposition~\ref{prop:La}), $r$, and every other prime of $n$, is decided at a slot
$\le r+H(r)<2(2u)^{0.01}\le u-H(u)$ (by \eqref{eq:C2} below). The set $\mathcal D$ and the density $\rhodet$ are deterministic, and the
parameters are deterministic functions of the height or of the scale. The oscillation $R$ on $[1,u]$ involves only those $R_k$ with $x_k+2H(x_k)<u$, and each
such $\gamma_k$ reads primes decided at slots up to $x_k+0.51H(x_k)\le u-H(u)$ (Lemma~\ref{lem:goodheights}, with $a_k=x_k$). Hence $\rho_{\Abs}$ on $[X_0,u]$, $R$ on $[1,u]$ and
the jumps of $T$ in $[1,u]$ are known before slot $u$.
\end{proof}

\subsection{Positivity of the target and the target error}
\begin{lemma}[Positivity margin]\label{lem:G1}
% v3: (G1), (1.3')
For every $u\ge X_0$ and every $\omega$,
\begin{equation}\label{eq:G1}
  T'_{\mathrm{cont}}(u)\ge\frac{0.99-0.1-\kappa_+}{\log u}-\frac{1}{\kappa u}=\frac{0.09}{\log u}-\frac1{\kappa u}>0,\qquad\kappa_+=0.8.
\end{equation}
In particular $T$ is non-decreasing on every $\omega$.
\end{lemma}
\begin{proof}
There are three inputs.
\begin{itemize}
\item \emph{Rigid part.} $1-\rhodet(u)\ge0.99/\log u-1/(\kappa u)$. Every window $[u,(1+\kappa)u]$ with $u\ge X_0$ lies in the range of
the short-interval prime number theorem (\eqref{eq:huxley}, Lemma~\ref{lem:margin}(a)), so it contains at least $0.99\kappa u/\log u$
rational primes among its at most $\kappa u+1$ integers. With the linear normalisation \eqref{eq:rhodet} there is no loss of
$\kappa/2$; such a loss belongs to the logarithmically normalised kernel, and a bound of the form $0.99/\log u-\kappa/2$ would be
negative once $\log u>2/\kappa\approx e^{3.97\cdot10^7}$, which is still below $x_1$.
\item \emph{Absorbed part.} $\bar\rho_{\Abs}\le0.1/\log u$, by the cap.
% Logged correction 76: the bound is run at x_k^+, where eq:ampl holds exactly for every omega (alpha_k is defined from x_k^+,
% and the fallback height of Lemma goodheights gives alpha_k = 2).
\item \emph{Oscillations.} $R\equiv0$ below $x_1^+$, and $R_k\equiv0$ below $x_k^+$. For $x\ge x_k^+$, by \eqref{eq:ampl} at $x_k^+$
(see after \eqref{eq:ampl}), for every $\omega$,
\[
  |R_k'(x)|\le\alpha_kx^{\theta-1}(\gamma_k+\theta+\|\phi'\|_\infty)\le2\alpha_k\gamma_kx^{\theta-1}
  =\frac{2\kappa_0(x/x_k^+)^{\theta-1}}{\log^3x_k^+}.
\]
For the last switched-on $k$: if $x\le(x_k^+)^2$ then $\log x\le2\log x_k^+$; otherwise $(x/x_k^+)^{\theta-1}\le x^{-(1-\theta)/2}$. Earlier $k$
contribute less, by the spacing $x_{k+1}\ge\exp(x_k)$. So
$|R'(x)|\le16\kappa_0/\log^3x+4\kappa_0x^{-(1-\theta)/2}\le\kappa_+/\log x$, since $\log x\ge X_0^2$.
\end{itemize}
% Logged correction 67: v3's "1/(kappa u) <= X_0^{-7/12} << 0.09/log u" fails once log u > 0.09 X_0^{7/12}; the bound below holds for all u.
Finally $\kappa u/\log u$ increases on $u\ge X_0$ and is $\ge X_0^{0.6}/\Lambda\ge100$ at $X_0$, so $1/(\kappa u)\le0.01/\log u$ for every
$u\ge X_0$.
\end{proof}

\begin{lemma}[Target error]\label{lem:T1}
% v3: (T1), (1.4)
Write $E(x):=T(x)-x-R(x)$. On the event of Lemma~\ref{lem:lemmaP},
\begin{equation}\label{eq:T1}
  |E(x)|\le C_{X_0}x^{\theta}+2x^{\theta/2}\qquad\text{for all }x\ge X_0,\quad C_{X_0}<\infty.
\end{equation}
Moreover $|R(x)|\le2x^{\theta}\log^*x$, and the rigid part $E_{\mathrm{rig}}(x):=\Psi(x,X_0)-\Psi(X_0,X_0)-\int_{X_0}^x\rhodet$ of $E$
satisfies, for every $x\ge X_0$,
% Logged correction 51
\begin{equation}\label{eq:Erig}
  |E_{\mathrm{rig}}(x)|\le\min\Bigl(\Psi(2x,X_0),\ \kappa X_0+1+\min\bigl(\kappa x+1,\Psi(2x,X_0)\bigr)\Bigr).
\end{equation}
\end{lemma}
\begin{proof}
% Logged correction 51
\emph{Rigid part.} Each rigid $d>X_0$ contributes a unit jump at $d$ and the ramp mass
$\kappa^{-1}\log\bigl(\min(x,d)/\max(X_0,d/(1+\kappa))\bigr)^+$ to $\int_{X_0}^x\rhodet$. So $E_{\mathrm{rig}}(x)=E_0+E_{\mathrm{dr}}-E_{\mathrm{fl}}$, where:
\begin{itemize}
\item $E_0\in[0,\kappa X_0+1]$ comes from the $d\in(X_0,X_0(1+\kappa))$, whose ramps are truncated at $X_0$ (Lemma~\ref{lem:blin}(4));
\item $E_{\mathrm{dr}}\in[0,\tfrac\kappa2\Psi(x,X_0)]$ is the drift: the $d\in[X_0(1+\kappa),x]$ each carry ramp mass
$\log(1+\kappa)/\kappa=1-\kappa/2+O(\kappa^2)$ (Lemma~\ref{lem:blin}(5));
\item $E_{\mathrm{fl}}\in[0,\#(\mathcal D\cap(x,(1+\kappa)x))]$ is the mass in flight.
\end{itemize}
Each rigid $d\le x$ contributes at most $1$ to $E_0+E_{\mathrm{dr}}$, so $E_0+E_{\mathrm{dr}}\le\Psi(x,X_0)$, and
$E_{\mathrm{fl}}\le\Psi(2x,X_0)-\Psi(x,X_0)$. Since $E_0+E_{\mathrm{dr}}$ and $E_{\mathrm{fl}}$ are non-negative,
$|E_{\mathrm{rig}}|\le\max(E_0+E_{\mathrm{dr}},E_{\mathrm{fl}})$. Both $E_{\mathrm{dr}}$ and $E_{\mathrm{fl}}$ are
$\le\min(\kappa x+1,\Psi(2x,X_0))$ (as $\Psi(x,X_0)\le x$), which gives \eqref{eq:Erig}.

% Logged correction 50
Rankin's method gives $\Psi(2x,X_0)\le(2x)^{\theta}C'_{X_0}$, where $C'_{X_0}:=\prod_{p<X_0}(1-p^{-\theta})^{-1}<\infty$, since
$\Psi(y,X_0)\le\sum_{d\in\mathcal D}(y/d)^{\theta}$. So $|E_{\mathrm{rig}}(x)|\le C_{X_0}x^{\theta}$ with $C_{X_0}:=2C'_{X_0}$. At the
scales $\log x\ge X_0^2$ of the zeros the drift $E_{\mathrm{dr}}$ is $\le\kappa x^{0.011}$, since by Rankin
$\Psi(x,X_0)\le x^{0.01}\prod_{p<X_0}(1-p^{-0.01})^{-1}\le x^{0.011}$ there: each factor is at most $(1-2^{-0.01})^{-1}<145<e^5$, so the
product is $\le e^{5\pi(X_0)}\le e^{0.001X_0^2}\le x^{0.001}$.

\emph{Absorbed part.}
% Logged correction 34 (CORRECTIONS-vs-v3.md): v3 omitted the mass deficit 1-M_a of the smoothed elements; added here.
Put $b:=\theta/2$ and, for $a\in\Abs$, let $u_a+h(u_a)=a$ and $M_a:=\int_{u_a}^adu/h(u)$, the mass of the smoothed element $a$.
An element $a$ contributes to $\rho_{\Abs}(u)$ exactly for $u\in[u_a,a]$, so
\[
  \#\Abs(\le x)-\int^x\rho_{\Abs}=\sum_{a\in\Abs,\,a\le x}(1-M_a)-\sum_{a\in\Abs,\,u_a<x<a}\int_{u_a}^x\frac{du}{h(u)} .
\]
The second sum runs over $a\in(x,x+h(x))$, since $a=u_a+h(u_a)<x+h(x)$, and each of its terms is at most $M_a\le1$; so
it is at most $\#(\Abs\cap(x,x+h(x)))\le h(x)(\log x)^{-4}$ by Lemma~\ref{lem:lemmaP}. In the first, $h$ is increasing, so $(u_a/a)^b=h(u_a)/h(a)\le M_a\le1$; with $a-u_a=h(u_a)\le a^b$ this gives
$0\le1-M_a\le-b\log(u_a/a)\le b\,u_a^{b-1}\le1.01\,b\,a^{b-1}$. To sum this, note that
$a^{b-1}\le M_a^{-1}\int_{u_a}^au^{b-1}du/h(u)$ with $M_a^{-1}\le1.001$, and that $\sum_{a\in\Abs}\mathbf 1[u_a\le u\le a]/h(u)=\rho_{\Abs}(u)
\le(\log u)^{-4}$ by Lemma~\ref{lem:rhoA}(a), where $\log u\ge10^7-1$ wherever $\rho_{\Abs}(u)\ne0$ (since $L_*=10^7$). Hence
\[
  \sum_{a\in\Abs,\,a\le x}a^{b-1}\le1.001\int_{e^{10^7-1}}^xu^{b-1}(\log u)^{-4}\,du\le\frac{1.01}{b}\,x^b(\log x)^{-4},
\]
because $u^{b-1}(\log u)^{-4}\le(1-4/(b\log u))^{-1}\frac{d}{du}\bigl(u^b(\log u)^{-4}/b\bigr)$ and $4/(b\log u)\le4\cdot10^{-6}$. So the
first sum is at most $1.01\,b\cdot1.01\,b^{-1}x^{\theta/2}(\log x)^{-4}\le1.03\,x^{\theta/2}(\log x)^{-4}$, and
$|\#\Abs(\le x)-\int^x\rho_{\Abs}|\le3.1\,x^{\theta/2}(\log x)^{-4}$ (there is no boundary term, since $\Abs$ is empty near
$X_0$), well within \eqref{eq:T1}.
% Logged correction 66: only Abs cap (x, x+h(x)) enters the second sum, and v3's +O(1) is superfluous.

\emph{Oscillations.} $|R(x)|\le2x^{\theta}\cdot\#\{k:x_k\le x\}\le2x^{\theta}\log^*x$.
\end{proof}

\subsection{Slots and the greedy rule}\label{sec:greedy}
% Logged correction 36: theta-offset restricted to (0,1/2] (used in Corollary cor:lag).
% Logged correction 77 (CORRECTIONS-vs-v3.md): v3 fixes the offset so that a.s. no jump of T starts on a level j+theta_s, and
% claims only countably many offsets are excluded. Above the onset of Abs or R the jump levels T(d-) are random and depend on
% the offset through the slots, so that claim is unproved; on a tie J_j is empty and s_j = inf J_j is undefined. Now every
% offset is allowed, s_j is a hitting time defined for every omega, and the tie case is an empty continuous slot.
\emph{Slots.} Fix any $\vartheta_{\mathrm s}\in(0,\tfrac12]$. For integers $j\ge X_0$ put $s_j:=\min\{x\ge1:\ T(x)\ge j+\vartheta_{\mathrm s}\}$,
which exists for every $\omega$, as $T$ is non-decreasing, right-continuous and unbounded. \emph{Slot $j$} is
$J_j:=T^{-1}([j+\vartheta_{\mathrm s},j+1+\vartheta_{\mathrm s}))=[s_j,s_{j+1})$, and $T(x)\in[j+\vartheta_{\mathrm s},j+1+\vartheta_{\mathrm s})$ for
$x\in J_j$. The jumps of $T$ are at $d\in\mathcal D$ and $n\in\Abs$. They are unit jumps off the event $\mathcal N_{=}$ that two distinct
elements of $\NN$ have equal values. $\mathcal N_{=}$ is null for every $\vartheta_{\mathrm s}$: after cancelling common factors, the latest-decided
prime of such a pair divides one side only, so it would have to land on a point fixed by the past, which has probability $0$ by the
box-uniform landing law below (and two products of rational primes with disjoint supports differ). The slot facts below are used
off $\mathcal N_{=}$; this costs no probability in Lemma~\ref{lem:budget}. Statements below called sure or pathwise that rest on slot
facts hold for every $\omega\notin\mathcal N_{=}$. Slot $j$ is a \emph{jump slot} if $T(s_j-)<j+\vartheta_{\mathrm s}$, that is, if a unit jump of $T$
reaches the level $j+\vartheta_{\mathrm s}$; that jump sits at $s_j$ and is the only one in $J_j$, as $T$ varies by less than $1$ on $J_j$. So
each unit jump of $T$ in $[s_{X_0},\infty)$ lies in exactly one jump slot. Every other slot is a \emph{continuous slot}: there $T(s_j-)=j+\vartheta_{\mathrm s}$. A
continuous slot is empty, $s_j=s_{j+1}$, exactly when a jump of $T$ starts on the level, $T(d-)=j+\vartheta_{\mathrm s}$; that jump then
lies in the jump slot $j+1$, and $J_j$ holds no element. If $j<j'$ are consecutive continuous slots, then $T(s_j-)=j+\vartheta_{\mathrm s}$, $T(s_{j'}-)=j'+\vartheta_{\mathrm s}$, and the jumps of $T$
in $[s_j,s_{j'})$ are exactly the unit jumps at the starts of the jump slots $j+1,\dots,j'-1$ (a jump that starts on the level
$j+\vartheta_{\mathrm s}$ sits at $s_j=s_{j+1}$ and belongs to slot $j+1$). So $\int_{s_j}^{s_{j'}}T'_{\mathrm{cont}}=1$: between
consecutive continuous-slot starts $T$ gains exactly one unit of continuous mass, and distinct continuous slots have distinct starts. Hence consecutive
continuous-slot starts are at least $1/(1+1/(\kappa u)+\kappa_+/\log u)$ apart at height $u$, since
$T'_{\mathrm{cont}}\le1+1/(\kappa u)+\kappa_+/\log u$. A single continuous slot can be shorter than $1$ where rigid integers are dense,
and has length $0$ when a jump starts on its level. A slot lies in a physical window $I$ when its start $s_j$ lies in $I$; so an
empty slot lies in $I$ exactly when its start does.

\emph{The hard greedy rule with box-centred jitter.} Process $j=X_0,X_0+1,\dots$ in order, keeping $Q_{j-1}\ge0$.
\begin{itemize}
\item $A_j$ is the number of composite elements whose value lies in $J_j$.
\item $P_j:=\max(0,1-Q_{j-1}-A_j)\in\{0,1\}$ and $Q_j:=Q_{j-1}+A_j+P_j-1\ge0$. No prime is placed while the queue is positive, and a
jump slot ($A_j\ge1$) never receives a prime.
\item If $P_j=1$, place one new prime at $s_j+H(s_j)(U_j-\tfrac12)$, where the $U_j$ are i.i.d.\ uniform on $[0,1]$ and independent
of the past. This holds for every slot, including those near $X_0$, so the landing density matches the slot count up to
$O(1/H)$; Section~\ref{sec:noise} uses this.
\item The \emph{jitter width} is
\begin{equation}\label{eq:jitter}
\begin{gathered}
  H(x):=\max\bigl(H_{\mathrm{std}}(x),\bar M_1(x)^2\bigr)\quad(x\ge X_0),\qquad H(x):=H_{\mathrm{std}}(x)\quad(x<X_0),\\
  H_{\mathrm{std}}(x):=\exp\bigl(8(\log\log x)^2\bigr),
\end{gathered}
\end{equation}
with the smooth majorant $\bar M_1\ge M_1$ of Section~\ref{sec:params}. So $H$ is a function of the height itself, not frozen per
scale; it is continuous and non-decreasing. The definition below $X_0$ makes $H$ continuous at $X_0$ and equal to
$H_{\mathrm{std}}$ on $[\Xm/2,X_0^2]$. \emph{Box-centred} means that the landing law of a prime decided at $s_j$ is uniform on
$[s_j-H(s_j)/2,s_j+H(s_j)/2)$.
\item \emph{Decision order.} $t(p)$ is the slot at which $p$ is decided, with $t(p):=p$ for the rational primes $p<X_0$. For an
element $w$, $\tau_w:=\max_{p\mid w}t(p)$ is the decision time of its latest prime.
\item $\mathcal F_t$ is the $\sigma$-field of all decisions and offsets at slots $\le t$.
\end{itemize}

\begin{lemma}[Causality]\label{lem:C2}
% v3: (C2)
$A_j$ is $\mathcal F_{s_j-}$-measurable:
\begin{equation}\label{eq:C2}
  A_j\in\mathcal F_{s_j-}.
\end{equation}
\end{lemma}
\begin{proof}
Let $n\in J_j$ be composite. It has a prime factor $p\le n/2<s_{j+1}/2$, which was decided at a slot $j'$ with
$s_{j'}-H(s_{j'})/2\le p<s_{j+1}/2<s_j-H(s_j)/2$, once $s_j>2H(s_j)+4$; as $v-H(v)/2$ increases, $s_{j'}<s_j$. Conversely, a prime
decided at a slot $\ge j$ lands at $\ge s_j-H(s_j)/2>s_{j+1}/2$, so it is not a factor of any composite in $J_j$.
\end{proof}

\subsection{Bookkeeping}
\begin{lemma}[Count identity]\label{lem:count}
% v3: Lemma 1.2
% Logged correction 36 (CORRECTIONS-vs-v3.md): v3 states V_j = j + Q_j; the initial value (offset 1, X_0 even) was not checked.
Let $j_0:=X_0$ be the first slot (recall that $X_0$ is an even integer) and $Q_{j_0-1}:=0$. For $j\ge j_0-1$ put
\[
\begin{aligned}
  V_j:={}&\#(\text{composites of value }<s_{j+1})\\
  &+\#(\text{primes decided at slots }\le j,\text{ including the rational primes }<X_0)+1.
\end{aligned}
\]
Then $V_j=j+1+Q_j$.
\end{lemma}
\begin{proof}
$V$ increases by $A_j+P_j$ at step $j$, and $Q_j=Q_{j-1}+A_j+P_j-1$, so $V_j-j-Q_j$ does not depend on $j$. It remains to
compute it at $j=j_0-1$, where no slot has been processed and $V_{j_0-1}$ counts $1$, the composites of value $<s_{j_0}$ and the
rational primes $<X_0$. By \eqref{eq:G1}, $T$ rises by $2$ within $30\log X_0$ of $X_0$, so $s_{j_0}<X_0+30\log X_0<2\Xm$. Hence
every composite of value $<s_{j_0}$ is an integer (an element with a jittered prime factor has value $\ge2\Xm$), and jittered
primes do not enter $V_{j_0-1}$. The composite integers in $[X_0,s_{j_0})$ are exactly $X_0$ itself: $T(X_0)=X_0<X_0+\vartheta_{\mathrm s}$,
so $X_0<s_{j_0}$, and $X_0$ is composite (it is even); a composite $n\in(X_0,s_{j_0})$ would be rigid (as $n<2X_0$), and its unit jump
would give $T(n)\ge X_0+1>j_0+\vartheta_{\mathrm s}$, which is impossible for $n<s_{j_0}$. So $V_{j_0-1}=\#(\N\cap[1,X_0))+1=j_0$.
\end{proof}

\begin{lemma}[Lindley]\label{lem:lindley}
% v3: Lemma 1.3
$Q_j=\max_{j_0-1\le i\le j}\bigl[\sum_{l=i+1}^jA_l-(j-i)\bigr]$, with $j_0$ and $Q_{j_0-1}=0$ as in Lemma~\ref{lem:count}.
In words, the queue is the largest excess of composite arrivals over slots,
taken over slot windows ending at $j$.
\end{lemma}
\begin{proof}
This is the standard solution of the recursion $Q_j=\max(0,Q_{j-1}+A_j-1)$, which is equivalent to the rule above.
\end{proof}

\begin{corollary}[Two-sided lag]\label{cor:lag}
% v3: Corollary 1.4, (1.5)
With $D:=N-T$, for $x\ge X_0+2H(x)$,
\begin{equation}\label{eq:lag}
  -\tfrac{H(x)}2\Bigl(1+\frac{\kappa_+}{\log x}\Bigr)-4\le D(x)\le\max_{s_j\le x}Q_j+\tfrac{H(x)}2\Bigl(1+\frac{\kappa_+}{\log x}\Bigr)+3.
\end{equation}
\end{corollary}
\begin{proof}
% Logged correction 36: exact accounting, including the offset 1 of Lemma count, the rounding j - T(x) and the arrivals inside slot j.
% Logged correction 77: x lies in exactly one (non-empty) slot; the continuous-slot spacing is the one of Section sec:greedy, which
% now covers a jump that starts on a level.
Let $j$ be the slot with $x\in J_j$, so $s_j\le x<s_{j+1}$ and $T(x)\in[j+\vartheta_{\mathrm s},j+1+\vartheta_{\mathrm s})$. Let $E_+$ be the number of
primes decided at slots $>j$ that land in $[1,x]$, let $E_-$ be the number decided at slots $\le j$ that land in $(x,\infty)$, and
let $C((x,s_{j+1}))$ be the number of composites of value in $(x,s_{j+1})$. The rational primes are decided before slot $j_0$ and
lie below $x$. So $N(x)=V_j-C((x,s_{j+1}))+E_+-E_-$, and Lemma~\ref{lem:count} gives
\[
  D(x)=\bigl(j-T(x)\bigr)+Q_j+1-C((x,s_{j+1}))+E_+-E_- .
\]
The terms are bounded as follows.
\begin{itemize}
\item $j-T(x)\in(-1-\vartheta_{\mathrm s},-\vartheta_{\mathrm s}]$.
\item $C((x,s_{j+1}))\le A_j$, and $Q_j-A_j=Q_{j-1}+P_j-1\ge-1$.
\item A prime decided at slot $j'$ lands within $H(s_{j'})/2$ of $s_{j'}$, and only continuous slots receive primes. The slots
counted by $E_-$ therefore start in $(x-H(x)/2,x]$, since $H$ is non-decreasing. Those counted by $E_+$ start in
$(x,x+H(s_{j'})/2]$, and $H(s_{j'})\le H(x)+10^{-6}$ by Theorem~\ref{thm:caplw}(c), since $s_{j'}-x\le x^{0.1}$. Between
consecutive continuous-slot starts $T$ gains exactly one unit of continuous mass (\S\ref{sec:greedy}, also when a jump starts on a
level); an interval $I$ therefore contains at most $\int_IT'_{\mathrm{cont}}+1$ of them. With
$T'_{\mathrm{cont}}\le1+1/(\kappa u)+\kappa_+/\log u$ this gives
$E_\pm\le\frac{H(x)}2(1+\kappa_+/\log x)+1+\epsilon$, where $\epsilon<10^{-6}$ covers the term $H/(\kappa x)$ and the variation
of $H$ and $\log u$ across the window. (Where $R'>0$ the slot density exceeds $1$ by up to $\kappa_+/\log x$, so $H/2+O(1)$ would
not suffice.)
\end{itemize}
Hence $D(x)\le Q_j+(1-\vartheta_{\mathrm s})+E_+<\max_{s_j\le x}Q_j+\frac{H(x)}2(1+\kappa_+/\log x)+2+\epsilon$, and
$D(x)>-(1+\vartheta_{\mathrm s})-1+1-E_-\ge-\frac{H(x)}2(1+\kappa_+/\log x)-2-\vartheta_{\mathrm s}-\epsilon$, which is
$\ge-\frac{H(x)}2(1+\kappa_+/\log x)-3$ because $\vartheta_{\mathrm s}\le\frac12$; in particular the lower bound in \eqref{eq:lag} holds.
\end{proof}

\begin{lemma}[Excess identity]\label{lem:excess}
% v3: Lemma 1.5
Let $C_{\mathrm{jit}}(W)$ be the number of composites in $W$ whose latest prime is jittered and which are not in $\Abs$. For every slot
window $W$,
\[
  C(W)-\#\mathrm{slots}(W)=C_{\mathrm{jit}}(W)-\int_WT'_{\mathrm{cont}}\pm2.
\]
\end{lemma}
\begin{proof}
The rigid set and $\Abs$ drop out identically, because each of their elements sits on its own jump slot; the continuous slots in $W$
number $\int_WT'_{\mathrm{cont}}\pm2$. No smooth-number or Tijdeman-type input enters.
\end{proof}

\emph{Reduction.} If $C(W)-\#\mathrm{slots}(W)\le B$ for every slot window $W\subset[X_0,x]$, then $D\le B+\frac H2(1+\kappa_+/\log x)+3$ on $[X_0+2H,x]$.
Everything therefore reduces to the queue bound of Section~\ref{sec:queue} (Theorem~\ref{thm:queue}), whose drift input is
Section~\ref{sec:noise}.
