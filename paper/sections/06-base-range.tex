\section{The base range: the linear-window target}\label{sec:baserange}

This section supplies the source term of the margin in Section~\ref{sec:noise}: the rigid part of the target, its short-interval
prime-number input, and the decomposition of the sharp reference density. The margin itself and its noise bound are proved in
Section~\ref{sec:noise}.

\emph{The design.} This is the hard greedy rule and the target of Section~\ref{sec:target}, with the rigid set averaged over the
\emph{linear} window $[u,(1+\kappa)u]$ and normalised by $1/(\kappa u)$, where $\kappa=X_0^{-5/12+2\eps_\kappa}$. For a measure $h$ on
$[1,\infty)$ and $w>0$ write
\begin{equation}\label{eq:MA}
  \mathrm{MA}_w[h](x):=\frac{h([x,x+w])}{w},
\end{equation}
so that $\rhodet(u)=\mathrm{MA}_{\kappa u}[\mathcal D](u)$, and write $h_{(m)}$ for the dilate of $h$ by $m$, i.e.\ $h_{(m)}(A)=h(A/m)$.

\begin{lemma}[Properties of the linear-window target]\label{lem:blin}
% v3: Lemma 5.1
\leavevmode
\begin{enumerate}[label=(\arabic*)]
\item \emph{Excess identity.} Rigid integers sit on unit jumps of $T$, so Lemma~\ref{lem:excess} holds verbatim. Also
$\rhodet\le1+1/(\kappa u)$, and $1-\rhodet\ge0.99/\log u-1/(\kappa u)$ by \eqref{eq:huxley}.
\item \emph{Dilation commutation.} For any point measure $h$ and $m\ge1$,
\[
  \mathrm{MA}_{\kappa x}[h_{(m)}](x)=m^{-1}\,\mathrm{MA}_{\kappa q}[h](q),\qquad q=x/m,
\]
because the window is linear. Write the non-rigid integers as $\bigsqcup_{m'}m'\cdot\mathbb P_{\ge\max(P^+(m'),X_0)}$, prime powers
included. The $m'$-component of the target at $x$ equals $1/m'$ times the target's fresh-prime density at $q=x/m'$. So the
mismatch between nested windows vanishes identically, for every $m'$.
\item \emph{Partial windows.} When $P^+(m')\in(q,(1+\kappa)q]$, the component is a partial window. Those $m'=pm$ have
% Logged correction 67: v3's "total weight << kappa/log x, a relative O(kappa)" replaced by an explicit bound (needed only in the
% margin, Lemma margin(b), where it is far below 10^{-3} kappa (log u)^3).
$p\in(\sqrt{x/m},\sqrt{(1+\kappa)x/m}]$, an interval holding at most $\frac\kappa2\sqrt{x/m}+1$ integers. For $q\ge X_0$ this gives
$\sum1/m'\le\sum_{m\le x}0.51\kappa/m\le0.51\kappa(\log x+1)$, and the components have density $\le1.01/\Lambda$, so the correction is
at most $\kappa\log x/\Lambda$.
\item \emph{Edges near $X_0$.} The ramps of integers in $[X_0,X_0(1+\kappa))$ are truncated at $X_0$. This gives $E$ a constant offset
$|E_0|\le\kappa X_0+1$, which changes only $C_{X_0}$. In the margin it is part of $\mu_0$, through $\Delta$ (Lemma~\ref{lem:margin}(b),
Step~1).
\item \emph{Drift.} With the linear normalisation the ramp mass is $1-\kappa/2+O(\kappa^2)$, so $E$ drifts by
$+(\kappa/2)\Psi(x,X_0)$. This is contained in \eqref{eq:T1}, and every consumer uses only $|\text{drift}|\le\kappa x^{0.011}$ at the
scales of the zeros. In the margin this smooth source is part of $\mu_0$, with $|\mu_0(u)|\le2\kappa(\log u)^3$
(Lemma~\ref{lem:margin}(b)).
\end{enumerate}
\end{lemma}
\begin{proof}
(2) $\mathrm{MA}_{\kappa x}[h_{(m)}](x)=h([x,(1+\kappa)x]/m)/(\kappa x)=m^{-1}h([q,(1+\kappa)q])/(\kappa q)$. The other items are immediate
from the definitions, as indicated.
\end{proof}

\subsection{The short-interval prime number theorem}
Every window of the construction lies in the range of the short-interval prime number theorem. Indeed, for every $q\ge X_0$,
\[
  \kappa q\ge q^{7/12+\eps_\kappa}\iff q^{5/12-\eps_\kappa}\ge X_0^{5/12-2\eps_\kappa}\Longleftarrow q\ge X_0,
\]
and the window has length $\le q$. The input used is the asymptotic count of primes in intervals of length between
$q^{7/12+\eps_\kappa}$ and $q$, uniformly.

\emph{Short windows.} Heath-Brown proved \cite[Theorem, (1.2), p.~23]{HeathBrown1988}: for any function $0\le\eps_{\mathrm{HB}}(x)\le\frac1{12}$,
$\pi(x)-\pi(x-y)=(y/\log x)\{1+O(\eps_{\mathrm{HB}}(x)^4)+O((\log\log x/\log x)^4)\}$ uniformly for
$x^{7/12-\eps_{\mathrm{HB}}(x)}\le y\le x/(\log x)^4$. We use
$\eps_{\mathrm{HB}}\equiv0$:
\begin{equation}\label{eq:HB}
  \pi(x)-\pi(x-y)=\frac{y}{\log x}\Bigl\{1+O\Bigl(\Bigl(\frac{\log\log x}{\log x}\Bigr)^4\Bigr)\Bigr\}\qquad\text{uniformly for }
  x^{7/12}\le y\le x/(\log x)^4.
\end{equation}
The asymptotic for $y\ge x^{7/12+\eps_0}$ with fixed $\eps_0>0$ is Huxley's \cite{Huxley1972}, as Heath-Brown credits
\cite[p.~22]{HeathBrown1988}; Heath-Brown removes the $\eps_0$.

\emph{Long windows.} For $y>x/(\log x)^4$, which the windows $\kappa q$ reach once $\log q>\kappa^{-1/4}$, we use the prime number
theorem with the de la Vall\'ee Poussin error, $\pi(t)=\Li(t)+O(t\exp(-c_{\mathrm{dVP}}\sqrt{\log t}))$. Applied at both ends of an interval, it gives
relative error $\le C(\log q)^5\exp(-c_{\mathrm{dVP}}\sqrt{\log q})$.

\emph{The threshold $X_{\mathrm{Hux}}$.} For $q\ge2$ put
\begin{equation}\label{eq:huxley}
\begin{gathered}
  \eta_{\mathrm H}(q):=\sup\Bigl\{\Bigl|\frac{\pi(I)}{\int_I dt/\log t}-1\Bigr|:\ I\subset[q,2q]\text{ an interval, }|I|\ge(2q)^{7/12}\Bigr\},\\
  X_{\mathrm{Hux}}:=\min\bigl\{X\ge e^{10^8}:\ \eta_{\mathrm H}(q)\le(\log q)^{-3}\text{ for all }q\ge X\bigr\}.
\end{gathered}
\end{equation}
Every window $[q,(1+\kappa)q]$ with $q\ge X_0$ is such an $I$. Let $x\le2q$ be the right end of $I$. If $|I|\le x/(\log x)^4$, then
\eqref{eq:HB} applies, since $|I|\ge(2q)^{7/12}\ge x^{7/12}$ and $\int_Idt/\log t=(|I|/\log x)(1+O((\log x)^{-5}))$; otherwise the
long-window bound applies. So
\[
  \eta_{\mathrm H}(q)\le\max\bigl(C_{\mathrm{HB}}(\log\log q/\log q)^4,\ C(\log q)^5e^{-c_{\mathrm{dVP}}\sqrt{\log q}}\bigr)=o((\log q)^{-3}),
\]
and $X_{\mathrm{Hux}}$ is finite. It depends on no parameter of the construction (in particular not on $\kappa$). It is effective in
principle, since every input is effective, but it has not been computed: we do not know of an explicit form of the Huxley or
Heath-Brown asymptotic. Theorem~\ref{thm:A} is an existence statement, so $X_0\ge X_{\mathrm{Hux}}$ is enough. The exponent $3$ is
what Lemma~\ref{lem:margin}(c) needs.

\emph{Consequence used throughout.} For every $q\ge X_0\ge X_{\mathrm{Hux}}$,
$\mathrm{MA}[\mathbb P](q)\log q\ge(1-(\log q)^{-3})(1-\kappa/\log q)\ge0.99$ (Lemma~\ref{lem:margin}(a)). This feeds \eqref{eq:G1},
the bound $1-\rhodet\ge0.99/\log u-1/(\kappa u)$, and the base values of Proposition~\ref{prop:g1} ($0.82$ for $x\le X_0^{1.01}$ and
$0.97$ for $x\ge X_0^{1.01}/2$; Lemma~\ref{lem:margin}(d)).

\begin{remark}[A fully explicit route, not used]
Johnston and Yang \cite[Thm.~1.1 and Table~1, row $X=10^8$]{JohnstonYang2023} prove $|\psi(x)-x|\le2.18\cdot10^{-4694}x$ for
$\log x\ge10^8$; for $\theta(x)$ their Corollary~1.2 gives the same form with $A_1=A+0.01$. With $\kappa\approx200\cdot e^{-10807.6}$
in place of the short-interval input above, this would remove the threshold $X_{\mathrm{Hux}}$. It would not by itself make $X_0$
explicit: the constants of the local prime number theorem are not explicit, and with this $\kappa$ the range
$\log x\le(0.0048/\kappa)^{1/5}\approx e^{2159}$ of the response bound for $\mu_0$ in Proposition~\ref{prop:g1} would no longer cover range (A) of
Proposition~\ref{prop:g1}.
\end{remark}

\subsection{The margin decomposition}

Write the non-rigid rational integers as $\bigsqcup_{m'}m'\cdot\mathbb P_{\ge\max(P^+(m'),X_0)}$, and put
\[
  \mathrm{MA}[\mathbb P](q):=\mathrm{MA}_{\kappa q}[\text{rational primes}\ge X_0](q),
\]
the window density of the new primes. It vanishes for $(1+\kappa)q\le X_0$, it is a partial window for $X_0/(1+\kappa)<q<X_0$, and
for $q\ge X_0$ it is the density of all rational primes in $[q,(1+\kappa)q]$. Let $K_{\mathbb Q}$ be the latest-prime renewal operator
with \emph{rational} cofactors.
% Logged correction 53: K_Q given in component form (independent check R7, C1).
It acts on a window density $h(q)=\mathrm{MA}_{\kappa q}[\nu](q)$ of a signed measure $\nu$ on $[X_0,\infty)$ by components,
\[
  (K_{\mathbb Q}h)(x):=\sum_{m'\in\N,\ m'\ge2}\frac1{m'}\,\mathrm{MA}_{\kappa q}\bigl[\nu|_{[P^+(m'),\infty)}\bigr](q)\Big|_{q=x/m'},
\]
and $K_{\mathbb Q}(h\mathbf 1_{[X_0,\infty)})$ keeps only the components with $x/m'\ge X_0$. For $h=\mathrm{MA}[\mathbb P]$ the component
of $m'$ is $m'^{-1}\mathrm{MA}_{\kappa q}[\text{rational primes}\ge\max(P^+(m'),X_0)](q)$ at $q=x/m'$. Let $K_{\mathrm{pt}}$ be the
sharp operator of Section~\ref{sec:noise}, with realised cofactors and latest prime in decision order (Section~\ref{sec:greedy}).
By dilation commutation (Lemma~\ref{lem:blin}(2)) we have, for $x\ge X_0$,
the exact rational identity
\begin{equation}\label{eq:ratid}
  T'_{\mathrm{cont}}-R'+\bar\rho_{\Abs}=(1+K_{\mathbb Q})\,\mathrm{MA}[\mathbb P]+O(1/(\kappa x)).
\end{equation}
The identity is stated only for $x\ge X_0$, where $T'_{\mathrm{cont}}$ is defined by \eqref{eq:target}; below $X_0$ the target
has $T'_{\mathrm{cont}}=0$, although $\mathrm{MA}[\mathbb P]>0$ on $(X_0/(1+\kappa),X_0)$. That window enters \eqref{eq:ratid} only through
$K_{\mathbb Q}$: a cofactor $N\ge2$ with $x/N\in(X_0/(1+\kappa),X_0)$ contributes the partial window of the integers $Np$, $p\ge X_0$, in
$[x,(1+\kappa)x]$.

Put $\rho^{\mathrm{sh}}:=(1+K_{\mathrm{pt}})^{-1}(T'_{\mathrm{cont}}-R'+\bar\rho_{\Abs})$, a function on $[\Xm,\infty)$ that vanishes on
$[\Xm,X_0)$. Near the cut-off the two operators include the following components.
\begin{itemize}
\item $K_{\mathbb Q}$ (rational system): every cofactor $N$ whose new prime $p\ge X_0$ lies in the window at $q=u/N$; for
$q\in(X_0/(1+\kappa),X_0)$ this is a partial window.
\item $K_{\mathrm{pt}}$ (cut-off $u/w\ge\Xm$): every realised cofactor $w$ with $u/w\ge\Xm$. On $[\Xm,X_0)$ it reads only functions that
vanish there ($\rho^{\mathrm{sh}}$ and the functions in \eqref{eq:shdecomp} below), so its cut-off may be taken to be $X_0$; and it never
reads $(X_0/(1+\kappa),\Xm)$.
\end{itemize}
So on $(X_0/(1+\kappa),X_0)$ only $K_{\mathbb Q}$ contributes, and its contribution there is exactly $\mu_{\mathrm{edge}}$ below. Writing
$T'_{\mathrm{cont}}-R'+\bar\rho_{\Abs}=(1+K_{\mathrm{pt}})(\mathrm{MA}[\mathbb P]\mathbf 1_{[X_0,\infty)})+(K_{\mathbb Q}-K_{\mathrm{pt}})(\mathrm{MA}[\mathbb P]\mathbf 1_{[X_0,\infty)})+\mu_{\mathrm{edge}}+O(1/(\kappa x))$
on $[X_0,\infty)$, and applying $(1+K_{\mathrm{pt}})^{-1}$, gives, for $x\ge X_0$,
\begin{equation}\label{eq:shdecomp}
  \rho^{\mathrm{sh}}=\mathrm{MA}[\mathbb P]+\rho_{\mathrm{edge}}+(1+K_{\mathrm{pt}})^{-1}(\mu_0+\mu_1),\qquad
  \rho_{\mathrm{edge}}:=(1+K_{\mathrm{pt}})^{-1}\mu_{\mathrm{edge}}.
\end{equation}
This $\rho^{\mathrm{sh}}$ is the part of the sharp reference of Section~\ref{sec:noise} that comes from $1-\rhodet$; the responses
to $R'$ and to the absorbed source are carried in Section~\ref{sec:noise}. Here we split
$\mathrm{MA}[\mathbb P]=f_0+f_1$ with $f_0(q):=\mathrm{MA}_{\kappa q}[\mathbf 1_{t\ge X_0}\,dt/\log t](q)$ and $f_1:=\mathrm{MA}[\mathbb P]-f_0$, so
that $|f_1(q)|\le\eta_{\mathrm H}(q)f_0(q)\le\eta_{\mathrm H}(q)/\log q$ for $q\ge X_0$, and we set
\[
\begin{gathered}
  \mu_{\mathrm{edge}}(u):=\sum_{N\ge2}\frac1N\,\mathrm{MA}_{\kappa q}[\text{rational primes}\ge\max(P^+(N),X_0)](q)\Big|_{q=u/N}
  \mathbf 1\Bigl[\frac uN\in\Bigl(\frac{X_0}{1+\kappa},X_0\Bigr)\Bigr],\\
  \mu_1:=(K^{\mathrm{sh}}_{\mathbb Q}-K_{\mathrm{pt}})(f_1\mathbf 1_{[X_0,\infty)}),\\
  \mu_0:=(K^{\mathrm{sh}}_{\mathbb Q}-K_{\mathrm{pt}})(f_0\mathbf 1_{[X_0,\infty)})
  +(K_{\mathbb Q}-K^{\mathrm{sh}}_{\mathbb Q})(\mathrm{MA}[\mathbb P]\mathbf 1_{[X_0,\infty)})+O(1/(\kappa x)).
\end{gathered}
\]
% Logged correction 74 (CORRECTIONS-vs-v3.md): mu_1 used the window-filtered K_Q, so it contained the partial windows of
% nu_1 = (primes - dt/log t) on [X_0,oo), which the proof of (c) does not bound; mu_1 now uses the sharp K_Q^sh, and all partial

Here $K^{\mathrm{sh}}_{\mathbb Q}h(x):=\sum_{m\ge2}m^{-1}\mathbf 1[P^+(m)\le x/m]\,h(x/m)$ is the sharp form of $K_{\mathbb Q}$. The
difference $K_{\mathbb Q}-K^{\mathrm{sh}}_{\mathbb Q}$ consists exactly of the partial windows $P^+(m')\in(q,(1+\kappa)q]$ of
Lemma~\ref{lem:blin}(3), so all partial windows of $\mathrm{MA}[\mathbb P]$ above $X_0$ are in $\mu_0$, and
$\mu_0+\mu_1=(K_{\mathbb Q}-K_{\mathrm{pt}})(\mathrm{MA}[\mathbb P]\mathbf 1_{[X_0,\infty)})+O(1/(\kappa x))$. Write
$K^{\mathrm{val}}_{\mathrm{pt}}$ for $K_{\mathrm{pt}}$ with its filter $\tau_w<y/w$ replaced by $P^+(w)\le y/w$ (value order, with the
non-strict filter of $K^{\mathrm{sh}}_{\mathbb Q}$). Since $K_{\mathrm{pt}}$ uses decision order, $\mu_0$ contains the decision-order mismatch
$(K^{\mathrm{val}}_{\mathrm{pt}}-K_{\mathrm{pt}})(f_0\mathbf 1_{[X_0,\infty)})$, which is bounded in the proof of (b); in $\mu_1$ the operator
$K_{\mathrm{pt}}$ is bounded directly in the proof of (c), so no separate mismatch term arises there. The term $\mu_{\mathrm{edge}}$ is the part of $K_{\mathbb Q}\mathrm{MA}[\mathbb P]$ that lies below $X_0$; for $N<X_0$ its summand is
$\mathrm{MA}[\mathbb P](u/N)/N$. For $u<2X_0$ every cofactor $w\ge2$ has $u/w<X_0$, so $\mu_0=O(1/(\kappa u))$ and $\mu_1=0$ there;
and $\mu_{\mathrm{edge}}(u)=0$ for $u\le2X_0/(1+\kappa)$, while $\mu_{\mathrm{edge}}(u)=\tfrac12\mathrm{MA}[\mathbb P](u/2)$ on
$(2X_0/(1+\kappa),2X_0)$.

\begin{lemma}[The margin decomposition]\label{lem:margin}
% v3: Lemma 5.2
\leavevmode
\begin{enumerate}[label=(\alph*)]
\item \emph{(Huxley part.)} For every $q\ge X_0\ge X_{\mathrm{Hux}}$, $\mathrm{MA}[\mathbb P](q)\log q\ge0.99$, and the left side tends to $1$.
\item \emph{(Smooth part.)} On $\Good(<u)$, for every $u\ge X_0$,
\[
  |\mu_0(u)|\le2\kappa(\log u)^3.
\]
\item \emph{(Huxley-error part.)} Put $v:=\log x/\Lambda$. On $\Good(<x)$, for every $x\ge X_0$,
\[
  |\mu_1(x)|\le2\Lambda^{-3}(\log v+1.01),
\]
and, for $v\ge4$ with $U:=1+2\log v$,
\[
  |\mu_1(x)|\le2\Bigl[\Bigl(\frac U{\log x}\Bigr)^3(\log U+1.01)+2e^{\gamma}e^{\operatorname{Ein}(2)}\frac vUv^{-4}\Lambda^{-3}\Bigr],
  \qquad\operatorname{Ein}(2)<3.69.
\]
Here $\operatorname{Ein}(a):=\int_0^a(e^t-1)\,dt/t$, the convention of Section~\ref{sec:zeros}, so that
$\operatorname{Ein}(2)=3.6838\ldots$ (Appendix~\ref{app:cert-consts}). Consequently
\begin{equation}\label{eq:mu1sum}
  \sum_{u\text{ dyadic}\ge X_0}\log(2u)\sup_{[u,2u]}|\mu_1|\le\frac{690}{\log X_0}\le7\cdot10^{-6}.
\end{equation}
% Logged correction 47: new part (d), the response to the edge term.
\item \emph{(Edge at $X_0$.)} On $\Good(<x)$:
\begin{enumerate}[label=(\roman*)]
\item for $X_0\le x\le X_0^{1.9}$, $\displaystyle\rho_{\mathrm{edge}}(x)=-\sum_{N\ge2:\ x/N\in(X_0/(1+\kappa),\,X_0)}\mu(N)\,
  \frac{\mathrm{MA}[\mathbb P](x/N)}N$;
\item for $X_0\le x\le X_0^{1.01}$, at most one $N$ occurs in (i), and $\rho_{\mathrm{edge}}(x)\log x\ge-0.167$;
\item for $x\ge X_0^{1.01}/2$ with $\log x\le X_0^{1/100}$, $|\rho_{\mathrm{edge}}(x)|\log x\le3X_0^{-0.01}+X_0^{-0.34}$.
\end{enumerate}
\end{enumerate}
\end{lemma}

\begin{proof}[Proof of (a)]
The window $I:=[q,(1+\kappa)q]$ lies in $[q,2q]$ and has $|I|=\kappa q\ge q^{7/12+\eps_\kappa}\ge(2q)^{7/12}$, so it is admissible in
\eqref{eq:huxley}, and $\eta_{\mathrm H}(q)\le(\log q)^{-3}$ there (by \eqref{eq:HB} if $|I|\le x/(\log x)^4$ with $x:=(1+\kappa)q$, and by the de
la Vall\'ee Poussin bound if $|I|$ is longer). Hence $\pi(I)\ge(1-\eta_{\mathrm H}(q))\int_Idt/\log t\ge(1-(\log q)^{-3})\kappa q/\log((1+\kappa)q)$.
Dividing by $\kappa q$,
$\mathrm{MA}[\mathbb P](q)\log q\ge(1-(\log q)^{-3})(1-\kappa/\log q)\ge1-10^{-20}$, since $\log q\ge10^8$ and $\kappa=X_0^{-0.39666\ldots}\le X_0^{-0.3966}$.
% Logged correction 69 (R13, C2): kappa = X_0^{-0.39666...} (was "= X_0^{-0.3967}").
\end{proof}

\begin{proof}[Proof of (b)]
% Logged correction 73 (CORRECTIONS-vs-v3.md): v3 bounds Delta and Delta_psi for all t <= u on Good(<u), where the queue bound
% is certified only up to u - 12 log u - 3. Step 3 needs them only for t <= sqrt(u): each replaced prime is <= sqrt(u/m_j).
\emph{Step 1: the element count.} Put $\Delta:=N_{\NN}-\lfloor\cdot\rfloor$. Then $\Delta=0$ on $[1,\Xm)$, and on $\Good(<u)$, for
$t\le\sqrt u$ (the only heights used in Step~3, and below the certified horizon of $\Good(<u)$, Remark~\ref{rem:horizon}),
\[
  |\Delta(t)|\le\delta(t):=2\kappa t+2t^{\theta}\log^*t+2t^{\theta/2}+B(t)+3H(t)+6.
\]
The pieces are: $|D|\le B+3H+3$ (\eqref{eq:lag} and the count in the proof of Proposition~\ref{prop:La});
$|E|\le|E_0|+\max\bigl((\kappa/2)\Psi(t,X_0),\,\Psi([t,(1+\kappa)t],X_0)\bigr)\le2\kappa t+2$ (\eqref{eq:Erig}; Lemma~\ref{lem:blin}(4),(5): the drift and
the mass still in flight in the top window have opposite signs, and $\Psi([t,(1+\kappa)t],X_0)\le\kappa t+1$); $|R|\le2t^{\theta}\log^*t$;
and the absorbed part of \eqref{eq:T1}.

\emph{Step 2: the prime count.} The Chebyshev identity $\log\cdot dN=d\psi*dN$ holds in both systems. Write $\Delta_\psi:=\psi_{\NN}-\psi$,
which is supported on $[\Xm,\infty)$ because the base primes coincide. Then $d\Delta_\psi*dN_{\NN}=dh$ with
$dh:=\log\cdot d\Delta-d\psi*d\Delta$, and
\[
  h(t)=\Delta(t)\log t-\int_1^t\Delta(v)\frac{dv}v-\int\Delta(t/a)\,d\psi(a).
\]
So $|h(t)|\le5\kappa t\log t$: the $\kappa$-part of $\delta$ gives $4\kappa t\log t+2\kappa t$, and the rest of $\delta$ gives
$\le\kappa t\,X_0^{-0.1}$, using $(B+3H+3)(s)\le s^{3\cdot10^{-5}}$ (proof of Proposition~\ref{prop:La}) and $\theta\le\tfrac12-10^{-4}$.
Invert with the M\"obius measure $\mu_{\NN}$ of $\NN$, using $|\mu_{\NN}|\le dN_{\NN}$ and $\sum_{v\in\NN,\,v\le s}1/v\le\log s+1$ on $\Good$.
This gives
\[
  |\psi_{\NN}(t)-\psi(t)|\le5\kappa t\log^2t\qquad(t\le\sqrt u).
\]

\emph{Step 3: telescoping the cofactors.} We have
$(K^{\mathrm{val}}_{\mathrm{pt}}-K^{\mathrm{sh}}_{\mathbb Q})(f_0\mathbf 1_{[X_0,\infty)})(u)=\sum_{w\in\NN}F(w)\chi_w-\sum_{m\in\N}F(m)\chi_m$,
where $F(v):=f_0(u/v)/v$ and $\chi$ is the latest-prime filter ($P^+\le u/v$, in value order; the decision-order mismatch
$K_{\mathrm{pt}}-K^{\mathrm{val}}_{\mathrm{pt}}$ is treated below). Write each cofactor as
$d\cdot(\text{a multiset of }r=k-1\text{ large primes})$, with $d\in\mathcal D$ (the same set in both systems) and the large primes
jittered and $\ge\Xm$, respectively rational and $\ge X_0$.
% Logged correction 53 (CORRECTIONS-vs-v3.md): symmetric telescoping replaces the ordered one-variable-at-a-time replacement,
% whose intermediate sums mix the two systems with an ordering across them (independent check R7, F1).
\begin{itemize}
\item \emph{Symmetric telescoping.} The sum over multisets of $r$ large primes is $1/r!$ times the sum over ordered $r$-tuples (the
product measure), plus a diagonal term from the multisets with a repeated large prime. The diagonal term has total weight
$\le1.01e^{\gamma}\Lambda\,e^{\ell_u}/(\Xm\Lambda)$ in each system, where $\ell_u$ is defined below. Telescope the product measure: the
$j$-th intermediate sum has its first $j-1$ variables realised and its last $r-j$ rational, with no ordering between them.
\item \emph{One replacement.} Fix the other variables, and put $m_j:=d\prod_{i\ne j}p_i$ and $M'$ for the largest of them (or $P^+(d)$).
The admissible set of the $j$-th variable $p$ is an interval $[\Xm,b]$. The filter $\max(p,M')\,m_jp\le u$ reads
$p\le\sqrt{u/m_j}$ if $M'^2m_j\le u$, and $p\le u/(M'm_j)$ otherwise; the support condition $u/(m_jp)\ge X_0$ of
$f_0\mathbf 1_{[X_0,\infty)}$ reads $p\le u/(X_0m_j)$; and $b$ is the smaller of the two bounds. So $f_0$ is evaluated only on
$[X_0,\infty)$; the partial windows below $X_0$ are in $\mu_{\mathrm{edge}}$, not here. In both cases of the filter
$b\le\sqrt{u/m_j}\le\sqrt u$ (in the second, $u/(M'm_j)<\sqrt{u/m_j}$ as $M'^2m_j>u$), so Step~2 is needed only on
$[1,\sqrt u]$. The summand is $G(p)=f_0(u/(m_jp))/(m_jp)$, and on $[X_0,\infty)$, $|qf_0'(q)|\le f_0(q)/\Lambda$, so
$|(G/\log)'|\le1.1G/(p\log p)$. Abel summation against
$d(\psi_{\NN}-\psi)/\log$ (the prime powers of the large primes cost at most $\sup f_0/(m_j\Xm)$) shows that one replacement changes the sum by at most
$10\kappa f_0\log b/m_j+2.8\kappa f_0\log^2b/m_j\le4\kappa\log^2u/(m_j\Lambda)$.
\item \emph{Summing over the other variables.} They range over a product set, so
$\sum1/m_j\le1.01e^{\gamma}\Lambda\cdot\ell_u^{r-1}$, with $\ell_u:=\sum_{\Xm\le p\le u/X_0}1/p\le\log(\log u/\Lambda)+2.2/\Lambda$ in
