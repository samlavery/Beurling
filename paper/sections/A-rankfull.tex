\section{Rank-full configurations}\label{app:rankfull}
% v3: Appendix A

This appendix proves the moment bound for rank-full clusters of generalized integers in a short window
(Theorem~\ref{thm:R}). It is used in the proof of the cluster event (Proposition~\ref{prop:ecl}). Its machinery
(Lemmas~\ref{lem:volume}--\ref{lem:SC} and Theorem~\ref{thm:svl}) is also used in the Cover Bound
(Proposition~\ref{prop:cover}). Probabilistically, the argument rests only on the standing facts
\ref{eq:sF1}--\ref{eq:sF4} of Section~\ref{sec:queue-standing}:
\begin{itemize}
\item the slot count \ref{eq:sF1};
\item the jitter density \ref{eq:sF2};
\item the harmonic sums \ref{eq:sF3p};
\item the sure Mertens bound \ref{eq:sF4}.
\end{itemize}
These are carried by progressive good events.

The idea is to integrate out the jitter of $k$ suitably chosen primes (the \emph{pivots}), one for each element of the
cluster. Each pivot is integrated against its conditional density. Every other prime (a \emph{free} prime) is summed over
its realised values by a sure bound.

Two difficulties have to be overcome:
\begin{itemize}
\item a prime that occurs in several elements constrains several of them at once;
\item a free prime may be decided after some pivot, so its realisation is not measurable at that pivot's decision time.
\end{itemize}
The \emph{Volume Lemma} (Lemma~\ref{lem:volume}) handles the first. The pivots are integrated one at a time, each
against its conditional density given the past (Lemma~\ref{lem:pil}). The resulting iterated Lebesgue integral is then
evaluated by a single change of variables, which replaces per-constraint widths by an exact Jacobian; no independence
between primes is used.
% Logged correction 41: clarifies that only conditional densities (tower property) are used, not independence.
The \emph{Pivot-Integration Lemma} (Lemma~\ref{lem:pil})
and the \emph{Staged Volume Lemma} (Theorem~\ref{thm:svl}) handle the second: integrate the pivots in stages, and bound
the free primes decided between stages by sure sums, each against a weight that is harmonic in some row.

\subsection{Setting}\label{sec:A-setting}
% v3: Appendix A, A.0

Throughout, $L=\log x$, $\ell=\log\log x$ and $\Lambda=\log X_0\ge 10^{8}$, as in Section~\ref{sec:queue-standing}.
Fix a dyadic height $z\ge X_0/2$ and a window $J\subset[z,2z]$ of length $\delta\le 1$, and put $\eps:=\delta/z$.

\begin{definition}[Configuration types]\label{def:config-type}
% v3: Appendix A, A.0
A \emph{configuration type} $\tau$ fixes:
\begin{enumerate}[label=(\alph*)]
\item $k$ rows $n_1,\dots,n_k\in J$, which are elements of $\NN$;
\item the jittered primes occurring in them (the \emph{columns}), listed in decision order;
\item the integer exponent matrix $E=(E_{ij})$, where $E_{ij}$ is the exponent of the $j$-th column in $n_i$.
\end{enumerate}
The \emph{tail} $d_i$ of a row, the product of its rational primes $<X_0$, is not part of the type; tails are summed in
the end step of Lemma~\ref{lem:pil}.
% Logged correction 55: v3 also fixes the tails in the type, although the end step sums them.
The type $\tau$ is \emph{rank-full} if $\operatorname{rank}E=k$. For a column $p$, write $e_p\in\Z_{\ge0}^{k}$ for its
column of $E$.
\end{definition}

\paragraph{Greedy pivots.}
Run through the columns latest-first, and declare $p$ a \emph{pivot} if and only if
\[
  e_p\notin\operatorname{span}\{e_q:\ q\text{ a pivot decided later than }p\}.
\]
Let $P$ be the set of pivots. For a rank-full type, $|P|=k=\operatorname{rank}E$ and the square matrix $E_P$ (the
columns of $E$ indexed by $P$) is invertible. The remaining columns are \emph{free}. For $i\le k$ write
\[
  n_i=c_i\prod_{j\in P}q_j^{E_{ij}},
\]
where $q_j$ is the value of the $j$-th pivot and $c_i\in\NN$ collects the free jittered primes of $n_i$ together with its
tail $d_i$.

\paragraph{Runs and blocks.}
Split $P$ into maximal runs $P_1,\dots,P_T$ of consecutive pivots in decision order, latest first. For $t<T$, the
\emph{late free block} $F_t$ consists of the free columns decided between $P_t$ and $P_{t+1}$. The \emph{early block}
$F_T$ consists of the free columns decided before all pivots. Write $P_{\le t}=P_1\cup\dots\cup P_t$, and similarly
$F_{\le t}$. By construction of the greedy pivots,
\begin{equation}\label{eq:A-span}
  e_f\in\operatorname{span}\{e_p:\ p\in P_{\le t}\}\qquad\text{for every }f\in F_{\le t}.
\end{equation}
Indeed, a free column $f$ decided after $P_{t+1}$ lies in the span of the pivots decided later than $f$, and these
belong to $P_{\le t}$.

\paragraph{Skeletons.}
The \emph{skeleton} $\mathfrak s$ of a rank-full type is its pivot pattern $E_P$, with the pivot columns in decision
order, together with its run structure $(|P_1|,\dots,|P_T|)$, a composition of $k$. Types with the same skeleton differ
only in their free columns: their number, their exponents, and the blocks in which they lie. The runs, the blocks
$F_t$ (as slot ranges) and the stage rows of Lemma~\ref{lem:SC} depend only on the skeleton. For $t<T$ the block $F_t$ is non-empty, since the runs are maximal; the early block may be
empty. There are $2^{k-1}$ run structures.
% Logged correction 55: Theorem R and the Cover Bound sum over all free structures; the lemmas below are stated for
% skeletons, i.e. for the union over the free columns, not for one full type.

\paragraph{Slot positions.}
Pivots are distinct primes, one per slot. Slot spacing is $\ge 0.99$, and the value $q$ of a jittered prime differs
from its slot $s_q$ by at most $H(s_q)/2$ (Section~\ref{sec:queue-standing}). Jittered primes are decided at slots
$\ge X_0$. Put $\sigma_j:=X_0+0.99(j-1)$. The $j$-th earliest pivot has slot $\ge\sigma_j$, and $v\mapsto v-H(v)/2$ is
increasing. So its value is at least $\sigma_j-H(\sigma_j)/2$, and at least $X_0-H(X_0)/2=\Xm$. Hence, whatever the labels
are, the $j$-th earliest pivot has value at least
\begin{equation}\label{eq:A-tau}
  \tau_j:=\max\bigl(\Xm,\ \sigma_j-2H(\sigma_j)\bigr).
\end{equation}
% Logged correction 42: v3 has max(X0, ...); jittered primes near X0 can have values in [X0^-, X0).
% Logged correction 32: H continuous (H := max(H_std, Mbar_1^2)); used for the monotonicity of v - H(v)/2 above.
Recall that $H$ is continuous and non-decreasing, and that $H(v)/v$ is decreasing (Theorem~\ref{thm:caplw}(c)). This
makes $v\mapsto v-H(v)/2$ increasing above: for $v<v'$, since $H(v')/v'\le H(v)/v<2$,
$H(v')-H(v)\le(v'-v)H(v)/v<2(v'-v)$. By \eqref{eq:jitter} and
Theorem~\ref{thm:caplw}(c), $\log H(X_0)\le8(\log\Lambda)^2+10^{-6}\Lambda$, so $H(X_0)/X_0\le e^{-0.99\Lambda}\le
e^{-10^{7}}$.
Values of jittered primes are only bounded below by $\Xm=X_0-H(X_0)/2$, as in Section~\ref{sec:queue-standing}; every
lower bound on such a value below uses $\Xm$, while slots are $\ge X_0$. Since $H(X_0)/X_0\le e^{-10^7}$, replacing $X_0$ by
$\Xm$ changes each constant by a relative amount at most $e^{-10^{7}}$.
% Logged correction 42.

\paragraph{Probabilistic set-up.}
$\mathcal F_{s-}$ is the information available before the decision at slot~$s$. $\Good(<s)$ is the good event of
Definition~\ref{def:good}, and $G_{<s}$ abbreviates it. For a window $J$, $t_J$ denotes the time at which the
cluster event of $J$ is examined; it satisfies $t_J\le t(E1)$ (Definition~\ref{def:good}). Every bound below is proved
on $\Good(<t_J)$, which is carried progressively along the pivots. No queue bound at heights inside the window is
used quantitatively.

\subsection{The Volume Lemma and the pattern sum}\label{sec:A-volume}

\begin{lemma}[Volume Lemma]\label{lem:volume}
% v3: Lemma A.1
Fix the cofactors $c_1,\dots,c_k$. The map $q_P=(q_j)_{j\in P}\mapsto(n_i)_{i\le k}$, $n_i=c_i\prod_jq_j^{E_{ij}}$,
is injective on $\R_{+}^{k}$, and
\begin{equation}\label{eq:volume}
  \operatorname{vol}\bigl\{q_P\in\R_+^k:\ n_i\in J\ \forall i\bigr\}
  =\int_{J^k}\frac{\prod_jq_j}{|\det E_P|\,\prod_in_i}\,dn,
  \qquad
  \prod_jq_j=\frac{\prod_i(n_i/c_i)}{\prod_jq_j^{\,s_j-1}},
\end{equation}
where $s_j:=\sum_iE_{ij}\ge1$ is the $j$-th column sum. In particular, if all $q_j\ge\Xm$, then the volume is at most
$\delta^k/\bigl(|\det E_P|\prod_ic_i\,(\Xm)^{M-k}\bigr)$, with $M:=\sum_{i,j}E_{ij}$.
% Logged correction 42: v3 has q_j >= X0 and X0^{M-k}.
\end{lemma}

\begin{proof}
Pass to logarithmic coordinates $x_j=\log q_j$. Then $\log n_i=\log c_i+\sum_jE_{ij}x_j$, an affine map with
invertible linear part $E_P$, so the map is injective. Its Jacobian matrix is $\partial n_i/\partial q_j=E_{ij}n_i/q_j$,
whose determinant is $\det(E_P)\prod_in_i/\prod_jq_j$. The change of variables $q_P\mapsto n$ therefore gives the
integral in \eqref{eq:volume}.

For the second identity, multiply the relations $n_i/c_i=\prod_jq_j^{E_{ij}}$ over $i$. This gives
$\prod_jq_j^{s_j}=\prod_i(n_i/c_i)$, hence $\prod_jq_j=\prod_i(n_i/c_i)\big/\prod_jq_j^{s_j-1}$. No column of the
invertible matrix $E_P$ vanishes, so $s_j\ge1$. If all $q_j\ge\Xm$, then $\prod_jq_j^{s_j-1}\ge(\Xm)^{M-k}$. Hence the
integrand is at most $1/\bigl(|\det E_P|\prod_ic_i(\Xm)^{M-k}\bigr)$, and $J^k$ has volume $\delta^k$.
\end{proof}

\begin{remark}
Lemma~\ref{lem:volume} involves no per-constraint widths. Bounding the constraints one at a time by Gaussian elimination
would combine them into integer combinations whose $\ell^1$-norms are not bounded by $k$ in general (this already
happens for $k=4$). The Jacobian avoids this entirely.
\end{remark}

\begin{lemma}[Pattern sum]\label{lem:pattern}\label{lem:patternsum}
% v3: Lemma A.2
Summing over the invertible nonnegative integer matrices $E_P$ with columns in decision order,
\[
  \frac1{k!}\sum_{E_P}\prod_j\tau_j^{-(s_j-1)}\ \le\ \prod_j\bigl(1-1/\tau_j\bigr)^{-k}
  \ \le\ e^{(3H(X_0)+2)k/X_0}\bigl(1+k/X_0\bigr)^{1.02k}.
\]
Since $H(X_0)/X_0\le e^{-10^{7}}$ (Section~\ref{sec:A-setting}), the right side is at most
$e^{k\,e^{-10^{6}}}\bigl(1+k/X_0\bigr)^{1.02k}$.
% CORRECTIONS-vs-v3.md entry 1: v3 states e^{2k/X0}; corrected constant e^{(3H(X0)+2)k/X0}.
\end{lemma}

\begin{proof}
By the Leibniz expansion, an invertible $E_P$ has a nonzero term $\prod_iE_{i,\sigma(i)}$ for some permutation $\sigma$.
So its support contains the permutation matrix of $\sigma$. Subtract that permutation matrix. The rest is a nonnegative
integer matrix $X$, and the column sums satisfy $s_j=1+\sum_iX_{ij}$. Choosing one such $\sigma$ for each $E_P$ makes the
map $E_P\mapsto(\sigma,X)$ injective. Hence
\[
  \sum_{E_P}\prod_j\tau_j^{-(s_j-1)}\le k!\prod_j\sum_{m\ge0}\binom{k+m-1}{m}\tau_j^{-m}=k!\prod_j(1-1/\tau_j)^{-k},
\]
since a column of extra mass $m$ can be distributed over the $k$ rows in $\binom{k+m-1}{m}$ ways. Next,
$(1-1/\tau)^{-k}\le\exp\bigl(k/(\tau-1)\bigr)$. Since $H(v)/v$ is decreasing, $H(\sigma+0.99)-H(\sigma)\le0.99\,H(X_0)/X_0$ for
$\sigma\ge X_0$. So $\sigma_j-2H(\sigma_j)$ increases by at least $0.989$ from one index to the next. Let $N_0$ be the
number of indices with $\tau_j=\Xm$. These satisfy $0.99(j-1)\le2H(\sigma_j)-H(X_0)/2\le2H(X_0)(1+10^{-8})$, so
$N_0\le2.03H(X_0)+1$. The remaining indices satisfy $\tau_{N_0+i}\ge\Xm+0.989(i-1)$ for $i\ge1$. Hence
\[
  \sum_{j\le k}\frac1{\tau_j-1}\le\frac{N_0+1}{\Xm-1}+\sum_{i=1}^{k}\frac1{\Xm-1+0.989i}
  \le\frac{3H(X_0)+2}{X_0}+1.02\log\Bigl(1+\frac{k}{X_0}\Bigr).
\]
Here we used $\Xm-1\ge X_0(1-H(X_0)/X_0)$, which gives $(2.03H(X_0)+2)/(\Xm-1)\le(3H(X_0)+2)/X_0$. We also used
$0.989/(\Xm-1)\le1/X_0$ and $1/0.989\le1.02$. Exponentiating gives the same bound as before,
$\prod_j(1-1/\tau_j)^{-k}\le e^{(3H(X_0)+2)k/X_0}(1+k/X_0)^{1.02k}$.
% Logged correction 42: floor X0^- in tau_j; the stated constant is unchanged.
\end{proof}

\subsection{The Pivot-Integration Lemma}\label{sec:A-pil}

Fix a skeleton $\mathfrak s$. Let $X_{\mathfrak s}$ be the number of realised configurations whose type has skeleton
$\mathfrak s$ and which satisfy all window and feasibility constraints. These are $k$-tuples of distinct elements of
$\NN\cap J$ with the pivot pattern and run structure of $\mathfrak s$; their free columns and tails are arbitrary. Let
$p_1,p_2,\dots$ be the pivots, latest first. Further data may be fixed in advance, as in Proposition~\ref{prop:cover};
the lemma holds with the same proof. Where such data restrict the block contents, the sums in~(iii) may still run over
all contents, since this only enlarges them.

\begin{lemma}[Pivot-Integration Lemma]\label{lem:pil}\label{lem:PIL}
% v3: Lemma A.3
For each run $P_t$, let $s_t$ be the slot of its last (earliest decided) pivot, let $f_t$ be the latest-decided column
of the block $F_t$ below it, and let $u_t$ be the first slot after the slot of $f_t$. For $t<T$ the block $F_t$ is
non-empty, since the runs are maximal. If the early block $F_T$ is empty, put $u_T:=X_0$, the first slot. Every pivot value is at most $2z$, since it divides some $n_i\in J$, and
$|q-s|\le H(s)/2$; so every slot label is at most $s_{\max}:=2z+0.51H(2z)$. The decision order gives $u_t\le s_t$. Put
\[
  \chi_t:=\mathbf 1(u_t\le s_{\max})\,\mathbf 1(G_{<u_t}),
\]
a function of the block data and of $\omega$ only. For a pending integrand $\Psi$, write $\Psi|_{s_t=u_t}$ for $\Psi$
with the label $s_t$ replaced by $u_t$. Define $S_{\mathfrak s}$ by the following recursion, run by run, latest first,
starting from
\[
  \Psi:=\prod_i\mathbf 1(n_i\in J)\prod_{j\in P}\mathbf 1(q_j\ge\tau_j),
\]
with $\tau_j$ as in \eqref{eq:A-tau} for the $j$-th earliest pivot. The caps hold for every realised configuration.
% Logged correction 55: v3 starts from prod 1(n_i in J); the caps are needed for the tau-replacement in Theorem svl.
For $t=1,\dots,T$:
\begin{enumerate}[label=(\roman*)]
\item $\Psi\leftarrow\chi_t\,\Psi|_{s_t=u_t}$;
\item at each pivot $p$ of $P_t$, latest first: $\Psi\leftarrow\int\Psi\,dq_{p}$ (Lebesgue measure; the window
  restriction of $p$ is dropped);
\item for $t<T$, at the late block $F_t$: $\Psi\leftarrow$ any bound for the realised sum $\sum_{F_t}\Psi$ over the
  block data (every non-empty content of $F_t$: the number of its free columns, their primes and their exponents) that
  holds for every $\omega$ and is a function of the label $s_{t+1}$ (and of no other label), of the variables not yet
  processed, and of the primes decided before $s_{t+1}$, non-increasing in $s_{t+1}$. An example is
  Lemma~\ref{lem:fac}\,(S3)(a). For $t=T$, at
  the early block and the tails: $\Psi\leftarrow$ a deterministic bound for the realised sum over them (the early block
  may be empty). For the harmonic Euler chains an example is Lemma~\ref{lem:fac}\,(S3)(c); tails of rows that carry no
  chain may instead be fixed in advance.
\end{enumerate}
Then
\[
  \E\bigl[X_{\mathfrak s}\,\mathbf 1(\Good(<t_J))\bigr]\le\theta^{k}\,\bar S_{\mathfrak s},\qquad\theta:=1+1/\Lambda .
\]
Step~(i) records the label sum at the last pivot of $P_t$ (see the proof). It comes before the integrals of the run, and
the block bound in~(iii) comes after them. In~(iii) the indicator $\chi_t$ is part of each summand, and the bound must
hold for every $\omega$. After the end step no label survives, so $S_{\mathfrak s}$ is a deterministic number; we
call it the \emph{envelope} and also write $\bar S_{\mathfrak s}:=S_{\mathfrak s}$.
% CORRECTIONS-vs-v3.md entry 12: v3 has theta = 1 + 1/L.
% Logged correction 46: the right side is the deterministic envelope \bar S_\tau (v3 states it with a realised sum).
% Logged correction 48: v3 sums each block inside the conditional expectation at s_t, before the label sum; there the
% harmonic weights are not yet available. Now the label sum comes first (step (i)), then the integrals, then the block.
\end{lemma}

\begin{proof}
Write $X_{\mathfrak s}$ as a sum, over the slot labels of the pivots and over the realised free data, of
$\prod_i\mathbf 1(n_i\in J)$; the caps may be inserted, since they hold for every realised configuration. Every pivot is decided before $t_J$. Along the pivots, latest first, the good events are
nested:
\[
  \mathbf 1(G_{<t_J})\le\mathbf 1(G_{<s_{p_1}})\le\mathbf 1(G_{<s_{p_2}})\le\cdots .
\]
Each $\mathbf 1(G_{<s_{p}})$ is $\mathcal F_{s_p-}$-measurable (Definition~\ref{def:good}). We induct along the decision
order, latest first, keeping the invariant that after each step the pending integrand, apart from the good-event
indicator carried along the pivots, is a nonnegative function of the variables not yet processed. As a function of the
labels of the run $P_t$ being processed, it depends only on $s_t$, through factors that are non-increasing in $s_t$ and
through the order constraint that $s_t$ is later than the slot of $f_t$. The pivots of a run are processed one after another, and the block
directly below a run right after the last pivot of the run. The factors changed in step~(i) do not involve the pivots
of $P_t$, so step~(i) commutes with their integrals; it is justified below, at the last pivot of $P_t$.

\emph{Pivot step.} Let $p$ be the current pivot, with slot $s$, and condition on $\mathcal F_{s-}$. Everything still
pending is then a function of $q_p$, of the label $s$, and of data decided before $s$. These data include the realised
primes of the block directly below $p$, if $p$ is the last pivot of its run. By \ref{eq:sF2}, the position of the prime
decided at slot $s$ has conditional density $\le1/H(s)$ on its window $W_s$. Hence
\[
  \E\bigl[\Psi\,\mathbf 1(G_{<s})\bigm|\mathcal F_{s-}\bigr]\le\mathbf 1(G_{<s})\,\frac1{H(s)}\int_{W_s}\Psi\,dq_p .
\]
Only continuous slots receive primes, so $s$ runs over continuous slots; the existence of a prime at slot $s$ is
bounded by $1$ and dropped. So are all decision-order constraints involving
the label $s$, between pivots and between a pivot and free data, except the constraint that $s$ is later than the slot
of $f_t$ when $p$ is the last pivot of $P_t$. Let $s'$ be the slot of the next pivot to be processed. If $p$ is not the last pivot of its run, $\Psi$ does not
depend on $s$, and we use $\mathbf 1(G_{<s})\le\mathbf 1(G_{<s'})$. If $p$ is the last pivot of the run $P_t$, so
$s=s_t$, then $\Psi$ depends on $s$ only through factors that are non-increasing in $s$, and through the decision order,
which requires $s$ to be later than the slot of $f_t$, i.e.\ $s\ge u_t$. Also $s\le s_{\max}$. So block data with
$u_t>s_{\max}$ have no admissible $s$, and contribute nothing. The good events are nested, so
$\mathbf 1(G_{<s})\le\mathbf 1(G_{<u_t})\,\mathbf 1(G_{<s'})$. Hence, for every admissible $s$,
\[
  \mathbf 1(G_{<s})\,\Psi\le\mathbf 1(G_{<s'})\,\chi_t\,\Psi|_{s_t=u_t},
\]
and the right side does not depend on $s$. (If $p$ is the earliest pivot, there is no $s'$ and its factor is omitted.)
This bound holds pointwise in $q_p$, in the block data and in $\omega$; in particular, no property of the value at which
the window constraints pin $q_p$ is used. Write $\Psi$ for the right side. All terms are nonnegative, so the sum over $s$
may be exchanged with the integral and with the sum over the block data, and
\[
  \sum_s\frac1{H(s)}\int_{W_s}\Psi\,dq=\int\Psi(q)\sum_s\frac{\mathbf 1(q\in W_s)}{H(s)}\,dq\le\theta\int\Psi\,dq .
\]
This uses \ref{eq:sF1}: at most $H(q)(1+\kappa_+/\log q)+3$ windows of continuous slots contain the point $q$, and $H$
varies by a factor
$1+o(1/\Lambda)$ over a window. Since $q\ge X_0^-$ and $\kappa_+=0.8$, the inner sum is at most
$1+0.8/\Lambda+3/H(X_0)+o(1/\Lambda)\le1+1/\Lambda=\theta$.

\emph{Block step.} After the label sum at $s_t$ and the integral over $q_p$, the pending integrand is
\[
  \theta\,\mathbf 1(G_{<s'})\int\chi_t\,\Psi|_{s_t=u_t}\,dq_p ,
\]
where $s'$ is the slot of the first pivot of $P_{t+1}$. The factor $\chi_t$ is a function of the block data and of
$\omega$, so it is carried through the integrals unchanged. The factor $\mathbf 1(G_{<s'})$ does not involve the block
$F_t$, whose primes are decided after $s'$. Sum over the realised block data, and replace the sum by the bound
of~(iii). The slot $u_t$ is a function of the block data: it is known inside the conditional expectation at $s_t$, but
not at the next pivot. It is consumed by the pathwise bound of~(iii), whose output does not contain it, so no
measurability of $u_t$ is needed. The result is a function of the label $s_{t+1}$, of the variables not yet processed
and of the primes decided before $s_{t+1}$, non-increasing in $s_{t+1}$. These primes are decided before every pivot of
$P_{t+1}$. So the invariant holds for the run $P_{t+1}$, and at its pivots other than the
last one the label sums are unaffected.

For the Euler chains, the bound is that of Lemma~\ref{lem:fac}\,(S3)(a). It sums over every content of the block,
and its output is the carried function $\Phi_t(s_{t+1})$ of Lemma~\ref{lem:fac}\,(S3)(b), which has the properties just
listed. It uses the sure bound of Lemma~\ref{lem:fac}\,(S3)(d) at slots $u\in[X_0,s_{\max}]$, valid on $G_{<u}$; the
factor $\chi_t$ provides $u_t\le s_{\max}$ and $G_{<u_t}$.

\emph{Order of decision versus order of value.} Slots and values are compared in three places:
\begin{itemize}
\item in the slot bounds $\tau_j$ of \eqref{eq:A-tau};
\item in $\mathrm{Proj}_t$ (Theorem~\ref{thm:svl}), which use slot spacing directly;
\item along the chains, which are indexed by slot, only in the sure bound for the top of a chain (Lemma~\ref{lem:fac}\,(S3),
  from the proof of \ref{eq:sF3p}).
\end{itemize}
In the last case the top is at a slot $u\in[X_0,s_{\max}]$, and the primes involved have values $\ge u-H(u)\ge u/2$.
So the cost is at most $e^{3(H(u)+3)/u}\le e^{4H(X_0)/X_0}$, since $H(v)/v$ is decreasing; it is the factor
$e^{5H(u)/u}$ already contained in that bound, and it is absorbed into $A_{\mathrm{top}}\le A_R\,e^{2.2/\Lambda}$.
% CORRECTIONS-vs-v3.md entry 13: v3 has e^{(H+3)/X0}, absorbed in theta.

\emph{End step.} At the earliest pivot, with slot $s_T$, the pivot step leaves
$\theta\int\chi_T\,\Psi|_{s_T=u_T}\,dq_p$. The early block and the tails are then summed over their realised values, and
the sum is replaced by the deterministic bound of~(iii). For the Euler chains this is Lemma~\ref{lem:fac}\,(S3)(c).
Taking expectations completes the induction.
\end{proof}

\subsection{Factorisation and sure sums}\label{sec:A-fac}

\begin{lemma}[Factorisation; sure sums]\label{lem:fac}\label{lem:FAC}
% v3: Lemma A.4
\begin{enumerate}[label=(\arabic*),leftmargin=*]
\item[\textup{(FAC)}] Let $f$ be a free column and $P_{\mathrm{late}}$ the set of pivots decided later than $f$, so
  $e_f=E_{P_{\mathrm{late}}}v_f$ for a unique $v_f$ by \eqref{eq:A-span}. After the pivots in $P_{\mathrm{late}}$ have
  been integrated,
  \[
    \Psi(q_f,\text{earlier})\le q_f^{-\beta_f}\,\mathbf 1\bigl(q_f\in C_\eps(\text{earlier})\bigr)\,\Psi_0(\text{earlier}),
    \qquad\beta_f:=\mathbf 1^{\mathsf T}v_f .
  \]
  Here $C_\eps$ is the projection of the feasibility caps (later pivots $\ge\tau_j$, and decision order) onto the
  $q_f$-axis, over the later-pivot values compatible with the window constraints. When the later pivots are pinned to
  relative width $O(\eps)$, this widens the caps by relative $O(\eps)$; the main line uses only the projection. The thin
  window constraints are invariant under the shift $x_f\mapsto x_f+s$,
  $x_{P_{\mathrm{late}}}\mapsto x_{P_{\mathrm{late}}}-s\,v_f$ (with $x=\log q$). So no late free prime ever enters a
  thin indicator.
\item[\textup{(S1$'$)}] Let $c$ range over cofactors all of whose primes are $\le P:=s_{p_t}+H(s_{p_t})$. Then for $\beta\ge1$,
  \[
    \sum_c c^{-\beta}\le\mathrm{Eu}(P),\qquad \mathrm{Eu}(x):=\prod_{\text{realised }p\le x}(1-1/p)^{-1}.
  \]
\item[\textup{(S2$'$)}] If $\beta<1$ and $c\le Y$, then $\sum_cc^{-\beta}\le Y^{1-\beta}\,\mathrm{Eu}(P)$. In particular
  $\#\{c\le Y:\ P^+(c)\le P\}\le Y\,\mathrm{Eu}(P)$.
\item[\textup{(S3)}] \emph{(Euler chains, indexed by slot.)} For a slot $s$ put
  \[
    \mathrm{Eu}_s(s):=\prod_{\text{realised }p\text{ decided before }s}(1-1/p)^{-1},
  \]
  the product over the realised primes of slot $<s$, rational primes $<X_0$ included. So $\mathrm{Eu}_s(X_0)$ is the
  product $\mathrm{Eu}_{\mathrm{rat}}$ over the rational primes $<X_0$. Let $s_t$, $f_t$, $u_t$, $s_{\max}$ and $\chi_t$
  be as in Lemma~\ref{lem:pil}, and put $s_{T+1}:=X_0$. Some rows are \emph{chain rows}, each with a starting run. Let
  $K_t$ be the number of chain rows starting at one of $P_1,\dots,P_t$ (so $K_0=0$), and $j_t:=K_t-K_{t-1}\ge0$ the
  number starting at $P_t$; $j_t=0$ is allowed. Suppose that every free column of $F_t$ occurs in a chain row starting
  at one of $P_1,\dots,P_t$, and give a prime $g$ of $F_t$ the weight $g^{-a_g}$, where $a_g\ge1$ is its total exponent
  in those chain rows. Each chain row $y$ has its tail $d_y$, with weight $1/d_y$. Put
  \[
    A_{\mathrm{top}}:=e^{\gamma}\log\bigl(2z+H(2z)\bigr)\cdot\exp\bigl(2.21/\Lambda+5H(X_0)/X_0\bigr),
  \]
  and define functions $G_t\ge0$ on $(0,\infty)$ by $G_0:\equiv1$ and, for $t\ge1$,
  \begin{equation}\label{eq:A-Grec}
    G_t(E):=\int_E^{A_{\mathrm{top}}}K_t\,x^{j_t-1}\,G_{t-1}(x)\,dx\quad(E<A_{\mathrm{top}}),\qquad
    G_t(E):=0\quad(E\ge A_{\mathrm{top}}).
  \end{equation}
  Each $G_t$ is non-increasing. Put $\Phi_t(u):=G_t(\mathrm{Eu}_s(u))/\mathrm{Eu}_s(u)^{K_t}$. Then:
  \begin{enumerate}[label=(\alph*)]
  \item \emph{(block bound with variable content)} on every $\omega$,
    \[
      \sum_{F_t}\chi_t\prod_{g\in F_t}g^{-a_g}\cdot\Phi_{t-1}(u_t)\ \le\ \Phi_t(s_{t+1}),
    \]
    where the sum runs over every non-empty content of $F_t$: every finite set of realised primes with slots in
    $[s_{t+1},s_{\max}]$, $f_t$ being the latest, and every choice of their exponents in the chain rows with all
    $a_g\ge1$;
  \item the carried function $\Phi_t$ is non-increasing. The factor $\Phi_t(s_{t+1})$ involves only the label $s_{t+1}$
    and primes decided before $s_{t+1}$. Step~(i) of Lemma~\ref{lem:pil} for the run $P_{t+1}$ turns it into
    $\Phi_t(u_{t+1})$, the carried factor in~(a) for the next block;
  \item at the end, for $t=T$: the sum of the left side of~(a) over all contents of the early block, the empty content
    included, multiplied by $\prod_y\sum_{d_y}1/d_y$ over the tails of the $K_T$ chain rows, is at most
    $G_T(\mathrm{Eu}_{\mathrm{rat}})+\mathrm{Eu}_{\mathrm{rat}}^{\,j_T}\,G_{T-1}(\mathrm{Eu}_{\mathrm{rat}})$;
  \item \emph{(sure top bound)} for a slot $u\in[X_0,s_{\max}]$, on $G_{<u}$,
    \[
      \mathrm{Eu}_s(u)\le e^{\gamma}\log u\cdot\exp\bigl(2.21/\Lambda+5H(u)/u\bigr)\le A_{\mathrm{top}}\le A_R\,e^{2.2/\Lambda},
    \]
    with $A_R=1.8(\log z+4)$ as in Theorem~\ref{thm:R}. The case $u=X_0$ involves only the rational primes, and gives
    $\mathrm{Eu}_{\mathrm{rat}}\le A_{\mathrm{top}}$.
  \end{enumerate}
  The recursion \eqref{eq:A-Grec} is evaluated in Theorem~\ref{thm:R} and in Proposition~\ref{prop:cover}. It does not
  factorise into one bound per row; its output, summed over the run structures, is at most
  $(k^k/k!)\,A_{\mathrm{top}}^{\,k}$ when every row is a chain row.
  % Logged correction 48: (a)-(d) replace v3's per-row cancellation inside the conditional expectation at s_t.
  % Logged correction 55: (a) now sums over every content of the block (the union over free structures that Theorem R
  % and the Cover Bound use); the per-type exact count of correction 48 is withdrawn.
\end{enumerate}
\end{lemma}

\begin{proof}
\emph{(FAC).} The window constraints read $\log n_i=\log c_i+\sum_{p}E_{ip}x_p\in\log J$. Since
$e_f=E_{P_{\mathrm{late}}}v_f$, the shift $x_f\mapsto x_f+s$, $x_{P_{\mathrm{late}}}\mapsto x_{P_{\mathrm{late}}}-sv_f$
leaves every $\sum_pE_{ip}x_p$ unchanged. In the integral over the later pivots, substitute
$y:=x_{P_{\mathrm{late}}}+v_fx_f$. The window indicators then no longer involve $x_f$. The Jacobian
$\prod_{p\in P_{\mathrm{late}}}q_p=e^{\mathbf 1^{\mathsf T}x_{P_{\mathrm{late}}}}=e^{\mathbf 1^{\mathsf T}y}\,q_f^{-\beta_f}$
produces the factor $q_f^{-\beta_f}$. The remaining dependence on $x_f$ is through the feasibility caps,
$y_p-v_{f,p}x_f\ge\log\tau_p$ and the decision order. These are half-spaces in $x_f$ for each $y$. Bounding them by their
projection over the $y$-values compatible with the window constraints gives $\mathbf 1(q_f\in C_\eps)$.

\emph{(S1$'$)--(S2$'$).} For $\beta\ge1$, $\sum_cc^{-\beta}\le\sum_cc^{-1}\le\prod_{p\le P}(1-1/p)^{-1}=\mathrm{Eu}(P)$,
since every $c$ is a product of realised primes $\le P$ (rational primes $<X_0$ included). For $\beta<1$ and $c\le Y$,
$c^{-\beta}=c^{-1}c^{1-\beta}\le Y^{1-\beta}c^{-1}$, and the same Euler product bounds $\sum c^{-1}$.

\emph{(S3).} (a) Fix the latest prime $f$ of the content, with slot $\sigma$, and put $u:=u_t$ (the first slot after
$\sigma$), $x:=1/f$, $E_1:=\mathrm{Eu}_s(\sigma)$ and $E_2:=\mathrm{Eu}_s(u)$. There is one prime per slot, so
$E_2=E_1/(1-x)$. Write $K:=K_t$, $j:=j_t$ and $m:=K_{t-1}$, so $K=m+j$, and $a:=s_{t+1}$. The other primes of the content
have slots in $[a,\sigma)$. For a realised prime $g$ with slot in $[a,\sigma)$, the sum of $g^{-a_g}$ over its exponent
vectors in the $K$ chain rows, the zero vector (meaning $g$ is absent) included, is $(1-1/g)^{-K}$; for $f$ itself the
zero vector is excluded, and the sum is $(1-x)^{-K}-1$. Dropping all other constraints, the contents with latest prime
$f$ contribute at most
\[
  \chi_t\Bigl(\frac{E_1}{\mathrm{Eu}_s(a)}\Bigr)^{K}\bigl((1-x)^{-K}-1\bigr)\frac{G_{t-1}(E_2)}{E_2^{\,m}}
  =\chi_t\,E_1^{\,j}\bigl((1-x)^{-j}-(1-x)^{m}\bigr)\frac{G_{t-1}(E_2)}{\mathrm{Eu}_s(a)^{K}} .
\]
Put $v:=\log(1/(1-x))>0$. If $j\ge1$, then, as $1-e^{-mv}\le mv\le(m/j)(e^{jv}-1)$,
\[
  (1-x)^{-j}-(1-x)^{m}=(e^{jv}-1)+(1-e^{-mv})\le\frac Kj\,(e^{jv}-1),
\]
so $E_1^{\,j}((1-x)^{-j}-(1-x)^m)\le(K/j)(E_2^{\,j}-E_1^{\,j})=\int_{E_1}^{E_2}Ky^{j-1}\,dy$. If $j=0$, then
$1-(1-x)^m\le mv=\int_{E_1}^{E_2}Ky^{-1}\,dy$. In both cases, since $G_{t-1}$ is non-increasing, the contribution is at
most
\[
  \chi_t\,\mathrm{Eu}_s(a)^{-K}\int_{E_1}^{E_2}Ky^{j-1}G_{t-1}(y)\,dy .
\]
The intervals $[\mathrm{Eu}_s(\sigma),\mathrm{Eu}_s(u)]$ of distinct realised primes are disjoint, since
$\mathrm{Eu}_s$ is non-decreasing along the slots, and they lie in $[\mathrm{Eu}_s(a),\infty)$. If $\chi_t=1$, then
$u\le s_{\max}$ and $G_{<u}$ holds, so $E_2\le A_{\mathrm{top}}$ by~(d). Summing over $f$ gives at most
$\mathrm{Eu}_s(a)^{-K}\int_{\mathrm{Eu}_s(a)}^{A_{\mathrm{top}}}Ky^{j-1}G_{t-1}(y)\,dy=\Phi_t(a)$, which is~(a); if
$\mathrm{Eu}_s(a)\ge A_{\mathrm{top}}$, no term has $\chi_t=1$ and both sides vanish. Both elementary inequalities are
sharp as $x\to0$, so~(a) loses almost nothing when the primes are large; the factor $k^k/k!$ of Theorem~\ref{thm:R}
comes from the order in which the labels are replaced, not from these inequalities.

(b) $G_t$ is non-increasing and $\mathrm{Eu}_s$ is non-decreasing. The primes of slot $<s_{t+1}$ are decided before every
pivot of $P_{t+1}$.

(c) For $t=T$, every prime of the early block has slot $\ge X_0=s_{T+1}$, so~(a) bounds the non-empty contents by
$\Phi_T(X_0)=G_T(\mathrm{Eu}_{\mathrm{rat}})/\mathrm{Eu}_{\mathrm{rat}}^{K_T}$. The empty content has $u_T=X_0$ and
contributes $\chi_T\Phi_{T-1}(X_0)\le G_{T-1}(\mathrm{Eu}_{\mathrm{rat}})/\mathrm{Eu}_{\mathrm{rat}}^{K_{T-1}}$. The tails
are the $X_0$-smooth numbers, so $\sum_d1/d=\mathrm{Eu}_{\mathrm{rat}}$ for each of the $K_T$ chain rows. Multiplying
gives~(c).

(d) The second case of the proof of \ref{eq:sF3p} bounds exactly the product over the realised primes decided before
$u$, on $\Good(<u)$, by $e^{\gamma}\log u\cdot\exp(2.21/\Lambda+5H(u)/u)$. That bound uses the rational primes
\cite[Thm.~8, (3.29)]{RosserSchoenfeld1962} and the sure local Mertens bound of Lemma~\ref{lem:cm}\,(iv) for the
realised primes with values in $[\Xm,u-H(u)]$, applied once per chain (Remark~\ref{rem:cm-chain}).

Finally, for $u\le s_{\max}\le2z+H(2z)$ we have $\log u\le\log(2z+H(2z))$ and $5H(u)/u\le5H(X_0)/X_0$, since $H(v)/v$
is decreasing; this gives $A_{\mathrm{top}}$. Moreover $\log(2z+H(2z))\le\log z+0.7$ and $5H(X_0)/X_0\le e^{-10^{6}}$,
so $A_{\mathrm{top}}\le e^{\gamma}(\log z+0.7)e^{2.21/\Lambda}(1+10^{-8})\le1.8(\log z+4)\,e^{2.2/\Lambda}$.
\end{proof}

\begin{remark}[Widened caps; an optional route]\label{rem:widened}
% v3: Lemma A.4, Remark (widened caps; optional)
Widen the caps of (FAC) by the pinned range $\rho_t$ of the later pivots, instead of by $O(\eps)$. Then they become
functions of outer variables only, the block sums commute with the later integrals, and the recursion of
Lemma~\ref{lem:pil} becomes a monotone functional of the final summand. The widened box costs a factor $e^{O(k\rho_t)}$.
Let $m:=\log2z/\Lambda$, the maximal exponent sum of a row. Then $\|E_P^{-1}\|_\infty\le k\,m^{k-1}$, by Cramer's rule and
Hadamard's inequality. The order is attained: rows $n_i=q_iq_{i+1}^{m-1}$ give $E_P=I+(m-1)N$ with $N$ the nilpotent
shift, whose inverse has a row of $\ell^1$-norm $\sum_{l<k}(m-1)^l$. So the route is harmless when
\[
  (k-1)\log m+O(\log k)\ \ll\ \log(z/\delta).
\]
The main line does not use this route. It uses the stagewise form (Theorem~\ref{thm:svl} with Lemma~\ref{lem:SC}),
which needs no conditioning hypothesis.
\end{remark}

\subsection{Nested stage rows}\label{sec:A-SC}

Let $B'$ be a rank-full set of $r$ rows, with exponent matrix $E$ on its jittered primes, and with pivots, runs and blocks
as in Section~\ref{sec:A-setting}. A \emph{cover} is a set $C\subseteq B'$ of rows such that every prime occurring in
$B'$ occurs in some row of $C$. For each $t$, let $M_t$ be the row matroid of $E_{\cdot,P_{\le t}}$, i.e.\ the matroid on
the rows $y\in B'$ in which a set of rows is independent iff their restrictions to the columns $P_{\le t}$ are linearly
independent.

\begin{lemma}[Nested stage rows]\label{lem:SC}\label{lem:sc}
% v3: Lemma A.7 (Lemma SC)
Order the rows so that those of $C$ come before those of $B'\setminus C$, and let $\mathcal R_{\le t}$ be the
lexicographically first basis of $M_t$ for this order. Then:
\begin{enumerate}[label=(\arabic*)]
\item $\mathcal R_{\le1}\subseteq\mathcal R_{\le2}\subseteq\dots\subseteq\mathcal R_{\le T}=B'$;
\item $|\mathcal R_{\le t}|=|P_{\le t}|$, and $E_{\mathcal R_{\le t},P_{\le t}}$ is invertible;
\item every column $j\in P_{\le t}\cup F_{\le t}$ has a nonzero entry in some row of $C\cap\mathcal R_{\le t}$.
\end{enumerate}
\end{lemma}

\begin{proof}
(2). Since $E_P$ is invertible, the columns $P_{\le t}$ are linearly independent, so $\operatorname{rank}M_t=|P_{\le t}|$.
A basis of $M_t$ is therefore exactly a set $\mathcal R$ of $|P_{\le t}|$ rows with $E_{\mathcal R,P_{\le t}}$
invertible.

(1). Suppose the restriction of a row $y$ to $P_{\le t+1}$ lies in the span of the restrictions of a set $X$ of rows.
Then so does its restriction to $P_{\le t}$. Hence $\cl_{M_{t+1}}(X)\subseteq\cl_{M_t}(X)$. A row $y$ lies in the
lexicographically first basis of a matroid $N$ iff $y\notin\cl_N(\text{rows listed before }y)$. Therefore
\[
  y\in\mathcal R_{\le t}\ \Rightarrow\ y\notin\cl_{M_t}(\text{earlier rows})\ \Rightarrow\
  y\notin\cl_{M_{t+1}}(\text{earlier rows})\ \Rightarrow\ y\in\mathcal R_{\le t+1}.
\]
Finally $\mathcal R_{\le T}=B'$, because $M_T$ has rank $r=|B'|$.

(3). Because the rows of $C$ are listed first, the lexicographically first basis contains a maximal independent subset of
$C$. So it maximises $|C\cap\mathcal R|$ among the bases $\mathcal R$ of $M_t$. Let $j\in P_{\le t}\cup F_{\le t}$, and
let $y\in C$ have $E_{y,j}\ne0$; such a $y$ exists because $C$ is a cover. If $y\in\mathcal R_{\le t}$, we are done.
Otherwise
\[
  E_{y,P_{\le t}}=\sum_{w\in\mathcal R_{\le t}}\alpha_wE_{w,P_{\le t}} .
\]
Column $j$ lies in the span of the columns $P_{\le t}$ over all rows. This is trivial for $j\in P_{\le t}$, and holds by
\eqref{eq:A-span} for $j\in F_{\le t}$. Hence $E_{y,j}=\sum_w\alpha_wE_{w,j}$, so some $w\in\mathcal R_{\le t}$ has
$\alpha_wE_{w,j}\ne0$. If $w\in C$, column $j$ is covered. If $w\notin C$, then $\alpha_w\ne0$, so
$\mathcal R_{\le t}-w+y$ is a basis of $M_t$ with one more row of $C$. This contradicts maximality.
\end{proof}

\begin{remark}
Neither minimality of $C$ nor any bound on $|C|$ is used. The order matters: if the rows of $B'\setminus C$ are listed
first, conclusion~(3) can fail. As an independent check, not a step of the proof, Lemma~\ref{lem:SC} was also confirmed by exact rational rank
computations: the nested sets exist on $32\,133$ random multi-stage configurations, and the explicit construction works on
$13\,433$ (configuration, cover) pairs (computer check; see Appendix~\ref{app:cert}).
\end{remark}

\subsection{The Staged Volume Lemma}\label{sec:A-svl}

Let $B'$ be a rank-full set of $k$ rows, $C\subseteq B'$ a cover, and $\mathcal R_{\le t}$ the nested bases of
Lemma~\ref{lem:SC}. The \emph{stage rows} are $C_t:=\mathcal R_{\le t}\setminus\mathcal R_{<t}$. For a row $y$ and a
stage $t$ put
\[
  c_y^{(\le t)}:=n_y\Big/\prod_{j\in P_{\le t}}q_j^{E_{yj}},
\]
the product of all primes of $y$ outside $P_{\le t}$, free primes and tail included. The entries of the free columns in
the rows outside $C$ are fixed in advance (skeleton (s4) of Proposition~\ref{prop:cover}).

\begin{theorem}[Staged Volume Lemma]\label{thm:svl}\label{thm:SVL}
% v3: Theorem A.5 (SVL)
In the recursion of Lemma~\ref{lem:pil}, integrate the pivots stage by stage, the stage-$t$ pivots $P_t$ by the
change of variables $q_{P_t}\mapsto(n_y)_{y\in C_t}$. Then the following hold.
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item \emph{(Stage identity with the cover split.)} After stage $t$, the Jacobian integrand is
  $\prod_{P_{\le t}}q\big/\allowbreak\bigl(|\det|\prod_{y\in\mathcal R_{\le t}}n_y\bigr)$, and
  \begin{equation}\label{eq:svl-identity}
    \frac{\prod_{j\in P_{\le t}}q_j}{\prod_{y\in\mathcal R_{\le t}}n_y}
    =W_t^{\mathrm{cov}}\prod_{y\in\mathcal R_{\le t}\setminus C}\frac1{n_y}\prod_{j\in P_{\le t}}q_j^{\,1-s_j^{\mathrm{cov}}}.
  \end{equation}
  Here $W_t^{\mathrm{cov}}:=\prod_{y\in C\cap\mathcal R_{\le t}}1/c_y^{(\le t)}$, and
  $s_j^{\mathrm{cov}}:=\sum_{y\in C\cap\mathcal R_{\le t}}E_{yj}\ge1$ by Lemma~\ref{lem:SC}\,(3). Further,
  $|\det|=|\det E_{\mathcal R_{\le t},P_{\le t}}|=\prod_{s\le t}|\det S_s|$, the product of the Schur complements. This
  is a nonzero integer, so $|\det|\ge1$, and the integrand is at most the right side of \eqref{eq:svl-identity}.
  Individual $|\det S_s|$ may be $<1$. The rows outside the cover keep the exact factor $1/n_y$, where $n_y\in J$ is
  their stage integration variable.
\item \emph{(Late primes are harmonic.)} For $f\in F_t$, Lemma~\ref{lem:SC}\,(3) puts $f$ in a cover row
  of $\mathcal R_{\le t}$, so $W_t^{\mathrm{cov}}$ carries $q_f^{-a}$ with $a\ge1$. The whole column of $f$ is
  determined by its $\mathcal R_{\le t}$-entries:
  \[
    e_f=E_{\cdot,P_{\le t}}\,E_{\mathcal R_{\le t},P_{\le t}}^{-1}\,e_{f,\mathcal R_{\le t}} .
  \]
  Hence the block data enter only through the weight $\prod_{g\in F_t}g^{-a_g}$, with $a_g\ge1$ the total exponent
  of $g$ in the cover rows of $\mathcal R_{\le t}$; the entries in the other rows of $\mathcal R_{\le t}$ carry no weight.
  The indicators of the remaining rows do not involve $f$. The sum over all contents of $F_t$ (the number, the primes
  and the exponents of its free columns) is bounded by Lemma~\ref{lem:fac}\,(S3)(a), with the rows of $C$ as chain rows,
  each starting at the stage where it joins; no count of free structures is needed.
  % Logged correction 55: v3 heads this item "no type count" but bounds one block structure at a time.
\item \emph{(Later stages; projected feasibility.)} The caps $\mathbf 1(q_j\ge\tau_j)$ of Lemma~\ref{lem:pil} hold at every
  point of the integration domain. After the block bound of each block, multiply by $\mathbf 1(\mathrm{Proj}_t)$.
  Here $\mathrm{Proj}_t$ is the projection onto the remaining variables of the real completions: summed primes
  $\ge\Xm$, and integrated pivots $\ge\tau_j$ in slot order.
  % Logged correction 42: summed primes >= X0^- (v3: >= X0).
  It is deterministic, and $\mathcal F_{p-}$-measurable at
  the next pivot $p$. On $\mathrm{Proj}_t$, a row $y$ joining at a stage $t'>t$ satisfies
  \begin{equation}\label{eq:B1}
    n_y\ \ge\ c_y^{\mathrm{noF}}\prod_{j\in P_{t'}}q_j^{E_{yj}}\prod_{j\in P_{<t'}}\tau_j^{E_{yj}}\,(\Xm)^{e_y(F_{\le t})}.
  \end{equation}
  Here $c_y^{\mathrm{noF}}$ is $c_y^{(\le t')}$ with the primes of $F_{\le t}$ removed, and $e_y(F_{\le t})$ is the total
  exponent of those primes in $y$. This is the only inequality the later stages use. In particular, the pivot mass of
  later rows carries $\tau_j^{-E}$, not $X_0^{-E}$.
\item \emph{(Slot-indexed chains.)} The Euler chains are indexed by slot, as in Lemma~\ref{lem:fac}\,(S3). They are
  carried from block to block by the functions $\Phi_t$ of Lemma~\ref{lem:fac}\,(S3)(b), and the end step (S3)(c) closes
  them. The block sums do not factorise over the rows.
\item \emph{(Joint caps.)} A row's free pieces across blocks, together with its tail, take one joint cap:
  $\mathbf 1\bigl(\prod_tc_t\le Y\bigr)\le Y/\prod_tc_t$. Capping block by block would give $Y^{T}$. For rows outside the
  cover the joint cap is not needed: their free part is a monomial in primes of the cover rows, and it is fixed in
  advance together with their $X_0$-smooth tail (skeleton (s4) of Proposition~\ref{prop:cover}).
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{(i).} We integrate the runs $P_1,P_2,\dots$ in this order, as Lemma~\ref{lem:pil} requires. Suppose stages
$1,\dots,t-1$ are done. The rows of $\mathcal R_{<t}$ have become integration variables $n_y\in J$. The later pivots
$P_{<t}$ are then functions of these variables and of the variables not yet processed; this uses the invertibility of
$E_{\mathcal R_{<t},P_{<t}}$ (Lemma~\ref{lem:SC}\,(2)). In logarithmic coordinates, for $y\in C_t$,
\[
  \log n_y=S_t\,x_{P_t}+(\text{terms in }n_{\mathcal R_{<t}}\text{ and in variables decided before }P_t),
\]
\[
  S_t:=E_{C_t,P_t}-E_{C_t,P_{<t}}E_{\mathcal R_{<t},P_{<t}}^{-1}E_{\mathcal R_{<t},P_t}.
\]
Here $S_t$ is the Schur complement of $E_{\mathcal R_{<t},P_{<t}}$ in $E_{\mathcal R_{\le t},P_{\le t}}$, and it is
invertible because both matrices are. The change of variables $q_{P_t}\mapsto(n_y)_{y\in C_t}$ has Jacobian
$|\det S_t|\prod_{y\in C_t}n_y/\prod_{j\in P_t}q_j$. Multiplying over $s\le t$, and using
$\det E_{\mathcal R_{\le t},P_{\le t}}=\prod_{s\le t}\det S_s$ (block $LU$ factorisation), gives the stated integrand.

For the identity \eqref{eq:svl-identity}, write $n_y=c_y^{(\le t)}\prod_{j\in P_{\le t}}q_j^{E_{yj}}$ for each
$y\in\mathcal R_{\le t}$. For the cover rows, the product is
\[
  \prod_{y\in C\cap\mathcal R_{\le t}}n_y=\prod_{y\in C\cap\mathcal R_{\le t}}c_y^{(\le t)}\cdot\prod_jq_j^{s_j^{\mathrm{cov}}} .
\]
Dividing $\prod_{P_{\le t}}q_j$ by this, and by the factors $n_y$ of the remaining rows of $\mathcal R_{\le t}$, gives
\eqref{eq:svl-identity}. The integer matrix $E_{\mathcal R_{\le t},P_{\le t}}$ is invertible, so
$|\det E_{\mathcal R_{\le t},P_{\le t}}|\ge1$.

\emph{(ii).} By \eqref{eq:A-span}, $e_f=E_{\cdot,P_{\le t}}v_f$ for some $v_f$. Restricting to the rows of
$\mathcal R_{\le t}$ gives $e_{f,\mathcal R_{\le t}}=E_{\mathcal R_{\le t},P_{\le t}}v_f$, hence
$v_f=E_{\mathcal R_{\le t},P_{\le t}}^{-1}e_{f,\mathcal R_{\le t}}$ and the displayed formula. In particular, once the
entries in the rows outside $C$ are fixed in advance (in Theorem~\ref{thm:R} there are no such rows; in
Proposition~\ref{prop:cover} they are part of the skeleton), the block data are determined by the cover-row pieces
$(c_{y,F_t})_{y\in C\cap\mathcal R_{\le t}}$, where $c_{y,F_t}$ is the product of the primes of $F_t$ in $y$ with their
exponents. This map is injective.

By Lemma~\ref{lem:SC}\,(3), $f$ has a nonzero entry in a cover row of $\mathcal R_{\le t}$. So
$W_t^{\mathrm{cov}}=\prod_{y\in C\cap\mathcal R_{\le t}}c_{y,F_t}^{-1}\cdot(\text{factors free of }F_t)$, with every
prime of $F_t$ appearing to a power $\ge1$. Before the block bound, replace $q_j^{1-s_j^{\mathrm{cov}}}$ by
$\tau_j^{1-s_j^{\mathrm{cov}}}$; this is valid at every point of the domain, by the caps of Lemma~\ref{lem:pil}, since
$s_j^{\mathrm{cov}}\ge1$. After that, the only dependence on $F_t$ is through the cover-row pieces. By injectivity, the
block sum is a sum over the realised primes of the columns of $F_t$ and their cover-row exponents, with weight
$\prod_{y\in C\cap\mathcal R_{\le t}}c_{y,F_t}^{-1}=\prod_{g\in F_t}g^{-a_g}$, $a_g\ge1$. Its sum over all contents of
$F_t$ is bounded by Lemma~\ref{lem:fac}\,(S3)(a), with the rows of $C$ as chain rows, each starting at the stage where it
joins.

For the remaining rows $y\notin\mathcal R_{\le t}$, the coefficient of $x_f$ in their constraints, after substituting
the pivots of $P_{\le t}$, is
\[
  e_{f,y}-E_{y,P_{\le t}}E_{\mathcal R_{\le t},P_{\le t}}^{-1}e_{f,\mathcal R_{\le t}}=e_{f,y}-E_{y,P_{\le t}}v_f=0 .
\]
So their indicators do not involve $f$. The non-cover rows of $\mathcal R_{\le t}$ carry $1/n_y$ with $n_y$ an
integration variable, which does not involve $f$ either.

\emph{(iii).} After a block bound, the summed primes are no longer variables. The remaining integrand is a function of the
variables not yet processed, and every actual configuration lies in $\mathrm{Proj}_t$. So inserting
$\mathbf 1(\mathrm{Proj}_t)$ costs nothing; with the caps of Lemma~\ref{lem:pil} it holds automatically for the
integrated pivots. $\mathrm{Proj}_t$ is defined by real inequalities among deterministic
quantities and variables decided before the next pivot, so it has the stated measurability.

For a row $y$ joining at stage $t'>t$, write
\[
  n_y=c_y^{\mathrm{noF}}\cdot(\text{primes of }F_{\le t}\text{ in }y)\cdot\prod_{j\in P_{t'}}q_j^{E_{yj}}
  \prod_{j\in P_{<t'}}q_j^{E_{yj}} .
\]
The left side, as a function of the remaining variables, does not depend on the completion. On $\mathrm{Proj}_t$ some
completion has all summed primes $\ge\Xm$ and all integrated pivots $\ge\tau_j$. This gives \eqref{eq:B1}. In the
stage-$t'$ identity, this inequality replaces the exact factorisation of the rows joining at $t'$; it drops the summed
primes from their cofactors and keeps $\tau_j^{-E_{yj}}$ for their earlier-integrated pivot mass.

\emph{(iv)} is Lemma~\ref{lem:fac}\,(S3). \emph{(v)}: if $\prod_tc_t\le Y$ then $Y/\prod_tc_t\ge1$. For a row outside
the cover, every prime lies in some cover row, and its exponents and its tail are fixed in advance (skeleton (s4) of
Proposition~\ref{prop:cover}).
\end{proof}

\begin{remark}
As a numerical corroboration only, and not part of the proof, some structural facts used in the proof of
Theorem~\ref{thm:svl} were checked by exact rational arithmetic in the case $C=B'$ (computer check; see
Appendix~\ref{app:cert}). There was no violation on random and adversarial families with $8$--$12$ rows and up to nine
stages. The facts checked are:
\begin{itemize}
\item the stage rows exist;
\item column masses are $\ge1$ on the stage prefix;
\item late free columns are covered and are determined by their stage-prefix entries, with vanishing coefficient in the
  remaining rows;
\item the cumulative determinants are nonzero integers.
\end{itemize}
The identity \eqref{eq:svl-identity} itself is not part of that check.
\end{remark}

\subsection{Theorem R}\label{sec:A-R}

\begin{theorem}[Rank-full clusters]\label{thm:R}
% v3: Theorem A.6 (Theorem R)
For every window $J\subset[z,2z]$ of length $\delta\le1$ and every $k\ge1$,
\[
\begin{aligned}
  &\E\Bigl[\#\{\text{rank-}k\ k\text{-subsets of }\NN\cap J\}\cdot\mathbf 1\bigl(\Good(<t_J)\bigr)\Bigr]\\
  &\qquad\le\frac{k^k}{k!}\,(\theta A_{\mathrm{top}}\delta)^k\,e^{(3H(X_0)+2)k/X_0}\bigl(1+k/X_0\bigr)^{1.02k}\\
  &\qquad\le(e\,\theta A_R\delta)^k\,e^{(3H(X_0)+2)k/X_0}\bigl(1+k/X_0\bigr)^{1.02k}\,e^{2.2k/\Lambda},
  \qquad A_R:=1.8(\log z+4),
\end{aligned}
\]
with $A_{\mathrm{top}}$ as in Lemma~\ref{lem:fac}\,(S3). Since $H(X_0)/X_0\le e^{-10^{7}}$, the factor
$e^{(3H(X_0)+2)k/X_0}$ is at most $e^{k\,e^{-10^{6}}}$.
% CORRECTIONS-vs-v3.md entry 1: v3 states e^{2k/X0}.
% Logged correction 55: v3 has (theta A_R delta)^k; the union over free structures costs the factor k^k/k! <= e^k.
\end{theorem}

\begin{proof}
\emph{Reduction to skeletons.} A $k$-subset of rank $k$ has $k!$ orderings. Fix one ordering. The type of the ordered
configuration has a skeleton $\mathfrak s$: its pattern $E_P$, with the pivot columns in decision order, and its run
structure. The expectation is at most $1/k!$ times the sum, over patterns $E_P$ and run structures, of
$\E[X_{\mathfrak s}\mathbf 1(\Good(<t_J))]$, where $X_{\mathfrak s}$ counts the ordered configurations with skeleton
$\mathfrak s$, with all their free columns and tails.

\emph{One skeleton.} Apply Lemma~\ref{lem:pil} to $X_{\mathfrak s}$, evaluating the recursion by Theorem~\ref{thm:svl}
with the cover $C:=B'$. Every row is then a chain row, starting at the stage where it joins, so $K_t=|P_{\le t}|$ and
$j_t=|P_t|\ge1$. Condition~(3) of Lemma~\ref{lem:SC} holds trivially: pivot columns are nonzero on $\mathcal R_{\le t}$
by~(2), and late free columns by \eqref{eq:A-span} and~(2). Collect the factors:
\begin{itemize}
\item every row's $n_y$ is integrated over $J$ in exactly one stage, which gives $\delta^k$;
\item $1/|\det E_P|\le1$;
\item the pivot label sums give $\theta^k$;
\item the caps give $\prod_jq_j^{1-s_j}\le\prod_j\tau_j^{-(s_j-1)}$, with $s_j$ the full column sums
  (\eqref{eq:svl-identity} and \eqref{eq:B1});
\item the free columns, in all blocks, and the tails: by Lemma~\ref{lem:fac}\,(S3)(a)--(c), at most
  \[
    \mathrm{Fin}(\mathfrak s):=G_T(\mathrm{Eu}_{\mathrm{rat}})+\mathrm{Eu}_{\mathrm{rat}}^{\,j_T}\,G_{T-1}(\mathrm{Eu}_{\mathrm{rat}}),
  \]
  with the recursion \eqref{eq:A-Grec} for the run structure $(j_1,\dots,j_T)$.
\end{itemize}
The factor $e^{2.2/\Lambda}$ in $A_{\mathrm{top}}\le A_Re^{2.2/\Lambda}$ is the error term of the sure Mertens bound
\ref{eq:sF4}; the rational primes (the tails) enter through \cite[Thm.~8, (3.29)]{RosserSchoenfeld1962}, inside the
proof of \ref{eq:sF3p}; and the slot--value fuzz of Lemma~\ref{lem:pil} is contained in $A_{\mathrm{top}}$, as its
factor $e^{5H(X_0)/X_0}$.

\emph{The sum over run structures.} Put $A:=A_{\mathrm{top}}$ and $\rho:=\mathrm{Eu}_{\mathrm{rat}}/A\in(0,1]$
(Lemma~\ref{lem:fac}\,(S3)(d)). Substituting $x=Ay$ in \eqref{eq:A-Grec} shows $G_t(E)=A^{K_t}g_t(E/A)$, where $g_t$
is the same recursion with $A=1$. Define polynomials $H_K$ on $[0,1]$ by $H_0:=1$ and
\begin{equation}\label{eq:A-HK}
  H_K(E):=K\sum_{j=1}^{K}\int_E^1x^{j-1}H_{K-j}(x)\,dx\qquad(K\ge1).
\end{equation}
Splitting off the last run, the sum of $g_T$ over the compositions $(j_1,\dots,j_T)$ of $K$ is $H_K$. Hence
\[
  \sum_{\text{run structures}}\mathrm{Fin}(\mathfrak s)=A^k\,F_k(\rho),\qquad
  F_k(\rho):=H_k(\rho)+\sum_{j=1}^{k}\rho^{\,j}H_{k-j}(\rho).
\]
Each $H_K$ is nonnegative and non-increasing, with $H_K'(E)=-K\sum_{j}E^{j-1}H_{K-j}(E)$. So
\[
  F_k'(\rho)=\sum_{j=1}^{k}(j-k)\rho^{\,j-1}H_{k-j}(\rho)+\sum_{j=1}^{k}\rho^{\,j}H_{k-j}'(\rho)\le0,
\]
and $F_k(\rho)\le F_k(0)=H_k(0)$. To evaluate $H_k(0)$, let $T(w):=\sum_{n\ge1}n^{n-1}w^n/n!$ be the tree function, the
inverse series of $w\mapsto we^{-w}$, so $T(w)e^{-T(w)}=w$. Consider the power series $P(E,z):=\sum_{K\ge0}H_K(E)z^K$.
Comparing coefficients, \eqref{eq:A-HK} is equivalent to
\[
  \partial_EP=-z\,\partial_z\Bigl(\frac{zP}{1-Ez}\Bigr),\qquad P(1,z)=1,
\]
which determine $P$. Put $\zeta:=T(ze^{-Ez})$, so $\log\zeta-\zeta=\log z-Ez$. Then $\partial_E\zeta=-z\zeta/(1-\zeta)$
and $\partial_z\zeta=\zeta(1-Ez)/(z(1-\zeta))$, and a direct computation shows that $P=(1-Ez)/(1-\zeta)$ satisfies the
equation: both sides equal $-z/(1-\zeta)-z\zeta(1-Ez)/(1-\zeta)^3$. Also $T(ze^{-z})=z$ gives $P(1,z)=1$. At $E=0$,
$P(0,z)=1/(1-T(z))=\sum_KK^Kz^K/K!$, since $zT'(z)=T/(1-T)$. Hence
\[
  \sum_{\text{run structures}}\mathrm{Fin}(\mathfrak s)\le A_{\mathrm{top}}^{\,k}\,\frac{k^k}{k!}\le e^k A_{\mathrm{top}}^{\,k}.
\]
(More precisely, $F_k(\rho)=[z^k]\,1/(1-T(ze^{-\rho z}))$, which equals $1$ at $\rho=1$ and $k^k/k!$ at $\rho=0$.)

\emph{The sum over patterns.} Summing $\prod_j\tau_j^{-(s_j-1)}$ over patterns with the factor $1/k!$ gives
$e^{(3H(X_0)+2)k/X_0}(1+k/X_0)^{1.02k}$, by Lemma~\ref{lem:pattern}. Finally $A_{\mathrm{top}}\le A_Re^{2.2/\Lambda}$ by
Lemma~\ref{lem:fac}\,(S3)(d).
\end{proof}

\begin{remark}
Theorem~\ref{thm:R} contains two simpler cases:
\begin{itemize}
\item tree-like types;
\item types in which every free prime is decided before every pivot. For these, Lemma~\ref{lem:volume} alone gives the
  bound, with the free data summed at the end.
\end{itemize}
\end{remark}

\begin{corollary}[The case of rank at least $r_0$]\label{cor:R-Ecl}
% v3: Theorem A.6, application to E_cl
Let $k=r_0=\lceil L/(3\ell)\rceil+3$ and $\delta=4L^{-6}$, and let $L\ge465$. Then the right side of
Theorem~\ref{thm:R}, multiplied by the number $2z/\delta$ of (overlapping) grid windows at height $z\le x$, is at most
$L^{-4}$. This is the
rank-$\ge r_0$ case of Proposition~\ref{prop:ecl}, and it is valid for every $L\ge465$.
\end{corollary}

\begin{proof}
Per element, the factor in Theorem~\ref{thm:R} is
$e\,\theta A_R\delta\,e^{(3H(X_0)+2)/X_0}(1+r_0/X_0)^{1.02}e^{2.2/\Lambda}$. Here:
\begin{itemize}
\item $\theta\,e^{(3H(X_0)+2)/X_0}e^{2.2/\Lambda}\le1.01$, since $H(X_0)/X_0\le e^{-10^{7}}$ (Section~\ref{sec:A-setting}),
  $\theta=1+1/\Lambda$ and $\Lambda\ge10^{8}$;
\item $A_R\le1.8(L+4)$, since $z\le x$;
\item $1+r_0/X_0\le1+L\le2L$.
\end{itemize}
Hence the factor is at most
\[
  e\cdot1.8(L+4)\cdot4L^{-6}\cdot(2L)^{1.02}\cdot1.01\le15e\,L^{-3.98}\qquad(L\ge465).
\]
Since $2z/\delta\le e^{L}L^{6}/2$ and $\log(15e)-3.98\ell<0$, and $r_0\ge L/(3\ell)+3$,
\[
\begin{aligned}
  \log\Bigl(\frac{2z}\delta\cdot\mathrm{RHS}\Bigr)&\le L+6\ell+r_0\bigl(\log(15e)-3.98\ell\bigr)
  \le L+6\ell+\Bigl(\frac L{3\ell}+3\Bigr)\bigl(\log(15e)-3.98\ell\bigr)\\
  &\le L\Bigl(-0.32666+\frac{1.2361}{\ell}\Bigr)-5.94\,\ell+11.13\ \le\ -4\ell
\end{aligned}
\]
for $L\ge465$. Here $1-3.98/3=-0.32666\ldots$, $\log(15e)/3\le1.2361$, $3\log(15e)\le11.13$, and the term
$-11.94\ell=3\cdot(-3.98\ell)$ is kept. For the last step, put $f(L):=L(-0.32666+1.2361/\ell)-1.94\ell+11.13$; then
$f(465)\le-59$, and $f'(L)\le-0.32666+1.2361/\ell<0$ for $L\ge465$.
% CORRECTIONS-vs-v3.md entry 17: v3 has -0.327.
% Logged correction 55: the per-element constant is 15e (factor k^k/k! <= e^k of Theorem R); the -11.94 ell term from
% r_0 >= L/(3 ell)+3 is now kept, since the simplified chain without it fails at L = 465.
\end{proof}
