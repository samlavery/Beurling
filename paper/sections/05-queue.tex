\section{The queue}\label{sec:queue}
% v3: §4 (opening paragraph)

This section proves that, on an event of probability greater than $\tfrac12$ (Lemma~\ref{lem:budget}), the Lindley queue of
Section~\ref{sec:construction} satisfies $Q_j\le B(s_j)$ at every slot (Theorem~\ref{thm:queue}). Here $B(x)$ is quasi-polylogarithmic in
$x$, and $B(x)\le H(x)$ at every scale (Theorem~\ref{thm:queue}). Consequently (L-a) holds:
% Logged correction 68 (check R12): v3 has "bounded by H(x)^{O(1)} on the raised range".
$|D(x)|\le B(x)+\tfrac{H(x)}2(1+\kappa_+/\log x)+4\le x^{\eta}$. The ingredients are:
% Approved deviation from v3 (CORRECTIONS-vs-v3.md, #15): v3 has H(x)/2+2.
% Logged correction 36: the symmetric lag constant is +4 (lower constant of eq:lag), not +3.
\begin{itemize}
\item the landing count and the local count (Lemma~\ref{lem:landing} and Corollary~\ref{cor:localcount});
\item the good event and the first-failure budget (Definition~\ref{def:good} and Lemma~\ref{lem:budget});
\item the cluster event $\qEcl$ and its per-prime increment bound (Proposition~\ref{prop:ecl} and Corollary~\ref{cor:increment});
\item the chain of lemmas proving $\qEcl$ at every scale (Sections~\ref{sec:queue-ecl}, \ref{sec:queue-cover} and~\ref{sec:queue-capl}, together
  with Appendix~\ref{app:rankfull});
\item the analysis of the absorbed set $\Abs$: its pointwise sparsity (Lemma~\ref{lem:lemmaP}) and the consequences
  (Section~\ref{sec:queue-abs}).
\end{itemize}

\subsection{Conventions, parameters and standing facts}\label{sec:queue-conv}\label{sec:queue-standing}
% v3: §4.1

\paragraph{Scales.}
A \emph{dyadic scale} is $x=2^mX_0$ with $m\ge0$. We write $L:=\log x$ and $\ell:=\log L$. All parameters are \emph{frozen per
scale}: they are evaluated at $x$ and used for heights in $[x,2x)$. The one exception is the jitter width $H(\cdot)$,
together with the functions $c(\cdot)$, $\bar r_0(\cdot)$ and $\bar M_1(\cdot)$ that enter it: these are functions of the
height itself, so that $H$ is continuous. Where a statement at scale $x$ writes $H$ without an argument, it means $H(x)$.
% Logged correction 32: v3 freezes H per scale and builds it from the integer r_0, so H jumps.
We write $\Lambda:=\log X_0\ge10^8$
(Section~\ref{sec:params}).

\paragraph{Parameters at scale $x$.}
These are the choices of Section~\ref{sec:params} and Section~\ref{sec:construction}.
\begin{itemize}

  $k\in\Z$, with left end $k\delta_1\in[z,2z)$. They have length $2\delta_1=4L^{-6}$ and lie in $[z,2z+1]$, and each grid
  window belongs to the dyadic height of its left end.
\item $r_0:=\lceil L/(3\ell)\rceil+3$ is the rank threshold (Theorem~\ref{thm:R}).

\paragraph{Filtration.}
% Logged correction 77 (R18, #77-2): slot starts are non-decreasing (an empty continuous slot shares its start); v3 had s_1<s_2<...
Slots $j=X_0,X_0+1,\dots$ (Section~\ref{sec:construction}) are processed in index order. Their starts satisfy $s_j\le s_{j+1}$,
with equality exactly for an empty continuous slot $j$, and distinct continuous slots, hence distinct decision slots, have
distinct starts. $\mathcal F_t$ is the $\sigma$-field of
all decisions and jitter offsets at slots $\le t$, and $\mathcal F_{s-}$ is the same for slots $<s$. For a decision slot $s$,
$\NN_{\prec s}$ denotes the set of elements of $\NN$ all of whose primes are decided at slots $<s$. It is
$\mathcal F_{s-}$-measurable.

\paragraph{The threshold $\Xm$.}
With centred placement (Section~\ref{sec:construction}), jittered primes have values at least
$\Xm:=X_0-H(X_0)/2$. Since $H(X_0)/X_0=e^{8\log^2\Lambda-\Lambda}\le e^{-0.98\Lambda}$, we have
$\log\Xm\ge\Lambda-e^{-0.98\Lambda}$. Every lower bound for a jittered prime below is stated with $\Xm$.
% Logged correction 43: v3's convention "read X_0 as X_0^-" is replaced by explicit X_0^- throughout; v3 also has
% X_0(1-O(e^{2715-Lambda})), which is uniform only near Lambda = 1e8.

\paragraph{The Lindley recursion.}
By Lemma~\ref{lem:lindley}, the queue is $Q_0=0$ and
\begin{equation}\label{eq:lindley}
Q_j=\max(0,\,Q_{j-1}+A_j-1),
\end{equation}
where $A_j$ is the number of arrivals assigned to slot $j$. The \emph{excess} of a slot window $W$ is
$\qexc(W):=C(W)-\#\{\text{slots of }W\}$. By the excess identity (Lemma~\ref{lem:excess}),
$\qexc(W)=\qCjit(W)-\qljit(W)\pm2$. Here $\qCjit$ counts the non-absorbed composites whose latest-decided prime is
jittered, and $\qljit(W)$ counts the jittered (continuous) slots of $W$. Lindley's formula gives
\begin{equation}\label{eq:lindleymax}
Q_j=\max\Big(0,\ \max_{k<j}\qexc((k,j])\Big).
\end{equation}

\paragraph{Standing facts at height $z$.}
All of the following are sure (pathwise) statements, except (F2), which is distributional.
\begin{enumerate}[label=\textup{(F\arabic*)},leftmargin=*]
\item\label{eq:sF1} At most $H(t)(1+\kappa_+/\log t)+3$ decided primes have jitter windows containing a given point $t$, since primes are decided only at continuous slots, whose starts have density
  $T'_{\mathrm{cont}}\le1+\kappa_+/\log t+1/(\kappa t)$ (Section~\ref{sec:construction}). Consequently, the label sums over slots in
  Lemma~\ref{lem:pil} carry the factor $\theta:=1+1/\Lambda$, which bounds $1+\kappa_+/\log q+3/H$ for jittered primes
  $q\ge\Xm$ (with $\kappa_+=0.8$).
  % Approved deviation from v3 (CORRECTIONS-vs-v3.md, #12): v3 has theta := 1+1/L.
\item\label{eq:sF2} \emph{(Conditional law of positions.)} Whether a prime is decided at slot $s$, and the slot $s$ itself,
  are $\mathcal F_{s-}$-measurable (Lemmas~\ref{lem:adapted} and~\ref{lem:C2}). Given $\mathcal F_{s-}$, the position of
  the prime decided at $s$ is uniform on $[s-H(s)/2,\,s+H(s)/2)$ (centred box jitter), so its conditional density is at
  most $1/H(s)$. The slots and positions of distinct primes are \emph{not} independent: which slots carry primes depends on
  the history of the greedy process. Joint statements follow by iterated conditioning along the decision order. Let
  $j_1<\dots<j_k$ be slot indices, let $D$ be the event that primes are decided at all of them, with positions
  $q_1,\dots,q_k$, and let $h_1,\dots,h_k$ be $\mathcal F_{s_{j_1}-}$-measurable with $H(s_{j_i})\ge h_i$ surely (for instance
  $h_i:=H(s_{j_1})$, as $H$ is non-decreasing). Then, for
  every Borel $g\ge0$ on $\R^k$,
  \[
  \E\big[\mathbf 1_D\,g(q_1,\dots,q_k)\,\big|\,\mathcal F_{s_{j_1}-}\big]\le\prod_{i=1}^kh_i^{-1}\int_{\R^k}g(t)\,dt.
  \]
  Indeed, given $\mathcal F_{s_{j_k}-}$ the event that $j_1,\dots,j_{k-1}$ are decision slots, the positions
  $q_1,\dots,q_{k-1}$, the event that $j_k$ is a decision slot, and the box at $s_{j_k}$ are all determined, while $q_k$ is
  uniform on a box of length at least $h_k$. So the conditional expectation given $\mathcal F_{s_{j_k}-}$ is at most
  $h_k^{-1}$ times $\mathbf 1_{D'}\int g(q_1,\dots,q_{k-1},t)\,dt$, where $D'$ is the event for $j_1,\dots,j_{k-1}$; now induct
  on $k$.

  % conditional law and its iterated form.
\end{enumerate}

If $Y+H(Y)\le s$, then all of $\NN\cap[1,Y]$ is decided before $s$, and the bound is the sure harmonic bound
$\log Y+B_1$ (the constant $B_1$ of \S\ref{sec:noise-notation}, from \textup{(H$_{\mathrm{BS}}$)}), which holds on $\Good(<s)$: if $Y\ge\Xm$ then $Y\le s/2$ or $Y\le s-H(Y)$ with $Y\ge s/2$, so

$\prod_{p\text{ decided before }s}(1-1/p)^{-1}$.

\begin{itemize}
\item The rational primes $p\le y_0$ give at most $e^{\gamma}\log y_0\,(1+1/(2\log^2y_0))\le e^{\gamma}\Lambda(1+1/(2\Lambda^2))$

\item The realised primes in $(y_0,s-H(s)]$ are the rational primes in $(y_0,X_0)$, the jittered primes that land in

\paragraph{Absorbed and non-absorbed elements.}
$\Abs$ (see \eqref{eq:absorbed}) is the extended absorption set: the elements $n$ with $\log n\ge L_*=10^7$ whose latest jittered
prime $r$ satisfies $r<y(n)$. In Lemma~\ref{lem:lemmaP}, $y(n)$ is replaced by $\sup_{[u,2u]}y$ on the dyadic block of
$n$. Absorbed elements are counted in the target $T$, as are the rigid composites (whose latest prime is rational, Lemma~\ref{lem:excess});
both are jumps of $T$ and drop out of the excess. The statements below about the latest primes of arrivals (the counts $X_q$, the
increment bound and the thresholds $y$, $y_-$) concern the non-absorbed composites whose latest prime is jittered, that is, the
composites counted by $C_{\mathrm{jit}}$.
\begin{itemize}
\item \emph{Stopped process.} Every statement of this section is proved on $\Good(<t)$ for the relevant $t$, that is, for the
  process stopped at the first failure time $\tau$. There Lemma~\ref{lem:lemmaP} gives $\rho_{\Abs}\le L^{-4}<0.1/\log u$, so the
  capped density $\bar\rho_{\Abs}=\min(\rho_{\Abs},0.1/\log u)$ of \eqref{eq:target} equals $\rho_{\Abs}$. No argument of this
  section uses $\rho_{\Abs}$ off the good event: Lemma~\ref{lem:lemmaP}, Proposition~\ref{prop:S1S2}, the compensator of
  $\sigma_{\Abs}$ (Lemma~\ref{lem:rhoA}(b)) and \eqref{eq:T1} all run on the good event.
\item \emph{Slot lengths.} This section uses $\rhodet$ only through $T'\le1+1/(\kappa u)\le1.01$, in the linear
  normalisation of Section~\ref{sec:construction} (see \eqref{eq:rhodet} and Lemma~\ref{lem:blin}(1)): $\rhodet\le$ (rigid fraction)
  $+1/(\kappa u)$, and the positivity margin is \eqref{eq:G1}, with no $\kappa/2$ loss. This enters as continuous-slot spacing at least
  $0.99$ in (F1) and Appendix~\ref{app:rankfull}, and in the candidate counts of Lemma~\ref{lem:landing}. No $\kappa/2$ form is used. The lower bound \eqref{eq:G1}, in the form
  $T'_{\mathrm{cont}}\ge0.0899/\log u$ (Remark~\ref{rem:horizon}), enters only through the slot length $s_{j+1}-s_j\le11.2\log s_j$
  and the physical length of the windows of $\mathcal W_x$ (Section~\ref{sec:queue-bound}).
\end{itemize}

\subsection{Landing count and local count}\label{sec:queue-landing}
% v3: §4.2

% Logged correction 77 (R18 P1): rho_H <= 1.33 (was 1.35, now proved: 16^{16 ell/L} at L >= 1099.3), so that 4 rho_H + 6.0001 <= 11.4 after #77-8.
Fix a dyadic $P$ and put $\lambda_P:=4\log P$, $H_-:=H(P/2)$, $H_+:=H(8P)$ and $\rho_H:=H_+/H_-$. Then $\rho_H\le1.33$ for
$\log P\ge1100$, and $\rho_H\le1+1.2\cdot10^{-5}$ for $\log P\ge10^8$, the raise of Theorem~\ref{thm:caplw} included. Indeed,
on $[X_0/2,\infty)$ both one-sided derivatives of $\log H$ in $\log z$ are at most $16\ell/L+10^{-7}\le3.1\cdot10^{-6}$: above
$X_0$ by Theorem~\ref{thm:caplw}(c), and below $X_0$ because $H=\qHstd$ there (Section~\ref{sec:queue-conv}). So
$\rho_H\le16^{3.1\cdot10^{-6}}\le1+1.2\cdot10^{-5}$ for $\log P\ge10^8$; for $\log P\ge1100$ the same bound $16\ell/L$ (where $H=\qHstd$,
with $\ell/L$ largest at $L=\log(P/2)\ge1099.3$) gives $\rho_H\le16^{0.10192}\le1.327$. For a window $J\subset[P,4P+1]$ with $|J|\le6H(P)$ we
define:
\begin{itemize}
\item the \emph{candidates} $\mathcal C(J)$: the decision slots $s$ with $|s-J|\le H(s)/2$. (Selection is by decision slot,
  which is predictable; selection by the value $q$ would not be.)
\item $\Lambda(J)$: the number of primes decided at slots in $\mathcal C(J)$ that land in $J$;
\item the compensator $\mu(J):=\sum_{s\in\mathcal C(J)}\Prob(\text{the prime at $s$ lands in $J$}\mid\mathcal F_{s-})$;
\item $\bar\mu(J):=(|J|+H_++2)\min(1,|J|/H_-)$. By (F1)--(F2) this is a \emph{sure} bound:
  $\mu(J)\le\bar\mu(J)\le(1+\rho_H)|J|+2$. Here the slot-density excess $\kappa_+/\log P+1/(\kappa P)$ in (F1) is absorbed by
  the gap between $H_+$ and the widths $H(s)\le H(4P+1+H)$ of the candidates. Above $x_1$, where $R'\ne0$ (Section~\ref{sec:target}),
  $H=\qHstd$ and this gap is at least $11\ell H/L$, more than the excess $7\cdot0.8H/L$ over the at most $7H$ candidate
  slots. Below $x_1$, $R'=0$ and the excess is at most $7H(P)/(\kappa P)\le7X_0^{0.4}H(X_0)/X_0\le e^{-0.5\Lambda}$, since
  $H(v)/v$ is decreasing.
\end{itemize}

\begin{lemma}[Landing count]\label{lem:landing}
% v3: Lemma 4.1
Let $\qLand(J)$ be the event
\begin{equation}\label{eq:landing}
|\Lambda(J)-\mu(J)|\le\sqrt{2\bar\mu(J)\lambda_P}+\tfrac23\lambda_P,\qquad\text{hence}\qquad
\Lambda(J)\le\min\big(\#\mathcal C(J),\,2\bar\mu(J)+\tfrac76\lambda_P\big).
\end{equation}
Then $\Prob(\qLand(J)^c)\le2e^{-\lambda_P}=2P^{-4}$, with no conditioning. Summed over the landing windows $J\in\mathcal J_P$
defined below, the total is at most $146H(P)P^{-3}$. The event $\qLand(J)$ is $\mathcal F_t$-measurable with
$t(\qLand(J)):=\max J+H(\max J)$.
\end{lemma}

\begin{proof}
Take the indicators $\mathbf 1(\text{the prime at $s$ lands in $J$})$, $s\in\mathcal C(J)$, in slot order. Their conditional
means sum to $\mu(J)\le\bar\mu(J)$ surely, and the increments are at most $1$. Apply Freedman's inequality
\cite[(4.1)]{Freedman1975} to both tails. We use it in the form
\begin{equation}\label{eq:freedman}
\Prob(\exists n:\ S_n\ge a,\ V_n\le v)\le\exp\Big(-\frac{a^2}{2v+2ab/3}\Big),
\end{equation}
for a martingale $S_n$ with increments at most $b$ and predictable quadratic variation $V_n$, which follows from \cite[(4.1)]{Freedman1975} by scaling and Bennett's bound $(1+u)\log(1+u)-u\ge u^2/(2(1+u/3))$. Take
$v=\bar\mu$ and $b=1$. The second form in \eqref{eq:landing} follows from $\sqrt{2\bar\mu\lambda}\le\bar\mu+\lambda/2$.
\end{proof}

\begin{remark}\label{rem:oldlanding}
% v3: §4.2, Remark after Lemma 4.1
A bound of the form $\Lambda(J)\le|J|(1+|J|/H)+\lambda$ with failure probability $e^{-\lambda}$ would need $\lambda\ge3.92\mu$,
which is false for $\lambda\lesssim|J|$; it would also condition on later decisions. It is not used.
\end{remark}

\paragraph{Landing windows.} Let $\mathcal J_P$ be the set of windows $[P+i/2,\,P+j/2)$ with integers $0\le i<j$,
$j-i\le12H(P)$ and $P+j/2\le4P+1$. There are at most $(6P+3)\cdot12H(P)\le73P\,H(P)$ of them. Every $J\subset[P,4P]$ with
$|J|\le6H(P)-1$ is contained in a smallest $J^+\in\mathcal J_P$, and $|J^+|\le|J|+1$.

\begin{theorem}[Queue bound]\label{thm:queue}
% v3: Theorem 4.3
Let $x$ be a dyadic scale. The drift bound R1$'$ is used in the form of Corollary~\ref{cor:r1}, on every window (slot or physical)
in $[1,\infty)$, with $L:=\log(2x+12\log x+2)$; this exceeds $\log x$ by less than $0.7$, which R1$'$'s slack absorbs.
\begin{enumerate}[label=\textup{(\roman*)}]
\item For each $I\in\mathcal I_x$, $\Prob(\qFr(I)^c\cap\Good(<t(I)))\le2(K_0+6)e^{-\lambda}$, with $t(I)=b$. Hence
  $\sum_{I\in\mathcal I_x}\Prob(\qFr(I)^c\cap\Good(<t(I)))\le30L(L_0+1)(K_0+6)x^{-3}\le x^{-2}$.
\item (Deterministic part.) Put $t_j:=\lceil s_{j+1}\rceil+1$, the time of the queue constraint at slot $j$
  (Definition~\ref{def:good}). On $\Good(<t_j)\cap\bigcap_{I\in\mathcal I_x,\,t(I)\le t_j}\qFr(I)$, with $s_j\in[x,2x)$,
\begin{equation}\label{eq:queueB}
Q_j\le B(x):=\frac{4M_1L^2}{c'}+\frac{16M_1L}{3}+4\sqrt{M_1L(K_d+4)}+8M_1L\Big(\frac L\Lambda+1\Big)+K_d+9.
\end{equation}
\end{enumerate}
In particular $B(x)\le5M_1L^2/c'\le34M_1L^2$. So $B\le17L^2\qHstd^{1/2}$ off the raised range (where $M_1\le\tfrac12\qHstd^{1/2}$), and
$B\le X_0^{7.6}L^2$ on it (Theorem~\ref{thm:caplw}). Moreover $B\le H$ at every scale: $\log M_1\le\tfrac12\log H$ since
$H\ge M_1^2$, and $34L^2\le\qHstd^{1/2}=e^{4\ell^2}$, so $B\le(M_1^2\qHstd)^{1/2}\le H$.
\end{theorem}

\frac{c'(K_0-1)}{L}\ge\sqrt{16M_1L(K_0+4+K_d)}+\frac{16M_1L}{3}+8M_1L\Big(\frac L\Lambda+1\Big)+L^3+K_d+10.
\end{equation}
Indeed, since $M_1\ge c\ge\tfrac14\qHstd^{1/2}=\tfrac14e^{4\ell^2}\ge L^2$, we have $L_0=48M_1L^3/c'^2$ at every scale, and
$K_0\ge L_0-1$. Then $c'(K_0-1)/L\ge48M_1L^2/c'-2c'/L$ and $\sqrt{16M_1L(K_0+4+K_d)}\le\sqrt{16M_1L(L_0+7)}\le27.72M_1L^2/c'$ (as
$\sqrt{768}\le27.713$ and $L_0\ge10^{40}$), so the left side minus the first term on the right is at least
$20.28M_1L^2/c'-2c'/L\ge135M_1L^2$, which exceeds the rest. So $\qexc(W)\le0$ on full blocks.

\emph{Lindley.} Let $k':=\max\{k<j:\ \qljit((k,j])=K_0\}$, if it exists; as $k$ decreases, $\qljit((k,j])$ grows by steps of
$0$ or $1$, so $k'$ exists whenever some $(k,j]$ has $\qljit\ge K_0$. By \eqref{eq:lindleymax},
\begin{align*}
Q_j&\le\max\Big(Q_{k'}+\qexc((k',j]),\ \max\{\qexc(W):\ W\in\mathcal W_x,\ W=(k,j],\ \qljit(W)<K_0\}\Big)\\
&\le\max(B(s_{k'}),B(x))=B(x).
\end{align*}
This uses: additivity of $\qexc$ over slots; $Q_{k'}\le B(s_{k'})$ on $\Good(<t_j)$ (trivially if $k'=j_0-1$, as $Q_{j_0-1}=0$), which holds because
$t_{k'}\le s_{k'+1}+2\le s_j$ (the $K_0$ continuous slots of $(k',j]$ have starts at least $0.99$ apart, so
$s_j-s_{k'+1}\ge0.99(K_0-1)\ge0.99(L^5-2)\ge12L+2$); and that $B$ is non-decreasing
across scales, because $M_1$ and $H$ are non-decreasing in $x$ (Theorem~\ref{thm:caplw}(c)). A window $(k,j]$ with
$\qljit>K_0$ splits at $k'$.
\end{proof}

\begin{remark}[The certified horizon]\label{rem:horizon}
For $z\ge X_0$ put $\check z:=z-12\log z-3$. Slots are short: by \eqref{eq:G1}, and as $\kappa u/\log u\ge X_0^{0.6}/\Lambda\ge10^4$

\begin{enumerate}[label=\textup{(\roman*)}]
\item $Q_j\le B(s_j)$ for every $j$ with $s_{j+1}\le z-2$; in particular for every $j$ with $s_j\le\check z$, since then
  $s_{j+1}\le\check z+11.2\log z<z-2$. A slot with $s_j\in(\check z,z)$ may have $t_j\ge z$, and then its queue bound is not
  certified on $\Good(<z)$.
\item At every $x\le\check z$ the lag bound \eqref{eq:lag} holds with $\max_{s_j\le x}Q_j\le B(x)$ (it reads only slots with
  $s_j\le x$, and $B$ is non-decreasing), and so do the start-up bound and \eqref{eq:La} of Proposition~\ref{prop:La}, for

  at every $x\le\check z$, and (LP-a) of Theorem~\ref{thm:lp} holds for every $y\le\check z$.
\item $H(u)/2\ge\frac12\exp(8(\log\log u)^2)\ge12\log(2u)+5$ for $u\ge\Xm/2$, by \eqref{eq:jitter}. So every height $\le0.9z$, and
  every height $\le z-H(v)/2$ with $v\ge z/2$, is at most $\check z$.
\end{enumerate}
On a good event $\Good(<z)$ these bounds are used only at heights $\le\check z$, so no event time changes. The heights read are
of the two kinds in (iii). They are $\le z-H(v)/2$ in \ref{eq:sF3p}, \ref{eq:sF4} and Appendix~\ref{app:rankfull} (heights
$\le s-H(s)$, or $Y$ with $Y+H(Y)\le s$), in Lemma~\ref{lem:noise} (on $\Good(<y+H(y))$, slots up to $y+H(y)/2$), in Proposition~\ref{prop:g1}(B) (the local
prime number theorem up to $x(1-\delta)\le x-x^{0.98}$, and the local count on the window that ends there), and in
Lemma~\ref{lem:goodheights} (on $\Good(<x_k+H(x_k))$, heights $\le x_k+2$, with $x_k$ the construction scale, written $a_k$ in
Section~\ref{sec:zeros}). They are $\le0.9z$ in all other uses, for instance in Lemmas~\ref{lem:threshold-mass}, \ref{lem:tau-near}, \ref{lem:S},
\ref{lem:margin}, \ref{lem:lemmaP}, \ref{lem:xb}, \ref{lem:short-recip}, \ref{lem:si}, Propositions~\ref{prop:true-kernels},
\ref{prop:xb-ranges}, and in the variance bound of the events E3, E3$'$; these contain the other uses of
Corollary~\ref{cor:localcount} (Remark~\ref{rem:localcount-uses}).
\end{remark}

\subsection{The cluster event and its reduction}\label{sec:queue-ecl}
% v3: §4.4

Fix a dyadic height $z\ge X_0/2$; this covers the band $[\Xm,X_0)$ of centred placement. Put $L:=\log z$, and let $J$ be a grid
window at height $z$, so $|J|=2\delta_1=4L^{-6}<1$. Since $|J|<1$, $J$ contains at most one element with no jittered prime.
Write $S(J)$ for the set of elements of $\NN$ in $J$ that have a jittered prime. For $n=M\cdot d$, with $M$ the jittered part
and $d$ the rigid ($X_0$-smooth) part, the \emph{exponent vector} $f_n\in\Z_{\ge0}^{\mathcal P_{\mathrm{jit}}}$ of $n$ is
that of $M$. \emph{Rank}, \emph{span} and \emph{degree} $\qdeg f:=\|f\|_1$ refer to these vectors.
% Logged correction 75 (CORRECTIONS-vs-v3.md): v3 has D := log 2z/Lambda (corrected to log 2z/log X_0^- by #43); the
% elements of J are below 2z+1, which may exceed (X_0^-)^D when D is just below an integer.
Since $J\subset[z,2z+1]$, every degree is at most $D:=\log(2z+1)/\log\Xm$, which equals $\log2z/\Lambda$ up to a relative
error at most $e^{-0.98\Lambda}$.
\begin{itemize}
\item \emph{(V1) One element per vector.} Two elements of $J$ with the same exponent vector $f$ have the same jittered part
  $M_f\ge\Xm$. They differ by $M_f|d-d'|\ge\Xm>|J|$ unless $d=d'$. So distinct elements of $S(J)$ have distinct exponent
  vectors.
\item \emph{Capacity.} For $S'\subset S(J)$ of rank $r$, put $\qCap(S'):=|S'|-r$.
\end{itemize}

\begin{proposition}[The cluster event]\label{prop:ecl}
% v3: Proposition 4.6
Let $\qEcl(z)$ be the event that every grid window $J$ at height $z$ satisfies $\#(\NN\cap J)\le M_1(z)$. Then, with
$t_z:=2z+H(2z)$,
\[
\Prob(\qEcl(z)^c\cap\Good(<t_z))\le L^{-2}.
\]
The event $\qEcl^{(u)}:=\bigcap_{X_0/2\le z\le4u/L_u^{11}}\qEcl(z)$ (over dyadic $z$) is the event E1 of Definition~\ref{def:good}. Its
new heights at scale $u$ lie in $(2u/L_{u/2}^{11},\,4u/L_u^{11}]$, which contains at most one dyadic $z$, and that $z$ has
$\log z\ge L_u-11\log L_u+\log2$. So its failure probability on the good event is at most
$(\log z)^{-2}\le L_u^{-2}(1+23\log L_u/L_u)\le1.0001L_u^{-2}$.
% Logged correction 68: v3 has L_u^{-2}(1+o(1)); the factor is now explicit.
\end{proposition}

\begin{proof}
% Logged correction 85 (R21, 85-1): M_1 is real, so the failure of E_cl is #(N cap J) > M_1, not >= M_1 + 1 (same class as Lemma heavy (ii)).
Suppose $\#(\NN\cap J)>M_1$. Since $J$ contains at most one element with no jittered prime, $|S(J)|>M_1-1=r_0+\varphi(r_0)$
($M_1$ is real, so the complement of $\#(\NN\cap J)\le M_1$ is $\#(\NN\cap J)>M_1$). Let $r:=\operatorname{rank}S(J)$.
\begin{itemize}
\item \emph{Low heights, $2z+1<(\Xm)^2$}, in particular every $z\in[X_0/2,X_0)$. Every element of $S(J)$ is $q\cdot d$ with
  exactly one jittered prime $q$, because two jittered primes give a product at least $(\Xm)^2>2z+1$. (Heights $z\ge X_0$ are in
  any case covered by the general argument below.) So the exponent vectors are unit vectors, distinct by (V1). Hence
  $\operatorname{rank}S(J)=|S(J)|$ and $\qCap=0$, so the case $r<r_0$ below is vacuous. The case $r\ge r_0$ is handled by
  Theorem~\ref{thm:R}, exactly as below; it is stated for $z\ge X_0/2$, and its bound includes the sums over the rigid
  cofactors $d$.
% Logged correction 75 (CORRECTIONS-vs-v3.md): the condition was 2z < (X_0^-)^2; the elements of J are below 2z+1.
% Logged correction 71 (CORRECTIONS-vs-v3.md): v3 calls this case "the direct (F1)-(F2) count E <= (1.01|J|)^{r_0}/r_0!",
% which omits the sums over the rigid cofactors d (an element is q*d); Theorem R, which includes them, is what is used.
\item \emph{Case $r\ge r_0$.} $S(J)$ contains a rank-full $r_0$-subset. By Theorem~\ref{thm:R} with $k=r_0$ and
  $\delta=|J|=4L^{-6}$, the expected number of rank-full $r_0$-subsets of $\NN\cap J$ on $\Good(<t_z)$ is at most
  $(e\,\theta A_R\delta)^{r_0}e^{(3H(X_0)+2)r_0/X_0}(1+r_0/X_0)^{1.02r_0}e^{2.2r_0/\Lambda}$; the factor $e^{(3H(X_0)+2)r_0/X_0}$
  is absorbed in the per-element factor $1.01$. There are at most $2z/\delta+1$ windows, since their left ends lie in
  $[z,2z)$. The total is at most $L^{-4}$ for $L\ge465$, by Corollary~\ref{cor:R-Ecl}.
  % Logged correction 75 (CORRECTIONS-vs-v3.md): v3 counts 2z/delta windows.
  % Approved deviation from v3 (CORRECTIONS-vs-v3.md, #1): v3 has e^{2r_0/X_0} in place of e^{(3H(X_0)+2)r_0/X_0}.
  % Logged correction 55: the factor e per element (k^k/k! <= e^k in Theorem R).
\item \emph{Case $r<r_0$.} Then $|S(J)|>r_0+\varphi(r_0)\ge r+\varphi(r)$. So $J$ is a window as in
  Theorem~\ref{thm:caplw}(a): with $V$ a heavy subspace of $S(J)$ of least dimension, Lemma~\ref{lem:heavy} gives
  $\qCap(S(J)\cap V)\ge c$. By Theorem~\ref{thm:caplw}(a), the probability that some window at height $z$ is of this kind
  is at most $L^{-3}$.
\end{itemize}
Summing the two cases gives the claim.
\end{proof}

\begin{corollary}[Per-prime increment bound]\label{cor:increment}
% v3: Corollary 4.7
% Logged correction 79 (CORRECTIONS-vs-v3.md): v3 states this for I in [u,2u+12L_u+2] and q >= y(u), but Theorem queue(ii)
% applies it to windows of I_x that begin below x, where the non-absorbed arrivals below x only have q >= y(x/2). Extended to
% I in [u/2,2u+12L_u+2], |I| <= 2H(u) (which also covers the windows of length 1.03H(u)+1 of Lemma noise) and q >= y_-(u).
% Logged correction 83 (CORRECTIONS-vs-v3.md): stated for jittered q only (every consumer counts jittered latest primes).
Let $u\ge X_0$ be a height, and put $y_-(u):=y(\max(u/2,X_0))$, the absorption threshold of the dyadic block of
$\max(u/2,X_0)$; so $y_-(u)=y(u/2)$ for $u\ge2X_0$. On $\qEcl^{(u)}$ the following holds for every window
$I\subset[u/2,\,2u+12L_u+2]$ (slot window or physical window) with $|I|\le2H(u)$, every jittered latest prime
$q\ge y_-(u)$, every $a\ge1$ and \emph{every} value of the jitter offset of $q$:
\[
X^{(a)}_q(I):=\#\{n\in I\setminus\Abs:\ q^a\,\|\,n,\ q\text{ latest}\}\le2M_1(u).
\]
Moreover, every $n\in I\setminus\Abs$ whose latest prime $q$ is jittered has $q\ge y_-(u)$. The admissible windows include
those of $\mathcal I_u$ (Section~\ref{sec:queue-bound}), which may begin below $u$, and the windows $W\subset[u,2u]$ of
length $\le1.03H(u)+1$ of Section~\ref{sec:noise-perscale}.
\end{corollary}

\begin{proof}
Put $L:=L_u$, and let $x'$ be the dyadic scale of $\max(u/2,X_0)$, so $x'\ge u/4$ and $y_-(u)=y(x')$.

\emph{The threshold.} Let $n\in I\setminus\Abs$ have jittered latest prime $q$. If $n\ge X_0$, then $\log n\ge\Lambda>L_*$, so
$q\ge y(n)\ge y(x')$, as $n\ge\max(u/2,X_0)$ and $y$ is non-decreasing across scales (Theorem~\ref{thm:caplw}(c)). If $n<X_0$,
then $u<2X_0$, and $n$ is the jittered prime $q$ itself, since a composite with a jittered prime is at least $2\Xm>X_0$; and
$q\ge\Xm>y(X_0)=y_-(u)$ by Theorem~\ref{thm:caplw}(b).

\begin{lemma}[Heavy subspace]\label{lem:heavy}
% v3: Lemma 4.8 (Participation+)
Let $S\subset\Z^m_{\ge0}\setminus\{0\}$ be finite, of rank less than $r_0$, with $|S|\ge r_0+\varphi(r_0)$. Call a subspace
$W$ spanned by elements of $S$ \emph{heavy} if $\#(S\cap W)\ge\varphi(\dim W)$. Let $V$ be a heavy subspace of least
dimension $r$. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\qCap(S\cap V)\ge\varphi(r)-r\ge c$;
\item every proper subspace $W'\subsetneq V$ spanned by elements of $S$ satisfies $\#(S\cap V\setminus W')>Q+r$;
\item if the elements of $S\cap V$ of degree at most $k$ number at most $Q$, then the elements of $S\cap V$ of degree at
  least $k+1$ span $V$.
\end{enumerate}
\end{lemma}

\begin{proof}
$\operatorname{span}S$ is heavy, since $\varphi$ is increasing and $|S|\ge\varphi(r_0)\ge\varphi(\operatorname{rank}S)$. So
$V$ exists, and (i) follows from $\varphi(r)\ge c+r$.

% Logged correction 85 (CORRECTIONS-vs-v3.md): v3 concludes #(S cap W') <= phi(dim W') - 1, but phi is real-valued; the correct
% conclusion is the strict #(S cap V minus W') > Q + r (counterexample to >= Q+r+1: c = 100.2, r_0 = 4, 155 vectors in Z^3). (iii) needs only > Q.
(ii) $W'$ is not heavy, by minimality of $V$, so $\#(S\cap W')<\varphi(\dim W')\le\varphi(r-1)$ ($\varphi$ is real-valued, so
non-heaviness gives only this strict inequality); and $\#(S\cap V)\ge\varphi(r)$, $\varphi(r)-\varphi(r-1)=Q+r$. Hence
$\#(S\cap V\setminus W')>\varphi(r)-\varphi(r-1)=Q+r$.

(iii) Otherwise the elements of high degree lie in a proper element-spanned subspace $W'$, so $S\cap V\setminus W'$
consists of elements of low degree only, of which there are at most $Q\le Q+r$. This contradicts (ii).
\end{proof}

\subsection{Capacity forces thin covers}\label{sec:queue-cover}
% v3: §4.5 (deterministic combinatorics)

Throughout this subsection $S\subset\Z^m_{\ge0}\setminus\{0\}$ is finite, of rank $r$, and every vector in $S$ has
$\ell^1$-norm at most $D$. Put $P(S):=\bigcup_{v\in S}\qsupp v$. A \emph{cover} is a subset $C\subset S$ with
$\bigcup_{y\in C}\qsupp y=P(S)$, and $\kappa(S)$ is the least size of a cover. For a basis $B\subset S$ and $y\in S\setminus B$
(an \emph{extra}), write $e_y=\sum_{i\in B}\alpha_ie_i$, $I_+:=\{i:\alpha_i>0\}$, $I_-:=\{i:\alpha_i<0\}$ and
$\qsupp_B(y):=I_+\cup I_-$. No affine-hyperplane assumption is used in this subsection, except in the last statement of
Theorem~\ref{thm:tcl}.

\begin{lemma}[Covering lemma; minimal covers]\label{lem:covering}
% v3: Lemma 4.9
\begin{enumerate}[label=\textup{(\alph*)}]
\item Every basis $B\subset S$ is a cover.
\item If $S$ has an extra $y$ that is not proportional to a basis vector, then $\kappa(S)\le r-1$.
\item A minimal cover $C$ (one with no proper subset that covers) is linearly independent.
\end{enumerate}
\end{lemma}

\begin{proof}
(a) Every $e_y$ is a combination of the $e_i$, $i\in B$, and nonnegativity gives
$\qsupp e_y\subset\bigcup_{\alpha_i\ne0}\qsupp e_i$.

(b) Rearrange as $e_y+\sum_{l\in I_-}|\alpha_l|e_l=\sum_{i\in I_+}\alpha_ie_i$. All coefficients are positive and all vectors
are nonnegative, so
\begin{equation}\label{eq:supportid}
\qsupp(y)\cup\bigcup_{l\in I_-}\qsupp(e_l)=\bigcup_{i\in I_+}\qsupp(e_i).
\end{equation}
If $I_-\ne\emptyset$, then $B\setminus I_-$ covers. If $I_-=\emptyset$, then $\{y\}\cup(B\setminus I_+)$ covers, and it has
size at most $r-1$ when $|I_+|\ge2$. The remaining case, $|I_+|=1$ with $I_-=\emptyset$, is the proportional case.

(c) Each $c\in C$ has a prime $p_c$ that lies in no other element of $C$, for otherwise $C\setminus\{c\}$ covers. The
$C\times\{p_c\}$ submatrix is diagonal with positive diagonal.
\end{proof}

\begin{lemma}[Coefficient support]\label{lem:coefsupport}
% v3: Lemma 4.10
If $\kappa(S)\ge r-j$, then for every basis $B$ and every extra $y$ we have $|I_-|\le j$ and $|I_+|\le j+1$. Hence
$|\qsupp_B(y)|\le2j+1$.
\end{lemma}

\begin{proof}
If $|I_-|\ge1$, then $B\setminus I_-$ covers by \eqref{eq:supportid}, so $r-|I_-|\ge\kappa\ge r-j$. In all cases
$\{y\}\cup\{e_l:l\in I_-\}\cup(B\setminus(I_+\cup I_-))$ covers by \eqref{eq:supportid}. It has size $r+1-|I_+|$, so
$|I_+|\le j+1$.
\end{proof}

\begin{theorem}[Capacity forces thin covers]\label{thm:binomial}
% v3: Theorem 4.11
For every $j\ge0$,
\[
\kappa(S)\ge r-j\ \Longrightarrow\ \qCap(S):=|S|-r\le N(r,D,j):=\sum_{s=1}^{2j+1}\binom rs\binom{D+s}s.
\]
Equivalently, $\qCap(S)>N(r,D,j)$ implies $\kappa(S)\le r-1-j$. Also $N(r,D,j)\le e\,(r(D+1))^{2j+1}$.
\end{theorem}

\begin{proof}
Fix a basis $B$. By Lemma~\ref{lem:coefsupport}, each extra lies in $\operatorname{span}\{e_i:i\in I\}$ for some $I\subset B$ with
$|I|=s\le2j+1$. Choose $s$ coordinates on which this span projects injectively. The extras with support $I$ then map
injectively into $\{v\in\Z^s_{\ge0}:\|v\|_1\le D\}$, which has $\binom{\lfloor D\rfloor+s}s\le\binom{D+s}s$ points. Sum over $I$.
\end{proof}

\begin{theorem}[Thin-Cover Lemma]\label{thm:tcl}
% v3: Theorem 4.12
If $\kappa(S)\ge r-1$, then $\qCap(S)\le4r(D+1)^3$. If $S$ lies on an affine hyperplane $w^{\mathsf T}e=1$ not through $0$,
the bound improves to $\qCap(S)\le3r(8D+4)$.
\end{theorem}

This is the case $j=1$ of Theorem~\ref{thm:binomial}, with $O(r)$ supports instead of $\binom r3$. It is recorded for
completeness; Section~\ref{sec:queue-capl} uses Theorem~\ref{thm:binomial}.

\begin{proof}
\emph{Supports have size at most $3$.} This is Lemma~\ref{lem:coefsupport} with $j=1$. A support of size $1$ means that $y$ is
proportional to a basis vector; there are at most $r$ such supports, each holding at most $D+1$ points.

\emph{Exchange rule.} Write $I(y):=\qsupp_B(y)$. Let $I(y)\cap I(y')=\{h\}$ for extras $y,y'$. Put $B':=B-e_h+y$ and
substitute for $e_h$ in $y'$. Nothing cancels, so $|\qsupp_{B'}(y')|=|I(y)|+|I(y')|-1\le3$, and hence
$|I(y)|=|I(y')|=2$. In particular a path of three $2$-supports $\{1,2\},\{2,3\},\{3,4\}$ is impossible: after exchanging
$e_2$ for $y_{12}$ it produces a $3$-support meeting a $2$-support in exactly one point.

\emph{Components.} In the hypergraph of supports of sizes $2$ and $3$, each component on $v_c$ vertices of $B$ is one of:
a star or a triangle of pairs; a book $\{i,j,l_t\}$, possibly together with the pair $\{i,j\}$; or a family inside a $4$-set.
Each has at most $3v_c$ supports, so there are at most $3r$ in all.

\emph{Points per support.} A support of size $s$ contributes at most $(D+1)^s$ points, as in Theorem~\ref{thm:binomial}. So
$\qCap\le r(D+1)+3r(D+1)^3\le4r(D+1)^3$.

\emph{Hyperplane case.} By Lemma~\ref{lem:coefsupport} with $j=1$, $S$ intersected with a plane is weakly convex: no point of $S$
lies strictly inside a triangle of points of $S$. Projecting to $[0,D]^2$ gives at most $4(2D+1)$ points per plane.
\end{proof}

\subsection{High-capacity configurations are rare at every scale}\label{sec:queue-capl}
% v3: §4.6 (CapL-w)

\paragraph{Setting.} Fix a dyadic height $z\ge X_0$, and put $L:=\log z$, $\ell:=\log L$ and $D:=\log(2z+1)/\log\Xm$, as in
Section~\ref{sec:queue-ecl}; every degree is at most $D$, and $L/\Lambda\le D\le2L/\Lambda$.
% Logged correction 43: v3 has D := log 2z/Lambda here, which is not a degree cap.
% Logged correction 75 (CORRECTIONS-vs-v3.md): the windows lie in [z,2z+1] and number at most 2z/delta+1 (v3: 2z/delta);
% hence 2z+1 in D, in (s4), in w_y and in Lemma rowweights, and 2z+delta in the proof of Lemma reduction(ii).
Let $J$ range
over the at most $2z/\delta+1$ grid windows at height $z$, where $\delta:=2\delta_1=|J|$; each lies in $[z,2z+1]$. A \emph{configuration} is a rank-full set
$B'\subset S(J)$ of rank $r<r_0$, together with a minimum cover $C\subset B'$ of $P(B')$, so $|C|=\kappa$. Write $R:=B'\setminus
C$ for the \emph{rows outside the cover}. For $y\in B'$ write $n_y=M_yd_y$, with $M_y$ jittered and $d_y$ rigid. Every
prime of $B'$ lies in $P(C)$, and every row has degree at most $D$, so $|P(C)|\le\pi:=\kappa D$.
% v3 also defines lambda_D here; it is no longer used, since the free primes enter only through the chain recursion.

\paragraph{Skeletons.}
The greedy pivots of $B'$ (Appendix~\ref{app:rankfull}) are $r$ primes of $P(C)$. Since $E_P$ is invertible, some term
of its Leibniz expansion is nonzero; it matches the rows bijectively with the pivots, each row with a pivot occurring in
it. Choose the matching by a fixed rule (say, the lexicographically first permutation with a nonzero Leibniz term), and
index the rows by the decision rank of their pivots, the $i$-th pivot being the $i$-th earliest. Index the primes of $P(C)$ by decision rank, earliest first; these are the \emph{labels}. The \emph{skeleton}
$\mathfrak s$ of a configuration records:
\begin{enumerate}[label=\textup{(s\arabic*)}]
\item $r$, and the indices of the rows of $C$;
\item the labels of the $r$ pivots;
\item the exponents of the pivots in the rows of $C$;
\item for each row $y\in R$, its monomial $M_y$ in the labels, and its tail $d_y$.
\end{enumerate}
The skeleton determines the pivot pattern $E_P$ in decision order, the run structure (two consecutive pivots lie in one
run if and only if their labels are adjacent) and the number of free primes in each block. It does not record the free
primes of the rows of $C$, their exponents in the rows of $C$, or the tails of the rows of $C$; these are summed by
Lemma~\ref{lem:pil}. Skeletons are decision-ordered, and the decision order of the pivots is recorded by~(s2), whose
count is part of Lemma~\ref{lem:reduction}(ii).
% Logged correction 55: v3 uses types "up to relabelling by value" that fix the free structure of the rows of C, while
% the Cover Bound sums over that structure and the PIL needs the decision order. Skeletons fix exactly what the PIL
% needs; the free structure of the rows of C is summed by the variable-content block bound (S3)(a).

\begin{proposition}[Cover Bound]\label{prop:cover}
% v3: Proposition 4.13
For every skeleton $\mathfrak s$, let $X_{\mathfrak s}$ be the number of configurations with skeleton $\mathfrak s$ in
$J$, and let $\bar S_{\mathfrak s}$ be the envelope of Lemma~\ref{lem:pil} (with $t_J:=t_z$). Then
\begin{equation}\label{eq:coverbound}
\E\big[X_{\mathfrak s};\ \Good(<t_z)\big]\le\theta^r\bar S_{\mathfrak s}\le(\theta\delta)^r\bar\Sigma_{\mathfrak s},\qquad
\bar\Sigma_{\mathfrak s}:=(1+\kappa\lambda)^{s_R}\,\frac{\kappa^\kappa}{\kappa!}\,\mathrm{Ch}^{\kappa}\cdot
\prod_{i=1}^{r}\tau_i^{-e_i}\cdot z^{-|R|},
\end{equation}
where:
\begin{itemize}
\item the constant $\mathrm{Ch}:=e^{\gamma}\log(2z+H(2z))\exp(2.21/\Lambda+10^{-9})$ is at least the constant
  $A_{\mathrm{top}}$ of Lemma~\ref{lem:fac}\,(S3), as $5H(X_0)/X_0\le10^{-9}$;
\item $\lambda:=\log(A_{\mathrm{top}}/\mathrm{Eu}_{\mathrm{rat}})$, and $s_R\le|R|$ is the number of runs at which no row of
  $C$ joins (in the staging of Lemma~\ref{lem:SC});
\item $e_i\ge0$ is the total exponent of the $i$-th pivot in the rows of $C$, minus $1$;
\item $\tau_i:=\max(\Xm,\sigma_i-2H(\sigma_i))$ with $\sigma_i:=X_0+0.99(i-1)$, as in \eqref{eq:A-tau}; the $i$-th pivot
  has value at least $\tau_i$. More generally, the label of rank $j$ has value at least $\tau_j$, because the primes of
  $P(C)$ are distinct jittered primes in distinct slots (Section~\ref{sec:A-setting}).
\end{itemize}
% Logged correction 44: v3 weights every excess occurrence uniformly by X_0^{-1}; the pattern sum then needs the
% per-index weights tau_j^{-e_j} of Appendix A.
% Logged correction 55: v3's Sigma-bar has Ch^kappa and the free-prime excess per value-ordered type; the union over the
% free structure of the rows of C costs (1+kappa lambda)^{s_R} kappa^kappa/kappa!, and the free-prime excess is inside the
% chain recursion.
\end{proposition}

\begin{proof}
\emph{From Lemma~\ref{lem:pil}.} The first inequality in \eqref{eq:coverbound} is Lemma~\ref{lem:pil}, applied with
$t_J:=t_z$ (every element of $J$ has all its primes decided before $2z+H(2z)$) and with the data (s1)--(s4) fixed in
advance. Its recursion processes the runs latest first. For each run $P_t$ it (i) records the label sum at $u_t$, with
$\chi_t:=\mathbf 1(u_t\le s_{\max})\mathbf 1(G_{<u_t})$; (ii) integrates the pivots of the run; and (iii) bounds the sum
over the contents of the late block below the run by Lemma~\ref{lem:fac}\,(S3)(a). The chain rows are the rows of $C$,
each starting at the stage where it joins; the covering hypothesis of (S3) is Lemma~\ref{lem:SC}\,(3). The block sums
run over the contents compatible with~(s1)--(s4), and are bounded by the sums over all contents. The entries of the rows
of $R$ in the block columns are fixed by~(s4) and carry no weight. The end step treats the early block and the tails of
the rows of $C$ by (S3)(c). So $\bar S_{\mathfrak s}$ is deterministic.
% Logged correction 48: the order of Lemma pil is now label sum (i), pivot integrals (ii), joint block bound (iii);
% v3 summed each block before the label sum.

\emph{The chain recursion.} Let $K_t$ be the number of rows of $C$ in $\mathcal R_{\le t}$, and $j_t$ the number of rows of
$C$ joining at stage $t$; $K_T=\kappa$. By Lemma~\ref{lem:fac}\,(S3), the free primes of the rows of $C$, in all blocks,
and their tails, contribute at most $\mathrm{Fin}:=G_T(\mathrm{Eu}_{\mathrm{rat}})+\mathrm{Eu}_{\mathrm{rat}}^{\,j_T}
G_{T-1}(\mathrm{Eu}_{\mathrm{rat}})$, with the recursion \eqref{eq:A-Grec}. At a stage with $j_t=0$, since $G_{t-1}$ is
non-increasing and $E\ge\mathrm{Eu}_{\mathrm{rat}}$,
\[
  G_t(E)=\int_E^{A_{\mathrm{top}}}\frac{K_t}xG_{t-1}(x)\,dx\le K_t\lambda\,G_{t-1}(E),
\]
and at the end, if $j_T=0$, $\mathrm{Fin}\le(1+\kappa\lambda)G_{T-1}(\mathrm{Eu}_{\mathrm{rat}})$. Every step of the
recursion is positive and monotone, so these factors can be pulled out. What remains is the recursion of
Theorem~\ref{thm:R} for the composition of $\kappa$ formed by the non-zero $j_t$, with the same $K_t$. By the proof of
Theorem~\ref{thm:R}, its output is at most $F_\kappa(\rho)A_{\mathrm{top}}^\kappa\le(\kappa^\kappa/\kappa!)A_{\mathrm{top}}^\kappa$.
Hence $\mathrm{Fin}\le(1+\kappa\lambda)^{s_R}(\kappa^\kappa/\kappa!)\,\mathrm{Ch}^\kappa$. (Every stage has $K_t\ge1$, by
Lemma~\ref{lem:SC}\,(3) applied to the pivot columns of $P_1$; so no stage has $K_t=0$.)

\emph{Exact identity.} By the Volume Lemma (Lemma~\ref{lem:volume}), the pivot-integral of $\prod_{y\in B'}\mathbf 1(n_y\in J)$
equals $\int_{J^r}\prod_{\mathrm{piv}}q\big/\big(|\det E_P|\prod_yn_y\big)\,dn$.
\begin{itemize}
\item Order the pivots latest first, and stage the rows by Lemma~\ref{lem:SC} (nested lexicographically-first bases with the
  rows of $C$ listed first).
\item The Staged Volume Lemma (Theorem~\ref{thm:svl}) then integrates the pivots stage by stage. Every late block is
  handled by the block bound (S3)(a), against the weight $\prod_gg^{-a_g}$ of the rows of $C$ of its stage. This is the
  condition that Lemma~\ref{lem:SC} guarantees.
\item $\prod_{\mathrm{piv}}q/\prod_yn_y=\prod_{y\in C}c_y^{-1}\cdot\prod_{y\in R}n_y^{-1}\cdot\prod_{\mathrm{piv}}q^{1-s^C_p}$,
  where $c_y:=n_y/(\text{pivot part of }n_y)$ and $s^C_p$ is the column sum of $p$ over the rows of $C$.
\end{itemize}

\emph{Allocation.} Every prime has an occurrence in some row of $C$, since $C$ covers $P(B')$. So:
\begin{itemize}
\item each pivot $p$ cancels the factor $q_p^{+1}$ of the Jacobian against one occurrence in a row of $C$, and its
  remaining $e_i=s^C_p-1$ occurrences there contribute at most $\tau_i^{-e_i}$, by the caps of Lemma~\ref{lem:pil};
\item the free primes of the rows of $C$ keep their weight $g^{-a_g}$ in the block bounds, and the tails of the rows of
  $C$ their weight $1/d_y$; together they give $\mathrm{Fin}$;
\item a row of $R$ gets $1/n_y\le1/z$; its monomial and tail are fixed by~(s4).
\end{itemize}

\emph{Bookkeeping.} The factor $\theta=1+1/\Lambda$ per pivot comes from (F1)--(F2) and the slot sums, each row's $n_y$ is
integrated over $J$ once, which gives $\delta^r$, and $|\det E_P|\ge1$.
\end{proof}

\begin{remark}\label{rem:jacobianrows}
% v3: §4.6, Remark after Proposition 4.13

\end{remark}

\begin{lemma}[Reduction to configurations; type union]\label{lem:reduction}
% v3: Lemma 4.14
\begin{enumerate}[label=\textup{(\roman*)}]
\item (Extras drop out.) Let $S':=S(J)\cap V$ with $V$ as in Lemma~\ref{lem:heavy}, of rank $r$ and with $\qCap(S')\ge c$. Then
  $\kappa(S')\le r-1-j(c)$, where $j(c):=\max\{j:\ e(r(D+1))^{2j+1}<c\}$ (Theorem~\ref{thm:binomial}). A minimum cover $C$ of $S'$
  is independent (Lemma~\ref{lem:covering}(c)), so it extends to a basis $B'\subset S'$ with $P(B')=P(S')$
  (Lemma~\ref{lem:covering}(a)). So $|R|=r-\kappa\ge1+j(c)$. If moreover the elements of $S'$ of degree at most $k$ number at most
  $Q$, the completion can be chosen with every row of $R$ of degree at least $k+1$ (Lemma~\ref{lem:heavy}(iii) and Steinitz
  exchange).
\item (Type union.) Sum \eqref{eq:coverbound} over skeletons and multiply by the at most $2z/\delta+1$ windows. For $L\ge60$, the result
  is, per height, at most
  \begin{equation}\label{eq:typeunion}
  13L^{5.1}\cdot z\cdot\prod_{y\in R}w_y,\qquad w_y:=z^{-1}\sum_{M_y}\Psi((2z+1)/M_y,X_0),
  \end{equation}
  where $M_y$ runs over the monomials of the degree class of the row in the $\pi$ labels, with the $j$-th label evaluated
  at $\tau_j$, as in \eqref{eq:coverbound}.
\item (Consequence.) The probability that some window at height $z$ contains a configuration with $|R|\ge1+j$ and all
  $w_y\le w$ is at most $r_0\cdot13L^{5.1}z\cdot w^{1+j}$. It is at most $L^{-3}$ as soon as
  \begin{equation}\label{eq:criterion}
  (1+j)\log(1/w)\ge L+10\ell+10.
  \end{equation}
\end{enumerate}
\end{lemma}

\begin{proof}
\emph{Proof of (ii).} Fix $r$. The quantity $\bar\Sigma_{\mathfrak s}$ of \eqref{eq:coverbound} is deterministic. We sum
it over the data (s1)--(s4) of the skeletons:
% Logged correction 55: v3 sums per-type bounds with factors Ch (kept free primes "in all blocks"), D (pivot positions)
% and the excess over pi labels; the free structure was not counted. Now the free structure of the rows of C is summed
% inside Fin (Proposition cover), and the skeleton data are counted here.
\begin{itemize}
\item (s1): at most $2^r$ choices;
\item (s2): the labels of the pivots form an $r$-subset of $\{1,\dots,\pi\}$, so there are at most
  $\binom{\pi}{r}\le(e\pi/r)^r\le(eD)^r$ choices, as $\kappa\le r$;
\item (s3): if the $i$-th pivot is matched to a row of $C$, its exponent there is at least $1$, and the sum of
  $\tau_i^{-e_i}$ over its exponents in the $\kappa$ rows of $C$ is $(1-1/\tau_i)^{-\kappa}$. If it is matched to a row of
  $R$, it still occurs in a row of $C$, and the sum is $\tau_i\big((1-1/\tau_i)^{-\kappa}-1\big)\le\kappa(1-1/\tau_i)^{-\kappa-1}$.
  By the estimate in the proof of Lemma~\ref{lem:pattern} (with $r$ in place of $k$),
  $\prod_{i\le r}(1-1/\tau_i)^{-1}\le B:=e^{(3H(X_0)+2)/X_0}(1+r/X_0)^{1.02}\le2.1L^{1.02}$, as $r\le L$. So (s3)
  contributes at most $\kappa^{|R|}B^{\kappa+1}\le\kappa^{|R|}B^{2\kappa}$;
\item (s4): the label of rank $j$ has value at least $\tau_j$, so a row of $R$, with $n_y\le2z+1$, has
  $M_y(\tau)d_y\le2z+1$, where $M_y(\tau)$ is $M_y$ with the $j$-th label evaluated at $\tau_j$. With the factor $z^{-1}$
  of $\bar\Sigma_{\mathfrak s}$, the sum over $(M_y,d_y)$ is at most $w_y$.
\end{itemize}
In $\bar\Sigma_{\mathfrak s}$ we have $\kappa^\kappa/\kappa!\le e^\kappa$, $\mathrm{Ch}\le e^{\gamma}(L+0.7)(1+10^{-7})\le2L$,
and $1+\kappa\lambda\le L$ for $L\ge60$, since $\mathrm{Eu}_{\mathrm{rat}}\ge1$ gives $\lambda\le\log(2L)$, and
$\kappa\le r-1\le r_0-2\le L/(3\ell)+2$. Also $\kappa\le r\le L$ and $D\le2L/\Lambda\le L$. Distributing the factors over the rows, a row of $C$ costs at most $\theta\delta C'$, with
\[
  C':=2\cdot eD\cdot e\cdot\mathrm{Ch}\cdot B^2\le2e^2\cdot L\cdot2L\cdot4.41L^{2.04}\le131L^{4.04}\le6L^{5.1}\qquad(L\ge60),
\]
and a row of $R$ costs at most $\theta\delta\cdot2\cdot eD\cdot\kappa(1+\kappa\lambda)\cdot w_y\le8e\theta L^{-3}w_y\le w_y$.
Hence the sum is at most
\[
  \Bigl(\frac{2z}\delta+1\Bigr)(\theta\delta C')^{\kappa}\prod_{y\in R}w_y\le(2z+\delta)\,\theta C'\cdot(\theta\delta C')^{\kappa-1}
  \prod_{y\in R}w_y\le13L^{5.1}z\prod_{y\in R}w_y,
\]
since $\kappa\ge1$, $\theta\delta C'=24\theta L^{-0.9}\le1$ for $L\ge60$, and $(2z+\delta)\theta C'\le12.01L^{5.1}z$. The rows are indexed by their pivots, so the
decision order costs nothing beyond~(s2).

\emph{Proof of (iii).} Sum (ii) over the fewer than $r_0$ possible ranks. Since $w\le1$, we have
$\prod_{y\in R}w_y\le w^{|R|}\le w^{1+j}$. As $z=e^L$, $r_0\le L=e^{\ell}$ and $\log13\le2.6$, the bound
$r_0\cdot13L^{5.1}z\,w^{1+j}$ is at most $L^{-3}$ once $(1+j)\log(1/w)\ge L+9.1\ell+2.6$, which \eqref{eq:criterion} implies.
\end{proof}

\begin{lemma}[Row weights]\label{lem:rowweights}
% v3: Lemma 4.15
Let $y\in R$ have degree $k\ge1$, so that $M_y\ge(\Xm)^k$ and $d_y\le(2z+1)/(\Xm)^k$.
% Logged correction 43: X_0 replaced by X_0^- in (a), (b), (d); (d) gets the -e^{-0.4 Lambda} term and the hypothesis on D.
% Logged correction 75 (CORRECTIONS-vs-v3.md): 2z+1 in place of 2z (the windows lie in [z,2z+1]); (a)-(e) keep their
% constants: 2.3 in (b), the -e^{-0.4 Lambda} in g in (d), and the factor 2 in (e) absorb the factor 1+1/(2z).
\begin{enumerate}[label=\textup{(\alph*)}]
\item $w_y\le\sum_{k'\ge k}\binom{\pi+k'-1}{k'}\Psi((2z+1)/(\Xm)^{k'},X_0)/z$.
\item (High degree.) If $e(\pi+k)/(k\Xm)\le1/3$, then $w_y\le3\,(e(\pi+k)/k)^k(\Xm)^{-k}$; that is,
  $\log(1/w_y)\ge k(\log\Xm-1-\log((\pi+k)/k))-\log3$.
\item (Smooth tails.) Put $u:=\log Y/\Lambda$. For $\max(10^4,\log\Lambda)\le u\le X_0^{0.3}$ we have
  $\Psi(Y,X_0)\le Y\exp(-0.9u\log u)$. For $u>X_0^{0.3}$ we have
  \[\Psi(Y,X_0)\le\exp\big(1.26X_0\log(2\log Y)/\Lambda\big).\]
\item (Low degree, smooth tails.) Put $u_1:=\max(10^4,\log\Lambda)$. If $2u_1\le D-1$, $D\le X_0^{0.3}$ and
  $\pi(\Xm)^{-1}(eD)^{0.9}\le1/3$, then $\log(1/w_y)\ge g:=\log\Xm-\log(3\pi)+0.9(D-1)\log(D-1)-e^{-0.4\Lambda}$.
\item (Joint Rankin bound at exponent $\tfrac34$; any $L$.)
  \[
  \log(1/w_y)\ge\frac L4-\log2-\log\Lambda-2-\frac{5.2X_0^{1/4}}{\Lambda}-5.5(X_0+\pi)^{1/4}.
  \]
  % Logged correction 49: v3 states (e) for every eta in [1/4,1); the rigid Euler product diverges as eta -> 1 and its
  % prime-power terms are not O(1) for eta >= 1/2. The only use is eta = 1/4 (Step 4 (T2) of Theorem caplw).
\end{enumerate}
\end{lemma}

\begin{proof}
(a) Count the pairs $(M_y,d_y)$.

(b) Use $\Psi(Y,X_0)\le Y$. The terms $\binom{\pi+k'-1}{k'}\cdot(2+1/z)(\Xm)^{-k'}$ have consecutive ratio
$(\pi+k')/((k'+1)\Xm)\le1/(3e)$, by the hypothesis. So the sum over $k'\ge k$ is at most
$(2+1/z)(1-1/(3e))^{-1}\binom{\pi+k-1}{k}(\Xm)^{-k}\le2.3\binom{\pi+k-1}{k}(\Xm)^{-k}\le3(e(\pi+k)/k)^k(\Xm)^{-k}$.

(c) We use Rankin's bound $\Psi(Y,y)\le Y^{\sigma}\prod_{p\le y}(1-p^{-\sigma})^{-1}$ with $y=X_0$ and $\sigma=1-\eta$, where
$\eta:=\log(u\log u)/\Lambda$. Since $u\le X_0^{0.3}$, we have $\eta\le0.3+\log(0.3\Lambda)/\Lambda\le0.31$, so $\sigma\ge0.69$.
% Logged correction 49: v3 states (c) up to u = X_0^{1/2}, where sigma ~ 1/2 and the prime-power sum R(sigma) exceeds 1;
% the range is cut to u <= X_0^{0.3} (sigma >= 0.69), which covers every use, and the bound is now proved in full.
\begin{itemize}
\item $Y^{\sigma}=Y\exp(-u\log u-u\log\log u)$.
\item Write $\log\prod_{p\le X_0}(1-p^{-\sigma})^{-1}=\sum_{p\le X_0}p^{-\sigma}+R(\sigma)$ with
  $R(\sigma):=\sum_p\big(-\log(1-p^{-\sigma})-p^{-\sigma}\big)$. This is decreasing in $\sigma$, and $R(0.69)\le0.77$ and
  $R(0.75)\le0.63$ (computer-assisted: the sum over $p\le10^7$ in interval arithmetic, plus the tail
  $\sum_{n>10^7}n^{-2\sigma}/(2(1-n^{-\sigma}))$; see Appendix~\ref{app:cert-rankin}).
\item $\sum_{p\le X_0}p^{-\sigma}=\sum_{p\le X_0}1/p+\sum_{p\le X_0}g(p)$ with $g(t):=(t^{\eta}-1)/t$. The first sum is at most
  $\log\Lambda+0.27$ \cite[Thm.~5, (3.18)]{RosserSchoenfeld1962}. The function $g$ decreases for $t\ge t_0:=(1-\eta)^{-1/\eta}$,
  and $t_0\in[e,e^{1.45}]$. Partial summation with $\pi(t)\le1.26t/\log t$ \cite[(3.6)]{RosserSchoenfeld1962}, dropping the
  negative part of $-\int\pi g'$ on $[2,t_0]$, makes the boundary terms at $X_0$ cancel and gives
  \begin{align*}
  \sum_{p\le X_0}g(p)&\le1.26\Big(\frac{t_0^{\eta}-1}{\log t_0}+\int_{t_0}^{X_0}\frac{t^{\eta}-1}{t\log t}\,dt\Big)
  \le1.26\Big(\frac{\eta}{1-\eta}+\mathrm{Ein}(\eta\Lambda)\Big),\\
  \mathrm{Ein}(x)&:=\int_0^x\frac{e^w-1}w\,dw.
  \end{align*}
  Here $\eta\Lambda=\log(u\log u)\ge11.4$ and $e^{\eta\Lambda}=u\log u$. For $x\ge11$,
  $\mathrm{Ein}(x)\le\mathrm{Ei}(x)\le(e^x/x)(1+2/x)$: integrating by parts twice on $[11,x]$ gives
  $\mathrm{Ei}(x)\le(e^x/x)(1+1/x+2.75/x^2)+134$, and $2.75/x+134x^2e^{-x}\le1$ there.
\item Altogether, with $1.26\eta/(1-\eta)\le0.6$,
  \begin{align*}
  \log\prod_{p\le X_0}(1-p^{-\sigma})^{-1}&\le\log\Lambda+0.27+0.6+1.26\,\frac{u\log u}{\log(u\log u)}\Big(1+\frac2{\log(u\log u)}\Big)+0.77\\
  &\le\log\Lambda+1.3u+2,
  \end{align*}
  since $\log u\,(1+2/\log(u\log u))\le\log(u\log u)$ when $\log\log u\ge2$.
\item So, for $u\ge\max(10^4,\log\Lambda)$,
  \[
  \log(\Psi/Y)\le-u\log u-u\log\log u+1.3u+\log\Lambda+2\le-0.9u\log u.
  \]
  This needs
  $0.1u\log u+u\log\log u-1.3u\ge\log\Lambda+2$, which holds there. (At $u=10^4$ alone it would need
  $\log\Lambda\le1.84\cdot10^4$; this is why the lower limit is $\max(10^4,\log\Lambda)$.)
\item For $u>X_0^{0.3}$: an $X_0$-smooth $d\le Y$ has at most $1+\log Y/\log2$ choices of exponent for each of the at most
  $1.26X_0/\Lambda$ primes $p<X_0$.
\end{itemize}

(d) The $k'$-th term of (a) has $Y=(2z+1)/(\Xm)^{k'}$, so $u=\log Y/\Lambda=(D-k')\log\Xm/\Lambda$. For $k'=1$ this gives
$(D-1)(1-\eps)\le u\le D-1$ with $\eps:=e^{-0.98\Lambda}/\Lambda$, so $u_1\le u\le X_0^{0.3}$ and (c) applies. Since
$D-1\le e^{0.3\Lambda}$, we have $u\log u\ge(D-1)\log(D-1)-(D-1)\eps(\log(D-1)+2)\ge(D-1)\log(D-1)-e^{-0.48\Lambda}$. So the
$k'=1$ term is at most $(2+1/z)\pi\exp(-0.9(D-1)\log(D-1)+e^{-0.48\Lambda})/\Xm$.
\begin{itemize}
\item Consecutive values of $u$ differ by $\log\Xm/\Lambda\le1$. So, while the next term still has $u\ge u_1$, the ratio
  of consecutive terms is at most $\pi(\Xm)^{-1}(eD)^{0.9}$, which is at most $1/3$ by hypothesis.
\item Once $u<u_1$, bound the term by $\Psi\le Y$ instead. The first such term is at most
  $\pi(\Xm)^{-1}e^{0.9(u_1+1)\log(u_1+1)}$ times the previous one, which is far less than $\tfrac13$, and the
  following ratios are at most $(\pi+k')/((k'+1)\Xm)\le\tfrac13$.
\item So $w_y\le\tfrac32(1+e^{-\Lambda})(2+1/z)\pi\exp(-0.9(D-1)\log(D-1)+e^{-0.48\Lambda})/\Xm$, which gives $g$, since
  $\log(1+e^{-\Lambda})+\log(1+1/(2z))+e^{-0.48\Lambda}\le e^{-0.4\Lambda}$ for $z\ge X_0$. Where (d)
  is used (Steps 3--4 of the proof of Theorem~\ref{thm:caplw}), $D-1\ge\Lambda e^{s_H+1}-1\ge2u_1$ and $D\le2L/\Lambda\le
  X_0^{0.3}$, as $\ell\le0.3\Lambda$ there.
\end{itemize}

(e) Rankin's bound $\mathbf 1(Md\le2z+1)\le((2z+1)/(Md))^{3/4}$ and $(2z+1)^{3/4}\le2z^{3/4}$ give
\[
w_y\le z^{-1}\sum_{(M,d)}\Big(\frac{2z+1}{Md}\Big)^{3/4}\le2z^{-1/4}\prod_{j\le\pi}(1-\tau_j^{-3/4})^{-1}
\prod_{p<X_0}(1-p^{-3/4})^{-1}.
\]
\begin{itemize}
\item The rigid product: its logarithm is $\sum_{p<X_0}p^{-3/4}+R(\tfrac34)$, with $R(\tfrac34)\le0.63$ as in (c).
  Partial summation with $\pi(t)\le1.26t/\log t$ gives
  $\sum_{p<X_0}p^{-3/4}\le1.26\big(X_0^{1/4}/\Lambda+\tfrac34\int_2^{X_0}t^{-3/4}dt/\log t\big)$, and
  $\int_2^{X_0}t^{-3/4}dt/\log t=\mathrm{Ei}(\Lambda/4)-\mathrm{Ei}(\tfrac14\log2)\le4X_0^{1/4}(1+8/\Lambda)/\Lambda+1$, by the
  bound on $\mathrm{Ei}$ in (c) and $\mathrm{Ei}(\tfrac14\log2)\ge-1$. So the logarithm is at most
  $5.05X_0^{1/4}/\Lambda+0.95+0.63\le\log\Lambda+2+5.2X_0^{1/4}/\Lambda$.
\item The labels are evaluated at $\tau_i=\max(\Xm,\sigma_i-2H(\sigma_i))\ge\sigma_i-2H(\sigma_i)\ge\tfrac23\sigma_i
  =\tfrac23(X_0+0.99(i-1))$, since $H(v)\le v/6$. With $(3/2)^{3/4}\le1.3555$ and $\tau_i^{-3/4}\le X_0^{-0.7}$,
  \begin{align*}
  \log\prod_{j\le\pi}(1-\tau_j^{-3/4})^{-1}&\le(1+2X_0^{-0.7})\,1.3555\sum_{i=1}^{\pi}(X_0+0.99(i-1))^{-3/4}\\
  &\le(1+2X_0^{-0.7})\,1.3555\Big(X_0^{-3/4}+\frac{4\big((X_0+\pi)^{1/4}-X_0^{1/4}\big)}{0.99}\Big),
  \end{align*}
  which is at most $5.5(X_0+\pi)^{1/4}$.
\item Since $\log(2z^{-1/4})=\log2-L/4$, the three bounds give the claim. \qedhere
\end{itemize}
\end{proof}

\begin{theorem}[High-capacity configurations at every scale; the capacity parameter]\label{thm:caplw}
% v3: Theorem 4.17 (CapL-w at every scale, uniformly in Lambda >= 10^8)
Let $\Lambda=\log X_0\ge10^8$ be arbitrary (Section~\ref{sec:params}), and let $\theta\in[\tfrac14,\theta_1]$ with
$\theta_1=\tfrac12-10^{-4}$, as in Section~\ref{sec:setup}, so that $\theta_b=\theta/2\ge\tfrac18$. For a height $z\ge X_0$
(not necessarily dyadic) put $\ell:=\log\log z$ and $s:=\ell-2\log\Lambda$, so that $L=\Lambda^2e^s$. Define
\begin{align*}
A(\Lambda,s)&:=\log\big(2\Lambda^3e^{2s}/(2\log\Lambda+s)\big)=\log\big(2L^2/(\ell\Lambda)\big),&
G(\Lambda,s)&:=\frac{2\sqrt{A(A+1)}\,e^{s/2}}{\sqrt{\Lambda-A-1}},\\
\Lambda_s&:=\max(10^8,e^{-s}),& \gamma_*(s)&:=1.3\,G(\Lambda_s,s).
\end{align*}
Let $s_H=5.9745\ldots$ be the root of $\gamma_*(s_H)=\tfrac13-10^{-5}$. Put $\gamma(s):=\gamma_*(s)$ for $s\le s_H$,
$\gamma(s):=\gamma_*(s_H)+(7.5-\gamma_*(s_H))(s-s_H)$ on $[s_H,s_H+1]$, and $\gamma(s):=7.5$ for $s\ge s_H+1$. Define
\[
\log c(z):=\max\big(4\ell^2-\log4,\ \gamma(s)\Lambda\big),\qquad\text{that is,}\qquad
c=\max\big(\tfrac14\qHstd^{1/2},\,X_0^{\gamma(s)}\big).
\]
Then, for every $\Lambda\ge10^8$, parts (a) and (b) hold at every dyadic height $z\ge X_0$, and part (c) holds for all
heights $z\ge X_0$:
\begin{enumerate}[label=\textup{(\alph*)}]
\item (Counting.) The probability, on $\Good(<t_z)$, that some grid window $J$ at height $z$ has
  $\operatorname{rank}S(J)<r_0$ and $|S(J)|\ge r_0+\varphi(r_0)$ is at most $L^{-3}$. For such $J$, let $V$ be a heavy
  subspace of least dimension for $S=S(J)$ (Lemma~\ref{lem:heavy}); then $S':=S(J)\cap V$ satisfies $\qCap(S')\ge c(z)$.
  % Approved deviation from v3 (CORRECTIONS-vs-v3.md, #18): v3 states (a) for any S' of rank < r_0 with Cap(S') >= c.
\item (Absorption compatibility.) If $s\le s_H$, then $y(z)<\Xm=X_0-H(X_0)/2$, so $\Abs$ has no element at height $z$. If
  $s>s_H$, condition \eqref{eq:twelvetwo} of Lemma~\ref{lem:lemmaP} holds at $z$.
\item (Regularity.)
  \begin{itemize}
  \item $c(z)$, $\bar M_1(z)$ and $H(z)=\max(\qHstd(z),\bar M_1(z)^2)$ are continuous, non-decreasing and piecewise $C^1$
    in $z$, with locally finitely many break points; in particular $H$ is absolutely continuous on compact intervals. The per-scale parameters $M_1$, $L_0$, $y$ and $B$ are non-decreasing
    across scales.
    % Logged correction 32: v3 has "c, M1, H = max(H_std, M1^2) and B are non-decreasing in z".
  \item At every $z\ge X_0$, both one-sided derivatives of $\log H$ with respect to $\log z$ are at most
    $16\ell/L+10^{-7}\le4\cdot10^{-6}$. (At a break point, $d\log H/d\log z$ denotes the larger one-sided derivative.) Here $16\ell/L$
    comes from $\qHstd$, and the raise contributes at most $10^{-7}$.
    % Approved deviation from v3 (CORRECTIONS-vs-v3.md, #18): v3 has 4.1e-8 here, from a chain valid only for Lambda = Lambda_s.
    Hence, $H$ being continuous, $H(vw)/H(v)\le w^{4\cdot10^{-6}}$ for $v\ge X_0$ and $w\ge1$ (the regularity
    \eqref{eq:RH}), $\rho_H\le1+1.2\cdot10^{-5}$, and $H(v)/v$ is strictly decreasing.
  \item $\log H(z)-\log\qHstd(z)\le10^{-6}\log z$; so the raise adds nothing to the lag.
  \item $c=\tfrac14\qHstd^{1/2}$ for $\ell\ge\ell_{\mathrm{std}}:=\sqrt{(7.5\Lambda+\log4)/4}$, which is $1.37\cdot10^4$ at
    $\Lambda=10^8$.
  \end{itemize}
\end{enumerate}
Parts (b) and (c) are deterministic statements about the parameters. Their proofs (Steps 5 and 6 below) use only the
definitions of $\gamma$, $c$, $M_1$, $\bar r_0$, $\bar M_1$, $L_0$, $y$, $H$ and $B$, and not part (a). In particular (F3$'$) and $\rho_H$, which use
(c), do not depend on (a).
\end{theorem}

\begin{proof}
\emph{Step 0 (reduction).} Let $J$ and $S'=S(J)\cap V$ be as in (a), so $\qCap(S')\ge c$ by Lemma~\ref{lem:heavy}(i). By
Lemma~\ref{lem:reduction}(i) and Theorem~\ref{thm:binomial}, $S'$ yields a configuration with $|R|\ge1+j$ rows outside a minimum cover, where $j\ge(\log c-1)/(2a)-3/2$ and
$a:=\log(r_0(D+1))$. By Lemma~\ref{lem:reduction}(iii) it suffices to show $(1+j)\log(1/w)\ge L+10\ell+10$ for a common bound $w$
on the row weights. We use:
\begin{itemize}
\item $r_0\le L/(2\ell)$ (as $L\ge\Lambda\ge10^8$) and $L/\Lambda\le D\le2L/\Lambda$;
\item $m:=r_0D\le L^2/(\ell\Lambda)$, so $\log2m\le A$. Also $a\le A-\log2+\log(1+1/D)\le A$, since $D\ge1$, and
  $a\le A-\log2+10^{-9}$ when $D\ge10^9$ (as in Step 3);
\item $A\le2\ell$, and $\log c\ge4\ell^2-\log4\ge3.9\ell^2$ (since $\ell\ge\log\Lambda\ge18.42$).
\end{itemize}

\emph{Step 1 ($G$ is decreasing in $\Lambda$).} At fixed $s$, $\partial A/\partial\Lambda=(3-2/\ell)/\Lambda\in(0,3/\Lambda)$.
So
\[
\partial_\Lambda\log G^2\le\frac3\Lambda\Big(\frac1A+\frac1{A+1}\Big)-\frac{1-3/\Lambda}{\Lambda-A-1}<0,
\]
because $A\ge\log(2\Lambda/\ell)\ge15.3>6$ (as $\ell\le2\log\Lambda+s_H$ for $s\le s_H$; at $\Lambda=10^8$ this is
$\log(2\cdot10^8/42.82)=15.357$, and it increases with $\Lambda$). Every admissible pair has $\Lambda\ge\Lambda_s$, since $z\ge X_0$ means
% Logged correction 72 (CORRECTIONS-vs-v3.md): v3 has "log(2 Lambda/ell) >= 16", which fails at Lambda = 10^8 (15.357);
% only A > 6 is used.
$s\ge-\log\Lambda$. Hence $\log c\ge\gamma_*(s)\Lambda\ge1.3\,G(\Lambda,s)\Lambda$ for every $s\le s_H$. Moreover $\gamma_*$
is non-decreasing in $s$, since at fixed $\Lambda$ both $e^{s/2}$ and $A$ increase with $s$ ($\partial_sA=2-1/\ell>0$, and $G$ increases with
$A$), while $\Lambda_s$ is non-increasing in $s$ and $\Lambda\mapsto G$ decreases.

\emph{Step 2 (rows of high degree: $s\le s_H+1$).} Recall $Q=c^{1/2}$.
\begin{itemize}
\item The vectors of degree at most $k$ in $m$ coordinates number at most $\binom{m+k}k\le\exp(k(1+\log2m))$. So the elements
  of degree at most $D^*:=\lfloor\log Q/(1+\log2m)\rfloor$ number at most $Q$.
\item If $D^*\ge D$, then $\qCap\ge c$ is impossible, by Lemma~\ref{lem:heavy}(iii) and (V1). Otherwise Lemma~\ref{lem:reduction}(i)
  completes the cover with rows of degree at least $k:=D^*+1\ge\log c/(2(A+1))$. If no such row exists,
  Lemma~\ref{lem:heavy}(iii) gives a contradiction; this includes the case $D^*=D-1$ with $D$ an integer.
\item Lemma~\ref{lem:rowweights}(b) applies, since $\pi\le m\le\Lambda^3e^{14}$ and $k\le D\le2\Lambda e^{7}$, so
  $e(\pi+k)/(k\Xm)\le e(\Lambda^3e^{14}+2\Lambda e^7)/\Xm\le1/3$. It gives
  $\log(1/w)\ge k(\log\Xm-A-1)-\log3\ge k(\Lambda-A-1)-1.1$, since $(\pi+k)/k\le2m$, $\log2m\le A$ and
  $k(\Lambda-\log\Xm)\le De^{-0.98\Lambda}\le10^{-3}$ (here $D\le2\Lambda e^{s_H+1}$).
  % Logged correction 43: the k e^{-0.98 Lambda} loss from X_0^- is absorbed in -1.1 (log 3 = 1.0986).
\item Therefore, using $(A+1)/\log c\le(2\ell+1)/(3.9\ell^2)\le0.03$,
  \[
  (1+j)\log(1/w)\ge\frac{(\log c)^2(\Lambda-A-1)}{4A(A+1)}(1-0.031)\ge1.69\cdot0.969\cdot\Lambda^2e^s\ge1.63L\ge L+10\ell+10,
  \]
  as soon as $\log c\ge1.3\,G(\Lambda,s)\Lambda$.
\item This holds for $s\le s_H$ by Step 1. On $[s_H,s_H+1]$ it holds because $\gamma(s)\ge\gamma_*(s)$ there: $\Lambda_s=10^8$ on this range, and $G(10^8,\cdot)$
  grows at logarithmic rate at most $\tfrac12+2/A+10^{-7}\le0.532$ (as $A\ge64.1$ there), so
  $\gamma_*(s_H+t)\le\gamma_*(s_H)e^{0.532t}\le\gamma_*(s_H)+0.24t\le\gamma(s_H+t)$ for $t\in[0,1]$.
\end{itemize}

\emph{Step 3 (Dickman rows: $s\ge s_H+1$).} Here $\log c\ge7.5\Lambda$. We have $D-1\ge\Lambda e^s-1\ge2u_1$, and for
$\ell\le0.3\Lambda$ the other hypotheses of Lemma~\ref{lem:rowweights}(d) hold: $D\le2L/\Lambda\le X_0^{0.3}$ and
$\pi(\Xm)^{-1}(eD)^{0.9}\le e^{0.87\Lambda-\Lambda+1}$. So $\log(1/w)\ge g\ge0.9\Lambda e^s(\log\Lambda+s)(1-10^{-9})$; the
term $-e^{-0.4\Lambda}$ in $g$ is absorbed by the factor $1-10^{-9}$, and $\log\Xm-\log(3\pi)\ge0$. Also $1+j\ge(\log c-1-a)/(2a)\ge(7.5\Lambda-1-a)/(2a)$, with
$a\le a_+:=3\log\Lambda+2s-2.9\le0.6\Lambda$ (as $\ell\le0.3\Lambda$). Hence, since the bound decreases in $a$,
\[
\frac{(1+j)g}{L}\ge0.45\,(1-10^{-9})\Big(7.5-\frac{1+a_+}{\Lambda}\Big)\frac{\log\Lambda+s}{a_+},\qquad
\frac{\log\Lambda+s}{a_+}=\frac13+\frac{s+2.9}{3a_+}.
\]
If $a_+\le10^{-3}\Lambda$, this is at least $0.45\cdot7.498/3\ge1.12$. Otherwise $s+2.9\ge(a_+-3\log\Lambda)/2$, so
$(\log\Lambda+s)/a_+\ge\tfrac12-10^{-4}$, and the bound is at least $0.45\cdot6.89\cdot0.4999\ge1.5$. In both cases it
exceeds $1+(10\ell+10)/L$.
Where the standard term dominates $\log c$, $j$ is only larger.

\emph{Step 4 (tails).}
\begin{itemize}
\item (T1) $\ell_{\mathrm{std}}\le\ell\le0.3\Lambda$. Here $\log c=4\ell^2-\log4$ and $j\ge\ell-2$. The hypotheses of Lemma~\ref{lem:rowweights}(d) hold as in Step 3, and it
  gives $(1+j)g/L\ge0.9(\ell-1)(\ell-\log\Lambda-0.01)(1-\Lambda/L)/\Lambda\ge0.9\cdot1.87>1.6$.
\item (T2) $\ell\ge0.3\Lambda$. Lemma~\ref{lem:rowweights}(e) gives
  $\log(1/w)\ge L/4-5.3X_0^{1/4}/\Lambda-5.6(X_0+L^2)^{1/4}-\log\Lambda-3\ge0.24L$, since $L\ge e^{0.3\Lambda}$. Then $j\ge4$
  suffices.
\end{itemize}
So (a) holds at every scale, including the scales $L\ge X_0^2$ of the zeros.

\emph{Step 5 (proof of (b)).}
\begin{itemize}
\item \emph{$s\le s_H$.} The bounds $\log M_1,\log\bar M_1\le\log c+\log2$, $\log H=\max(8\ell^2,2\log\bar M_1)$ and
  $\log L_0\le\log48-2\log c'+\log M_1+3\ell=7.67+\log M_1+3\ell$ (with $c'=0.15$) give $\log y\le3\log c+9\ell+11.14$. The
  constant is $11.14$ when $\qHstd$ dominates ($8\ell^2\le2\log c+2\log4$), and $9.8$ when $\bar M_1^2$ does. If
  $\log c=\gamma\Lambda\le(\tfrac13-10^{-5})\Lambda$, this is at most $\Lambda-2597<\log\Xm$, since
  $9\ell+11.14\le3\cdot10^{-5}\Lambda-2597$ for $\ell\le2\log\Lambda+6$, and $\log\Xm\ge\Lambda-e^{-0.98\Lambda}$. If
  $\log c=4\ell^2-\log4$, it is at most $12\ell^2+9\ell+7\le\Lambda$.
\item \emph{$s\ge s_H$ with $\log c=\gamma\Lambda\le7.5\Lambda$.} Here $\log2y\le3.01\gamma\Lambda$ and
  $\log C_d+4\ell\le1.01\gamma\Lambda$. With $u_0\ge\theta_b\Lambda e^s/(3.01\gamma)$, condition \eqref{eq:twelvetwo} follows from
  $e^s\log(\theta_b\Lambda e^s/(3.01\gamma))\ge107.6\gamma^2$. The left side is increasing in $\Lambda$ and $s$. At
  $\Lambda=10^8$ and $s=s_H+t$ it is at least $393.28e^t(19.19+t)\ge7547$, against $107.6\gamma^2\le107.6\cdot7.5^2=6052.5$, since
  $\gamma\le7.5$ everywhere. Also $u_0\ge100$.
% Logged correction 72 (CORRECTIONS-vs-v3.md): v3 bounds the ramp by 107.6(0.334+7.17t)^2 <= 6053, which fails at t = 1
% (6058.96); the linear form overestimates gamma there, and gamma <= 7.5 gives 6052.5. The factors are now

\item \emph{Standard regime, $\ell\ge\ell_{\mathrm{std}}$.} Here $\log2y\le12.1\ell^2$, the right side is at most
  $16.3\ell^2$, and $u_0\ge e^\ell/(97\ell^2)$. So \eqref{eq:twelvetwo} holds with an enormous margin.
\end{itemize}

\emph{Step 6 (proof of (c)).}
\begin{itemize}
\item \emph{Continuity and monotonicity.} $\Lambda_s=\max(10^8,e^{-s})$ and $\gamma_*$ are continuous; $\gamma$ is continuous
  at $s_H$ and at $s_H+1$ by construction; and $\gamma$ is non-decreasing. So $c(z)=\exp(\max(4\ell^2-\log4,\gamma(s)\Lambda))$
  is continuous and non-decreasing in $z$, and so are $\bar r_0$, $\bar M_1$ and $H$. Each is built from analytic functions of
  $\log\log z$ by finitely many maxima and the three break points $s=-\log10^8,s_H,s_H+1$ of $\gamma$, so each is piecewise
  $C^1$ with locally finitely many break points. The per-scale $M_1=r_0+\varphi(r_0)+1$, $L_0$, $y$ and $B$ are non-decreasing
  across scales, since $r_0$ and $c(x)$ are.
  % Logged correction 32: continuity of c, M1bar, H made explicit; v3 asserts monotonicity only.
\item \emph{Derivative.} Where $H=\qHstd$, $d\log H/dL=16\ell/L$. Where $H=\bar M_1^2>\qHstd$, we have
  $c\ge\bar M_1/2>\tfrac14\qHstd^{1/2}$, so we are on the raised range and $\log c=\gamma(s)\Lambda$. There $\bar M_1\ge c$,
  $d\bar r_0/dL\le1/(3\ell)$, $\partial\bar M_1/\partial\bar r_0=\tfrac32+c^{1/2}+\bar r_0\le1.01c^{1/2}$ and
  $\partial\bar M_1/\partial c=1+\bar r_0/(2c^{1/2})$, so
  \[
  \frac{d\log H}{dL}=2\,\frac{d\log\bar M_1}{dL}\le2\big(1+\bar r_0c^{-1/2}\big)\frac{\Lambda\gamma'(s)}L+\frac{0.7}{\ell c^{1/2}}.
  \]
  Here $\bar r_0c^{-1/2}\le2Le^{-2\ell^2}\le e^{-600}$ and $0.7/(\ell c^{1/2})\le e^{-600}$, because $c\ge\tfrac14e^{4\ell^2}$
  and $\ell\ge18.42$. On the ramp $2\Lambda\gamma'(s)/L\le2\cdot7.2/(\Lambda e^{s_H})\le4\cdot10^{-10}$; for $s\ge s_H+1$,
  $\gamma'=0$; for $s\le s_H$ the next item applies.
  % Logged correction 32: the r0bar term 0.7/(l c^{1/2}) is new.
\item For $s\le s_H$, where $\gamma_*\Lambda\ge4\ell^2-\log4$ (so that $\gamma_*$ dominates the standard term), the raise
  contributes $2\Lambda\gamma_*'(s)/L=2\kappa\gamma_*\Lambda/L$, with $\kappa:=d\log\gamma_*/ds$. Put
  $\ell_s:=2\log\Lambda_s+s$ and $A_s:=A(\Lambda_s,s)$; then $18.42\le\ell_s\le\ell$, since $\Lambda_s\le\Lambda$ and
  $s\ge-\log\Lambda$. Dominance gives $1/\Lambda\le\gamma_*/(4\ell^2-\log4)$, so
  \[
  \frac{\gamma_*\Lambda}L=\frac{\gamma_*}{\Lambda e^s}\le\frac{\gamma_*^2}{(4\ell^2-\log4)e^s}
  =\frac{6.76A_s(A_s+1)}{(4\ell^2-\log4)(\Lambda_s-A_s-1)}\le\frac{1.7A_s(A_s+1)}{\ell_s^2\Lambda_s}.
  \]
  \begin{itemize}
  \item If $s\ge-\log10^8$, then $\Lambda_s=10^8$ and $\ell_s\le42.82$. Here
    $\kappa=\tfrac12+\tfrac12A'(\tfrac1A+\tfrac1{A+1})+\tfrac12A'/(\Lambda_s-A-1)$ with $A'=2-1/\ell_s$, so
    $\kappa\le\tfrac12+2/A_s+10^{-7}\le0.63$, as $A_s\ge16.2$. Also $A_s\le2\ell_s-20.6$, so
    $A_s(A_s+1)/\ell_s^2\le(2-19.6/42.82)^2\le2.38$. The contribution is at most
    $2\cdot0.63\cdot1.7\cdot2.38\cdot10^{-8}\le5.1\cdot10^{-8}$.
  \item If $s<-\log10^8$, then $\Lambda_s=e^{-s}=e^{\ell_s}$ and $A_s=\log2+\ell_s-\log\ell_s$, so $A_s+1\le\ell_s$. The
    variation of $\Lambda_s$ adds at most $\tfrac12(1+10^{-6})$ to the logarithmic rate, while the $A$-terms are now
    non-positive, so $\kappa\le1+10^{-6}$. The contribution is at most $2(1+10^{-6})\cdot1.7\cdot e^{-\ell_s}\le3.5\cdot10^{-8}$.
  \end{itemize}
  So $2\Lambda\gamma_*'(s)/L\le5.1\cdot10^{-8}$ for $s\le s_H$.
\item \emph{Conclusion.} The pieces are $C^1$ up to their break points, so the bounds above hold for both one-sided
  derivatives at every point:
  $d\log H/dL\le\max\big(16\ell/L,\ (1+e^{-600})\,5.1\cdot10^{-8}+e^{-600}\big)\le16\ell/L+10^{-7}$, and
  $16\ell/L\le16\log\Lambda/\Lambda\le3\cdot10^{-6}$. Since $H$ is continuous and piecewise $C^1$, integration gives
  $\log H(vw)-\log H(v)\le4\cdot10^{-6}\log w$ for $v\ge X_0$ and $w\ge1$, which is \eqref{eq:RH}; in particular
  $\rho_H\le16^{4\cdot10^{-6}}\le1+1.2\cdot10^{-5}$. Also $d\log(H(v)/v)/d\log v\le4\cdot10^{-6}-1<0$, so $H(v)/v$ is
  strictly decreasing.
\item $(\log H-\log\qHstd)/L\le\max(0,2\log\bar M_1-8\ell^2)/L\le(2\gamma\Lambda+2)/L\le10^{-6}$ at every height, using
  $\bar M_1\le2c$. \qedhere
\end{itemize}
\end{proof}

\subsection{Absorption}\label{sec:queue-abs}
% v3: §4.7 (Lemma P, S1, S2 and the raised threshold)

Recall from Section~\ref{sec:construction} the absorption set
\[
\Abs=\{n\in\NN:\ \log n\ge L_*,\ \text{the latest prime $r$ of $n$ is jittered, and }r<y(n)\},
\]
with $L_*=10^7$ and $y(n):=y(x)=H(x)L_0(x)L(x)^6$, where $x$ is the dyadic scale with $n\in[x,2x)$; that is, $y$ is frozen
per scale, as in Section~\ref{sec:queue-conv}.
% Logged correction 32: one reading of y(n) fixed (frozen per scale).
Put $\theta_b:=\theta/2$
(so $\theta_b=\tfrac18$ for $\theta=\tfrac14$), $h(u):=u^{\theta_b}$ and $\rho_{\Abs}(u):=\#(\Abs\cap[u,u+h])/h$. The target $T$
gets a unit jump at each $n\in\Abs$, and $T'$ gets $-\rho_{\Abs}$ (see \eqref{eq:target}). Membership is decided when $r$ is
decided, at slot approximately $r<y(n)\le n^{0.01}$ (Remark~\ref{rem:caplw-sizes}), so $T$ is adapted.

Every jittered prime is at least $\Xm=X_0-H(X_0)/2$ (centred placement), so \emph{$\Abs$ has no element at heights where
$y<\Xm$.} By Theorem~\ref{thm:caplw}(b) these are all heights with $s=\log\log n-2\log\Lambda\le s_H=5.9745$ (that is,
$\log\log n\le42.82$ at $\Lambda=10^8$). So below that height Lemma~\ref{lem:lemmaP} is vacuous, and, since $\Lambda\ge10^8>L_*$, the condition $\log n\ge L_*$
plays no role.

\begin{lemma}[Pointwise sparsity of $\Abs$]\label{lem:lemmaP}
% v3: Lemma 4.18 (Lemma P)
Let $u\ge X_0$ and $L:=\log u$, and put $y:=\sup_{v\in[u,2u]}y(v)$ and $C_d:=M_1(u)L^6/2$. Suppose
\begin{equation}\label{eq:twelvetwo}
0.9u_0\log u_0\ge\log(2y)+\log C_d+4\log L,\qquad u_0:=\frac{\theta_bL}{\log(2y)},\qquad u_0\ge100.
\end{equation}
Then $\#(\Abs\cap[u,u+h])\le h\,L^{-4}$ on $\Good(<t)$ for every $t\ge t_u:=9u/L^{11}+H(u)$. In particular this holds on
$\Good(<t)$ at heights up to $t+1.01H(t)$, since $9(t+1.01H)/L^{11}+H<t$. Condition \eqref{eq:twelvetwo} holds at every $u$ at
which $\Abs$ meets $[u,2u]$, by Theorem~\ref{thm:caplw}(b). That theorem is proved at dyadic $z$ with $y(z)$; at a
non-dyadic $u$, with $y$ the supremum over the two scales meeting $[u,2u]$, the quantities in \eqref{eq:twelvetwo} differ from
their values at the dyadic scale of $u$ by relative factors at most $1+10^{-5}$ (Lemma~\ref{lem:raisedy} for $y$ on the ramp, and
Theorem~\ref{thm:caplw}(c) for $M_1$, $H$ and, through $L_0$, for $y$ in the standard regime), and $L$ by at most $\log2$. Step 5 of the proof of Theorem~\ref{thm:caplw} has
far more room than this: on the raised range the ratio of its two sides is at least $3.5$ (at $t=1$ on the ramp, and
larger elsewhere), and in the standard regime the margin is larger still.
% Logged correction 66: caplw(b) is proved at dyadic z; the passage to non-dyadic u and two-scale y is made explicit.
\end{lemma}

\begin{proof}
% Logged correction 80 (CORRECTIONS-vs-v3.md): the value bound for primes decided before r is derived with the jitter
% allowance (v3: "decided no later than r, so value less than r+H(r)").
(i) \emph{Smoothness.} Every prime $p$ of $n\in\Abs\cap[u,u+h]$ is decided no later than its latest prime $r<y$, so $p<2y$.
Indeed, a rational $p$ is $<X_0<2\Xm\le2r$. A jittered $p$, decided at a slot $s_p\le s_r$, has
$p\le s_p+H(s_p)/2\le s_r+H(s_r)/2\le r+H(s_r)$, as $v+H(v)/2$ increases and $s_r-H(s_r)/2\le r$; and $H(s_r)\le e^{-10^6}r$, as
$H(v)/v\le e^{-10^7}$ for $v\ge X_0$. So $p\le(1+e^{-10^6})r<2y$.

(ii) \emph{Divisor trick.} Multiply the primes of $n$ in any order and stop at the first partial product $d>h$. Then
$d\in(h,2yh]$ is a $2y$-smooth element of $\NN$, and $n=d\cdot m$ with $m\in\NN$.

(iii) Hence $\#(\Abs\cap[u,u+h])\le\sum_{d\in\mathcal G_{2y}\cap(h,2yh]}N(I_d)$, where $\mathcal G_{2y}$ is the set of
$2y$-smooth elements of $\NN$ and $I_d:=[u/d,(u+h)/d]$ has length less than $1$.

% Logged correction 75 (CORRECTIONS-vs-v3.md): v3 covers I_d by grid windows "there"; with windows assigned to heights by
% their left end, the adjacent heights z/2 and 2z may be needed (as in Corollary cor:increment), and below X_0^- there are none.
(iv) \emph{Window count.} $I_d$ lies at height $u/d\le u^{1-\theta_b}\le2u/L^{11}$. Below $\Xm$ it holds at most one element
of $\NN$, a rational integer, since $|I_d|<1$. The rest of $I_d$ is covered by at most $L^6/4+5$ grid windows, at the
dyadic height $z$ of its left end, at $2z$, and at $z/2$ if $z\ge X_0$, since grid windows belong to the height of their left end (as in
the proof of Corollary~\ref{cor:increment}); these heights are at most $2\max I_d\le2u^{1-\theta_b}+2\le2u/L^{11}$, as
$u^{\theta_b}\ge e^{L/8}\ge2L^{11}$. Let $x'$ be the dyadic scale of $u$. The event E1 at scale $x'$, namely $\qEcl^{(x')}$, has time
$8x'/L_{x'}^{11}+H(x')\le8.0001u/L^{11}+H(u)<t_u$, so $\Good(<t_u)\subset\qEcl^{(x')}$; and it covers every height up to
$4x'/L_{x'}^{11}\ge2u/L^{11}$, hence all these heights. On it each of these windows holds at most $M_1(2z)\le M_1(u)$
elements, so $N(I_d)\le1+(L^6/4+5)M_1(u)\le C_d$. Lemma~\ref{lem:cm} at heights $\le2y$, used in (v), also holds on $\Good(<t_u)$, as
$2y\le u^{1/100}\le0.9t_u$ (by \eqref{eq:twelvetwo}, $\log2y\le\theta_bL/100$; Remark~\ref{rem:horizon}). So the bound holds on
$\Good(<t)$ for every $t\ge t_u$.
% Logged correction 73: "2y < t_u" is not enough on Good(<t_u); 2y <= 0.9 t_u puts it below the certified horizon.
% Logged correction 73 (R17, C73-4): the bound 2y <= u^{1/100} is taken from eq:twelvetwo.
% Logged correction 66: v3 argues on Good(<u) only; the statement and its uses need Good(<t_u).
% v3's aside "(Alternatively, use Corollary 4.2 at height u/d.)" is deleted: it was never checked against (12.2).

(v) \emph{Rankin's bound for the realised system.} Let $Y\ge2y$, and put $u_*:=\log Y/\log(2y)$ and
$\eta:=\log u_*/\log(2y)$. Suppose
\[
u_*\ge2\Lambda,\qquad\log\log2y\le1.79u_*,\qquad\eta\le0.31.\tag{$*$}
\]
Then
\[
\Psi_{\NN}(Y,2y)/Y\le\exp\big(-u_*\log u_*+2.3u_*/\log u_*+\log\log2y\big)\le\exp(-0.9u_*\log u_*).
\]
% Logged correction 65: v3 states this for every Y >= h; for large Y the Rankin exponent 1-eta goes to 0 or below
% (with Y = z^U, z = 2y, U = z^2 the claim would give Psi/Y <= Y^{-1.8} < 1/Y). It now carries (*), and the proof is explicit.
\begin{itemize}
\item Rankin's bound with $\sigma:=1-\eta\ge0.69$ gives $\Psi_{\NN}(Y,2y)\le Y^{\sigma}\prod_p(1-p^{-\sigma})^{-1}$, over the
  realised primes $p<2y$, and $Y^{\sigma}=Y\exp(-u_*\log u_*)$.
\item The realised primes satisfy $\pi(s)\le2s/\log s$ for all $s\ge2$. Below $\Xm$ they are the rational primes, with
  $\pi(s)\le1.26s/\log s$ \cite[(3.6)]{RosserSchoenfeld1962}. Above, $\pi(s)\le\vartheta(s)/(0.9\log s)+\pi(s^{0.9})\le1.578s/\log s+2.05s^{0.9}\le2s/\log s$, since
  $\vartheta(t)\le\psi(t)\le1.42t$ and $\pi(t)\le\psi(t)/\log2\le2.05t$ for all $t$ (Lemma~\ref{lem:cm}(i)), and $s\ge\Xm$.
\item The logarithm of the product is $\sum_pp^{-\sigma}+R_{\NN}(\sigma)$, where $R_{\NN}(\sigma)\le R(0.69)+0.01\le0.78$: the
  rational primes contribute at most $R(\sigma)\le R(0.69)$ (Lemma~\ref{lem:rowweights}, proof of (c)), and the jittered
  primes, which are at least $\Xm$, contribute at most $\sum_{p\ge\Xm}2p^{-2\sigma}\le0.01$.
\item $\sum_{p<2y}1/p\le\log\log2y+4.1$, by Lemma~\ref{lem:cm}(iv) with $y=2$. The partial summation of
  Lemma~\ref{lem:rowweights}(c), with $2s/\log s$ in place of $1.26s/\log s$, gives
  \[
  \sum_{p<2y}\frac{p^{\eta}-1}p\le2\Big(\frac{\eta}{1-\eta}+\mathrm{Ein}(\log u_*)\Big)\le1+\frac{2u_*}{\log u_*}\Big(1+\frac2{\log u_*}\Big),
  \]
  since $\eta\log2y=\log u_*\ge\log(2\cdot10^8)\ge19.11$.
\item Altogether
  \[
  \log(\Psi/Y)\le-u_*\log u_*+\log\log2y+5.9+2.22\,u_*/\log u_*,
  \]
  and $5.9\le0.08u_*/\log u_*$. This is the first bound. The second needs $0.1u_*\log u_*\ge2.3u_*/\log u_*+\log\log2y$; since $0.1\log u_*-2.3/\log u_*\ge1.79$, it
  follows from $(*)$.
\end{itemize}
In step (vi), $Y=2yh$, so $u_*=1+u_0$. Wherever $\Abs$ meets $[u,2u]$, $(*)$ holds there:
\begin{itemize}
\item $u_*\ge u_0\ge2\Lambda$, by Step 5 of the proof of Theorem~\ref{thm:caplw} ($u_0\ge\theta_b\Lambda e^s/(3.01\gamma)\ge5\Lambda$
  on the raised range, and $u_0\ge e^\ell/(97\ell^2)$ beyond it);
\item the same step gives $\log2y\le\max(22.6\Lambda,12.1\ell^2)$, so $\log\log2y\le1.79u_*$;
\item $\log2y\ge\max(\Lambda-1,7.9\ell^2)\ge\ell\sqrt{7.9(\Lambda-1)}$, since $y\ge\Xm$ (as $\Abs\ne\emptyset$) and $y\ge H\ge\qHstd$;
  and $u_*\le1+L/8\le L$. So $\eta\le\ell/\log2y\le(7.9(\Lambda-1))^{-1/2}\le10^{-4}$.
\end{itemize}

(vi) So $\#(\Abs\cap[u,u+h])\le C_d\Psi_{\NN}(2yh,2y)\le C_d\cdot2yh\cdot\exp(-0.9u_0\log u_0)\le hL^{-4}$, by (v) with
$u_*\ge u_0$ and by \eqref{eq:twelvetwo}.
\end{proof}

No Diophantine input is used: rigid runs are tamed in step (iv), where every count is moved into windows of length less
than $1$.

  below $X_0$.

\item The source perturbation $\sigma_{\Abs}:=K_{\Abs}\tilde\rho-\rho_{\Abs}$ of Section~\ref{sec:noise} satisfies
  $|\sigma_{\Abs}|\le2L^{-4}$ pointwise. Its compensator form uses $u=L/\log2y\ge10^8$ wherever $\Abs\ne\emptyset$, the raised

  \[
  o_{\Abs}(u)\le2\sup|\sigma_{\Abs}|\le4L^{-4}

\end{enumerate}

\begin{proof}
(a) is Lemma~\ref{lem:lemmaP}. (b) Consider the kernel at height $v$, with $L:=\log v$. $K_{\Abs}$ is the part of
$K_{\mathrm{pt}}$ whose landing is absorbed, that is, whose latest prime is less than $y(v)$. So its cofactors $w$ are
$2y(v)$-smooth (their primes are decided before the latest prime, as in step (i) of Lemma~\ref{lem:lemmaP}), with
$w\ge v/y(v)$. With the \emph{sure} bound $\tilde\rho\le1.01$ (at most one prime per continuous slot),
$K_{\Abs}\tilde\rho\le1.01\sum1/w$ over such $w$. Rankin's bound with exponent $1-\beta$, $\beta:=1/\log2y(v)$, gives
\[
\sum_{w\ge v/y(v)}\frac1w\le\Big(\frac v{y(v)}\Big)^{-\beta}\prod_{p<2y(v)}\big(1-p^{\beta-1}\big)^{-1}\le e^{8.2}\log(2y(v))\,e^{-u},
\qquad u:=\frac L{\log2y(v)},
\]
where the product, over the realised primes below $2y(v)$ (decided long before), is at most $e^{7.2}\log2y(v)$ as in the
proof of Lemma~\ref{lem:rankin-crude} (Lemma~\ref{lem:cm}(iii),(iv)), and $(v/y(v))^{-\beta}\le e^{1-u}$. Wherever $K_{\Abs}\ne0$ (so $s>s_H$),
$u\ge L/\max(22.6\Lambda,12.1\ell^2)\ge4\ell+\log\log2y(v)+9$, and this gives $K_{\Abs}\tilde\rho\le L^{-4}$.
% Logged correction 66: the Rankin exponent and the use of Lemma cm(iii),(iv) are now stated.
Together with (a) this gives $|\sigma_{\Abs}|\le2L^{-4}$. No local count at the current height is
used. (c) An oscillation is at most twice the supremum.
\end{proof}

\begin{proposition}[Small jittered latest primes are absorbed]\label{prop:S1S2}
% v3: Proposition 4.20
\begin{enumerate}[label=\textup{(S\arabic*)}]
\item (The small-$q$ truncation in the noise bounds of Lemma~\ref{lem:noise}.) The case of latest primes $q<y$ with jittered
  cofactors needs $\log n<L_*$. Since $L_*=10^7<\Lambda$, it is vacuous. Jittered latest primes $q<y(n)$ (all are $\ge\Xm$)
  are absorbed. A rational latest prime, even one in $[\Xm,X_0)$, belongs to a rigid composite, which sits on its own jump of
  $T$ and drops out of the excess (Lemma~\ref{lem:excess}).
% Logged correction 83 (CORRECTIONS-vs-v3.md): v3 has "latest primes q in [X_0^-, y) are absorbed", false for rational q in
% [X_0^-, X_0); only jittered latest primes are meant.
\item (The full blocks of Theorem~\ref{thm:queue} at large heights.) The arrivals that would break \eqref{eq:fullblock} are rigid runs
  $r\cdot r'\cdot d$ driven by small jittered latest primes. These have latest prime less than $y(n)$, so they lie in $\Abs$.
  A non-absorbed composite $n$ in a window $I\in\mathcal I_x$ whose latest prime $q$ is jittered (the only ones counted in
  $\qCjit$) has $q\ge y(n)\ge y_-(x)$; for $n<x$ (the windows of
  $\mathcal I_x$ may begin below $x$) this is only $q\ge y(x/2)$. Corollary~\ref{cor:increment}, with $u=x$, covers these
  windows and these primes, and \eqref{eq:fullblock} holds as shown in Theorem~\ref{thm:queue}.
% Logged correction 79 (CORRECTIONS-vs-v3.md): v3 has "non-absorbed arrivals have latest prime q >= y"; in a window crossing x
% the arrivals below x have q >= y(x/2) only.
\end{enumerate}
In particular no small-latest-prime event appears in the budget (Definition~\ref{def:good}).
\end{proposition}

\begin{proof}
This is contained in the statement: (S2) combines the definition of $\Abs$ with Corollary~\ref{cor:increment} and the proof of
Theorem~\ref{thm:queue}(ii); (S1) is the comparison $L_*<\Lambda$.
\end{proof}

\begin{lemma}[Raised threshold]\label{lem:raisedy}
% v3: Lemma 4.21
On the raised range of Theorem~\ref{thm:caplw}, where $\gamma(s)\Lambda>4\ell^2-\log4$, we have
$y(n)\ge y_{\mathrm{std}}(n):=\qHstd(x)L_{0,\mathrm{std}}(x)L(x)^6$ (with $x$ the dyadic scale of $n$), since $H$, $M_1$ and $L_0$ are all at least their
standard values. So $\Abs_{\mathrm{std}}\subset\Abs$: raising $y$ only \emph{adds} to $\Abs$ the composites whose latest
jittered prime $q$ lies in $[y_{\mathrm{std}}(n),y(n))$. The consumers of $\Abs$ are unaffected:
\begin{itemize}
% Logged correction 79 (CORRECTIONS-vs-v3.md): the threshold seen by Theorem queue is y_-(x) = y(x/2) on windows that begin below x.
\item Theorem~\ref{thm:queue} and the noise bounds (Lemma~\ref{lem:noise}) see only jittered latest primes $q\ge y_-(u)$ of
  Corollary~\ref{cor:increment} ($y(u/2)$ on the windows of Theorem~\ref{thm:queue} that begin below $u$, and $y(u)$ in
  Lemma~\ref{lem:noise}). Corollary~\ref{cor:increment} needs $|I/q|\le2H(u)/y_-(u)\le2.00002L_u^{-11}$, which holds because $y$
  uses the same raised $H$, and $L_0\ge L^5$. No consumer assumes $y=y_{\mathrm{std}}$.
  Jittered latest primes in $[y_{\mathrm{std}},y)$ are absorbed, not repaired; a rational latest prime there belongs to a rigid
  element, which drops out of the excess (Lemma~\ref{lem:excess}).
\item Lemma~\ref{lem:S}: its ``$y$'' is the start of the operator, not the absorption threshold. Its bound uses Mertens' bound
  over all of $\NN$, so it is independent of $y$. Its threshold $H(v)\ge(\log v)^{8.5+\delta}$ only gets easier as $H$ grows.
\item Proposition~\ref{prop:pointwise}: its only $\Abs$-input is Lemma~\ref{lem:rhoA}(c), which does not involve $y$ and reads only heights
  below $u$.
\item Lemma~\ref{lem:lemmaP}: \eqref{eq:twelvetwo} holds by Theorem~\ref{thm:caplw}(b). It is read with $y:=\sup_{[u,2u]}y$. Since $y$
  is frozen per scale, this sup involves at most two scales, and along the ramp $y(2x)/y(x)\le2^{10^{-5}}(1+2c^{-1/2})$ by
  Theorem~\ref{thm:caplw}(c) and $M_1\le\bar M_1\le M_1(1+2c^{-1/2})$. So this changes nothing numerically.
  % Logged correction 32: v3 has "along the ramp d log y/d log u <= 1e-5"; y is frozen per scale.
% Logged correction 68: explicit exponents (v3: y = X_0^{O(1)} <= n^{o(1)} and B << x^theta).
\item Lemma~\ref{lem:ZA}: the bound on the absorbed part holds for $y\le X_0^{22.6}L^9\le n^{0.01}$
  (Remark~\ref{rem:caplw-sizes}).
\item Queue size and \eqref{eq:La}: $B\le5M_1L^2/c'\le X_0^{7.6}L^2$ enters additively or as $x^{\eta}$, and
  $B\le x^{10^{-5}}\le x^{\eta}$ at these heights, since $\eta\ge5\cdot10^{-5}$ and $\log B\le\log c+2\ell+4.3$: for
  $s>s_H$, $\log c\le7.5\Lambda$ and $\log x\ge393\Lambda^2$; for $s\le s_H$, $\log c=\gamma_*\Lambda\le0.03\Lambda e^{s/2}
  =0.03\sqrt L$ (Step 6 of the proof of Theorem~\ref{thm:caplw}) and $L\ge10^8$.
% Logged correction 68 (check R12): the raised range also has s <= s_H, where log x >= 393 Lambda^2 fails; handled separately.
\item Causality: $y(n)=y(x)$ is a deterministic function of the dyadic scale $x$ of $n$, so membership in $\Abs$ is decided at the decision
  of the latest prime.
\end{itemize}
\end{lemma}

\begin{proof}
Each item is a direct check against the cited statement, as indicated.
\end{proof}

\paragraph{Non-circularity.}
\begin{itemize}
\item Theorem~\ref{thm:queue} at slot $s_j$ reads only events with times at most $s_j$: $\qEcl$ at cofactor heights at most
  $4u/L^{11}$, and Corollary~\ref{cor:r1} and Proposition~\ref{prop:pointwise}.
\item $\qEcl$ reads no queue bound at its own heights. Its inputs are (F1)--(F4) at heights at most $s-H$, carried by
  progressive events.
\item Corollary~\ref{cor:localcount} is used only at strictly smaller heights.
\item Lemma~\ref{lem:lemmaP} uses $\qEcl$ only at heights $u/d<u$.
\item The first-failure argument (Lemma~\ref{lem:budget}) orders all of these by event time.
\end{itemize}
